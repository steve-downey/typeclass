36// A generic lambda-expression is a distinct, unrelated type at every
37// occurrence in the source, so it cannot be the shared expression a
38// disjunctive requires-clause needs at both the in-class declaration and
39// the out-of-line definition (they must be spelled with the same tokens
40// naming the same entity). discard_first_eval_t and discard_second_eval_t
41// give `discard_first` and `discard_second` a named, stable callable for
42// that purpose -- the same role applicative_eval_t plays for `ap`.
43
45struct discard_first_eval_t {
46 template <class FIRST, class SECOND>
47 constexpr auto operator()(const FIRST &, SECOND &&second) const
48 -> SECOND && {
49 return std::forward<SECOND>(second);
50 }
51};
52
54struct discard_second_eval_t {
55 template <class FIRST, class SECOND>
56 constexpr auto operator()(FIRST &&first, const SECOND &) const -> FIRST && {
57 return std::forward<FIRST>(first);
58 }
59};

118template <class IMPL, class CONTEXT>
119concept applicative_impl =
120 requires(const IMPL &impl) {
121 impl.pure(std::declval<applicative_value_t<CONTEXT>>());
122 } &&
123 (requires(const IMPL &impl, const CONTEXT &context) {
124 impl.invoke(detail::probe_witness<applicative_value_t<CONTEXT>>{}, context);
125 } || requires(const IMPL &impl, const CONTEXT &context) {
126 impl.ap(impl.pure(detail::probe_witness<applicative_value_t<CONTEXT>>{}),
127 context);
128 });

280 constexpr auto ap_chain(this auto &&self, ACCUMULATED &&accumulated,
281 NEXT_ARGUMENT &&next_argument,
282 REST_ARGUMENTS &&...rest_arguments) {
283 auto next = self.ap(std::forward<ACCUMULATED>(accumulated),
284 std::forward<NEXT_ARGUMENT>(next_argument));
285 if constexpr (sizeof...(REST_ARGUMENTS) == 0) {

319// -- Operation objects --
320//
321// See functor.hpp for the shape and the motivation. Applicative is where the
322// deducing form meets its one structural limit: `pure` has no in-context
323// argument, so nothing at the call site names the context to look up. It is
324// spelled as a variable template of operation objects, taking the context --
325// `pure<std::optional<int>>(42)` -- and it is the only operation in this
326// header that does not deduce.

350 REST_ARGUMENTS &&...rest_arguments) const {
351 return TC.invoke(std::forward<FUNCTION>(function),
352 std::forward<FIRST_ARGUMENT>(first_argument),
353 std::forward<REST_ARGUMENTS>(rest_arguments)...);
354 }

482template <class OBJ, class CONTEXT>
483concept applicative_object =
484 requires(const OBJ &obj, const CONTEXT &context) {
485 obj.pure(std::declval<applicative_value_t<CONTEXT>>());
486 obj.invoke(detail::probe_witness<applicative_value_t<CONTEXT>>{}, context);
487 obj.map(detail::probe_witness<applicative_value_t<CONTEXT>>{}, context);
488 obj.lift(std::declval<applicative_value_t<CONTEXT>>());
489 obj.zip_with(detail::probe_witness2<applicative_value_t<CONTEXT>>{}, context,
490 context);
491 obj.discard_first(std::declval<CONTEXT>(), std::declval<CONTEXT>());
492 obj.discard_second(std::declval<CONTEXT>(), std::declval<CONTEXT>());
493 obj.invoke_with(obj, detail::probe_witness<applicative_value_t<CONTEXT>>{},
494 context);
495 } &&
496 (!requires(const OBJ &obj) {
497 obj.pure(detail::probe_witness<applicative_value_t<CONTEXT>>{});
498 } || requires(const OBJ &obj, const CONTEXT &context) {
499 obj.ap(obj.pure(detail::probe_witness<applicative_value_t<CONTEXT>>{}),
500 context);
501 }) &&
502 (!graded_context<CONTEXT> ||
503 requires(const OBJ &obj, const CONTEXT &context) {
504 obj.template subsume<grade_of_t<CONTEXT>>(context);
505 });

511template <class VALUE_TYPE>
512struct optional_applicative_impl {
513 // \ref{transpose.applicative.optional}, applicative instance for optional
514 template <class VALUE>
515 constexpr auto pure(this auto &&, VALUE &&value)
516 -> std::optional<std::remove_cvref_t<VALUE>>;
517
518 template <class FUNCTION, class FIRST, class... REST>
519 requires detail::is_optional_v<FIRST> &&
520 (detail::is_optional_v<REST> && ...) &&
521 std::invocable<FUNCTION &, detail::contained_ref_t<FIRST>,
522 detail::contained_ref_t<REST>...> &&

527 REST &&...rest)
528 -> std::optional<std::remove_cvref_t<
529 std::invoke_result_t<FUNCTION &, detail::contained_ref_t<FIRST>,
530 detail::contained_ref_t<REST>...>>>;
531};

563template <class Impl>
564template <class FUNCTION, class FIRST_ARGUMENT, class... REST_ARGUMENTS>
565constexpr auto
566derive_applicative<Impl>::invoke(this auto &&self, FUNCTION &&function,
567 FIRST_ARGUMENT &&first_argument,
568 REST_ARGUMENTS &&...rest_arguments)
569 requires requires(const Impl &impl) {
570 impl.invoke(std::forward<FUNCTION>(function),
571 std::forward<FIRST_ARGUMENT>(first_argument),
572 std::forward<REST_ARGUMENTS>(rest_arguments)...);
573 } || detail::invocable_through_ap<decltype(self), FUNCTION, FIRST_ARGUMENT,
574 REST_ARGUMENTS...>
575{
576 if constexpr (requires {
577 impl_of(self).invoke(
578 std::forward<FUNCTION>(function),
579 std::forward<FIRST_ARGUMENT>(first_argument),
580 std::forward<REST_ARGUMENTS>(rest_arguments)...);
581 }) {
582 return impl_of(self).invoke(
583 std::forward<FUNCTION>(function),
584 std::forward<FIRST_ARGUMENT>(first_argument),
585 std::forward<REST_ARGUMENTS>(rest_arguments)...);
586 } else {
587 // Derivation from the ap basis:
588 // invoke(f, x1, ..., xn) = pure(curried f) `ap` x1 `ap` ... xn
589 // This static_assert is provably redundant: the declaration's
590 // constraint already established that the ap chain is well-formed,
591 // which needs an ap basis. It is the GHC-MINIMAL-style last-resort
592 // message fold_map also keeps, not load-bearing SFINAE.
593 static_assert(
594 applicative_impl<Impl, std::remove_cvref_t<FIRST_ARGUMENT>>,
595 "Applicative Impl must provide pure and at least one basis: "
596 "invoke(f, args_in_context...) or "
597 "ap(f_in_context, arg_in_context).");
598 auto lifted_function = self.pure(
599 detail::make_terminating_partial<1 + sizeof...(REST_ARGUMENTS)>(
600 std::forward<FUNCTION>(function)));
601 return self.ap_chain(
602 self.ap(std::move(lifted_function),
603 std::forward<FIRST_ARGUMENT>(first_argument)),
604 std::forward<REST_ARGUMENTS>(rest_arguments)...);
605 }
606}
