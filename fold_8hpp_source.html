27// The two fold programs fold_left and fold_right derive through are
28// endomorphisms `STATE -> STATE`, type-erased through std::function so that
29// combine returns what it takes: a monoid's value type must close under
30// combine, and a bare lambda type does not. One value type, two instances
31// -- composition in traversal order for fold_left, in reverse for
32// fold_right -- which is docs/transpose/decisions.md#monoid-selection in
33// miniature: the choice between them is made at the operation, not by wrapping
34// the function in a differently named carrier.
35
40template <class STATE>
41struct left_compose_monoid {
42 using value_type = std::function<STATE(STATE)>;
43
44 auto identity() const -> value_type {
45 return [](STATE s) { return s; };
46 }
47
48 auto combine(const value_type &lhs, const value_type &rhs) const
49 -> value_type {
50 return [lhs, rhs](STATE s) { return rhs(lhs(std::move(s))); };
51 }
52};
53
57template <class STATE>
58struct right_compose_monoid {
59 using value_type = std::function<STATE(STATE)>;
60
61 auto identity() const -> value_type {
62 return [](STATE s) { return s; };
63 }
64
65 auto combine(const value_type &lhs, const value_type &rhs) const
66 -> value_type {
67 return [lhs, rhs](STATE s) { return lhs(rhs(std::move(s))); };
68 }
69};

81// Named witnesses for the fold family's availability probes, completing
82// probe_witness/probe_witness2 with the argument-dependent return shapes
83// the derived operations need. A lambda inside a requires-clause mints a
84// distinct closure type at every constraint check; MSVC's backend ICEs
85// (fatal error C1001, p2) in exactly the translation units that evaluate
86// the fold family's clauses, and hoisting the probes into named callables
87// is both the workaround and consistent with the concepts, which already
88// probe with named witnesses.
89//
90// Both are constrained on `copy_constructible`. Unlike probe_witness,
91// whose body establishes a result type and nothing else, these two stand in
92// for derivations that really do copy the element -- `combine_all` folds
93// with the identity function, `to_vector` collects each element into a
94// vector. (`find_first`, which carries a matching element out in an
95// optional, probes with find_first_probe_witness below.) That requirement
96// has to be stated rather than discovered in a body: probing an operation
97// with a deduced return type instantiates enough of it to instantiate the
98// witness call, so an unstated copy would reach the compiler as a
99// diagnostic from inside the probe, and a concept that diagnoses cannot be
100// used to detect anything. Stated, the probe answers `false` for an element
101// type that cannot be copied -- which is the true answer, since the
102// derivation could not have run.
103
108struct identity_probe_witness {
109 template <class ARGUMENT>
110 requires std::copy_constructible<ARGUMENT>
111 constexpr auto operator()(const ARGUMENT &argument) const -> ARGUMENT {
112 return argument;
113 }
114};
115
119struct vector_probe_witness {
120 template <class ARGUMENT>
121 requires std::copy_constructible<ARGUMENT>
122 constexpr auto operator()(const ARGUMENT &argument) const
123 -> std::vector<ARGUMENT> {
124 return std::vector<ARGUMENT>{argument};
125 }
126};

128// Witnesses that carry the caller's callable. The witnesses above stand in
129// for callables the derivation writes itself, so any callable of the right
130// shape will do. A derivation that applies the CALLER's callable to each
131// element has to ask whether that callable can take the element, and only
132// the fold knows the element's type: the probed fold_map deduces it when it
133// calls the witness. So each of these is constrained on exactly what its
134// derivation's lambda does with the caller's callable, and a fold_map that
135// cannot call it -- the shipped instances spell their result through
136// invoke_result_t on the callable -- fails the probe softly instead of
137// accepting a callable that the body then cannot use. Like probe_witness,
138// each body establishes only the result type.
139
144template <class F, class MONOID>
145struct fold_map_step_probe_witness {
146 template <class ELEMENT, class RESULT>
147 requires std::invocable<F &, const ELEMENT &> &&
148 requires(MONOID &monoid, RESULT acc) {
149 {
150 monoid.combine(
151 std::invoke(std::declval<F &>(),
152 std::declval<const ELEMENT &>()),
153 std::move(acc))
154 } -> std::convertible_to<RESULT>;
155 }
156 constexpr auto operator()(const ELEMENT &, const RESULT &) const -> RESULT {
157 std::unreachable();
158 }
159};

232// Foldable pattern invariants:
233// - Generic entry point is fold_map(F, T, MONOID) via foldable<T>.
234// The monoid is an instance object over the result type of F, passed by
235// value (docs/transpose/decisions.md#monoid-selection); the two-argument
236// fold_map(F, T) is derived, defaulting to the registered Monoid of the
237// result type, and exists only where the Impl declares `element_type`
238// and that type has a registration.
239// - Instances provide an object with fold_map(F, T, MONOID) and specialize
240// foldable<Concrete>. The instance never chooses a monoid.
241// - Derived APIs (length, fold_left, fold_right, combine_all, any_of, all_of,
242// empty, to_vector, find_first) live on the same looked-up object and pass
243// the library's own instances over bare types.
244// - Traversal order is instance-defined but must be coherent per instance.
245
254template <class IMPL, class STRUCTURE>
255concept foldable_impl = requires(const IMPL &impl, const STRUCTURE &structure) {
256 impl.fold_map(detail::probe_witness<std::size_t>{}, structure,
257 sum_monoid<std::size_t>{});
258} || requires(const IMPL &impl, const STRUCTURE &structure) {
259 typename IMPL::element_type;
260 impl.fold_right(structure, int{}, detail::probe_witness2<int>{});
261};

271template <class Impl>
272struct derive_foldable : protected Impl {
273 static_assert(!std::is_same_v<Impl, std::false_type>,
274 "No foldable<T> specialization found. "
275 "Specialize beman::transpose::foldable<T> for your "
276 "type T and provide fold_map(F, T, MONOID) or "
277 "fold_right(T, STATE, F) + element_type.");
278 // Alternate-core: Impl provides either fold_map or fold_right as
279 // primitive. Haskell equivalent: {-# MINIMAL foldMap | foldr #-}
280 //
281 // fold_map and fold_right are a mutually-derivable pair, done the
282 // apply.hpp way: each probes Impl for its own native version first, and
283 // each derivation addresses Impl directly (impl_of(self), never self)
284 // for the other operation. Before this conversion, both derivations were
285 // self-routed, and the only thing standing between that and unbounded
286 // mutual template recursion -- fold_map deriving from fold_right
287 // deriving from fold_map, each round nesting another fold program --
288 // was a Map's using-declaration happening to shadow one side. A Map that
289 // omitted it fell into the cycle. Addressing Impl directly closes that
290 // structurally: neither derivation can re-enter the other through self,
291 // because self is never named. The Maps carry no using-declaration for
292 // fold_map at all; the base decides, per instantiation, from whichever
293 // basis Impl actually provides.
294
295 // fold_map with an explicit monoid instance: the basis. Native
296 // Impl::fold_map preferred; derived from fold_right otherwise.
297 // foldMap f = foldr (\x acc -> f x <> acc) mempty
298 template <class F, class T, class MONOID>
299 constexpr auto fold_map(this auto &&self, F &&function, T &&value,
300 MONOID monoid)
301 requires requires(const Impl &impl) {
302 impl.fold_map(std::forward<F>(function), std::forward<T>(value),
303 monoid);
304 } || requires(const Impl &impl) {
305 // The derivation applies function to each element and combines
306 // the result into the accumulator. That is checked here against
307 // element_type, which this alternative requires anyway, so the
308 // answer does not depend on whether Impl's fold_right checks the
309 // callable it is given; the witness asks the same of each
310 // element as fold_right will actually pass it.
311 typename Impl::element_type;
312 requires std::invocable<F &, const typename Impl::element_type &>;
313 {
314 monoid.combine(
315 std::invoke(
316 function,
317 std::declval<const typename Impl::element_type &>()),
318 monoid.identity())
319 } -> std::convertible_to<monoid_value_t<MONOID>>;
320 impl.fold_right(std::forward<T>(value), monoid.identity(),
321 detail::fold_map_step_probe_witness<F, MONOID>{});
322 }
323 {
324 if constexpr (requires {
325 impl_of(self).fold_map(std::forward<F>(function),
326 std::forward<T>(value),
327 monoid);
328 }) {
329 return impl_of(self).fold_map(std::forward<F>(function),
330 std::forward<T>(value), monoid);
331 } else {
332 using Result = monoid_value_t<MONOID>;
333 // The declaration's second alternative probes with a named
334 // witness, never a lambda: a capturing lambda in a
335 // requires-clause is rejected by Clang's front end, and the
336 // closure types lambdas mint per constraint check are what ICE
337 // MSVC's backend (see the witnesses' note in detail). This
338 // static_assert is provable redundant given the declaration's
339 // constraint already selected this branch -- it is the
340 // GHC-MINIMAL-style last-resort message, not load-bearing SFINAE
341 // -- so the concept-based condition is exactly as informative.
342 static_assert(foldable_impl<Impl, std::remove_cvref_t<T>>,
343 "Foldable Impl must provide at least one basis: "
344 "fold_map(f, container, monoid), or "
345 "fold_right(container, state, f) plus element_type.");
346 return impl_of(self).fold_right(
347 std::forward<T>(value), monoid.identity(),
348 [&function, &monoid](const auto &elem, Result acc) {
349 return monoid.combine(std::invoke(function, elem),
350 std::move(acc));
351 });
352 }
353 }
354
355 // fold_map without an instance: defaults to the registered Monoid of
356 // the result type. Available only where Impl declares element_type (so
357 // the result type can be named) and that type has a registration. A
358 // fold over a type with several monoids -- int, bool -- is therefore
359 // not a candidate here; the caller passes the instance. An Impl may
360 // still supply a native two-argument fold_map of its own.
361 template <class F, class T>
362 constexpr auto fold_map(this auto &&self, F &&function, T &&value)
363 requires requires(const Impl &impl) {
364 impl.fold_map(std::forward<F>(function), std::forward<T>(value));
365 } || requires {
366 typename Impl::element_type;
367 requires detail::has_registered_monoid<
368 detail::fold_result_t<Impl, F>>;
369 self.fold_map(std::forward<F>(function), std::forward<T>(value),
370 Monoid<detail::fold_result_t<Impl, F>>{});
371 }
372 {
373 if constexpr (requires {
374 impl_of(self).fold_map(std::forward<F>(function),
375 std::forward<T>(value));
376 }) {
377 return impl_of(self).fold_map(std::forward<F>(function),
378 std::forward<T>(value));
379 } else {
380 return self.fold_map(std::forward<F>(function),
381 std::forward<T>(value),
382 Monoid<detail::fold_result_t<Impl, F>>{});
383 }
384 }

387 template <class T>
388 constexpr auto length(this auto &&self, T &&value) -> std::size_t
389 requires requires(const Impl &impl) {
390 impl.length(std::forward<T>(value));
391 } || requires {
392 // self, not impl: fold_map is itself now a probing member with
393 // its own either-basis constraint, so checking availability
394 // through self is what correctly admits a fold_right +
395 // element_type Impl. Checking impl.fold_map directly would
396 // reject that Impl, since it has no fold_map member at all.
397 self.fold_map(detail::probe_witness<std::size_t>{},
398 std::forward<T>(value), sum_monoid<std::size_t>{});
399 }
400 {
401 if constexpr (requires {
402 impl_of(self).length(std::forward<T>(value));
403 }) {
404 return impl_of(self).length(std::forward<T>(value));
405 } else {
406 return self.fold_map([](const auto &) { return std::size_t{1}; },
407 std::forward<T>(value),
408 sum_monoid<std::size_t>{});
409 }
410 }

415 template <class T, class STATE, class F>
416 constexpr auto fold_left(this auto &&self, T &&value, STATE initial_state,
417 F &&function)
418 requires requires(const Impl &impl) {
419 impl.fold_left(std::forward<T>(value), std::move(initial_state),
420 std::forward<F>(function));
421 } || requires {
422 // self, not impl -- see length's second alternative. A named
423 // witness, not a lambda: it carries the step's type, so the
424 // probed fold_map asks whether the step can take each element
425 // as the derivation will pass it.
426 requires std::constructible_from<std::remove_cvref_t<F>, F>;
427 self.fold_map(
428 detail::fold_left_probe_witness<std::remove_cvref_t<F>,
429 std::remove_cvref_t<STATE>>{},
430 std::forward<T>(value),
431 detail::left_compose_monoid<std::remove_cvref_t<STATE>>{});
432 }
433 {
434 if constexpr (requires {
435 impl_of(self).fold_left(std::forward<T>(value),
436 std::move(initial_state),
437 std::forward<F>(function));
438 }) {
439 return impl_of(self).fold_left(std::forward<T>(value),
440 std::move(initial_state),
441 std::forward<F>(function));
442 } else {
443 using StateType = std::remove_cvref_t<STATE>;
444 using Program = std::function<StateType(StateType)>;
445 auto step = std::forward<F>(function);
446
447 const auto program = self.fold_map(
448 [&step](const auto &x) -> Program {
449 using ValueType = std::remove_cvref_t<decltype(x)>;
450 return [x_copy = ValueType(x), &step](StateType s) {
451 return std::invoke(step, std::move(s), x_copy);
452 };
453 },
454 std::forward<T>(value),
455 detail::left_compose_monoid<StateType>{});
456
457 return program(StateType(std::move(initial_state)));
458 }
459 }

715 constexpr auto find_first(this auto &&self, T &&value,
716 PREDICATE &&predicate)
717 requires requires(const Impl &impl) {
718 impl.find_first(std::forward<T>(value),
719 std::forward<PREDICATE>(predicate));
720 } || requires {
721 // self, not impl; see any_of's second alternative.
722 typename Impl::element_type;
723 self.fold_map(
724 detail::find_first_probe_witness<PREDICATE,
725 typename Impl::element_type>{},
726 std::forward<T>(value),
727 first_monoid<typename Impl::element_type>{});
728 }
729 {
730 if constexpr (requires {
731 impl_of(self).find_first(
732 std::forward<T>(value),
733 std::forward<PREDICATE>(predicate));
734 }) {
735 return impl_of(self).find_first(std::forward<T>(value),
736 std::forward<PREDICATE>(predicate));
737 } else {
738 using Element = typename Impl::element_type;
739 return self.fold_map(
740 [&predicate](const auto &x) -> std::optional<Element> {
741 if (std::invoke(predicate, x)) {
742 return std::optional<Element>{x};
743 }
744 return std::optional<Element>{};
745 },
746 std::forward<T>(value), first_monoid<Element>{});
747 }
748 }

754 return static_cast<impl_ref_t<Impl, SELF>>(self);

763// -- Operation objects --
764//
765// See functor.hpp for the shape, the motivation, and why every typeclass
766// operation shares one namespace. Foldable is the widest of the typeclasses
767// -- one primitive and nine derived operations -- and it supplies most of the
768// names with namespace-std counterparts: `empty`, `length`, `to_vector`,
769// `any_of` and `all_of`, of which `empty` against `std::empty` is the worst.
770//
771// `empty` stays the predicate here, which is the C++ reading (std::empty,
772// std::ranges::empty, container .empty()) and the reverse of the FP one --
773// Haskell and PureScript call this `null`, Cats calls it `isEmpty`, and all
774// three reserve `empty` for Alternative's identity. See the note in
775// functor.hpp for why the host language wins that one.
776//
777// Every operation gets one, primitive and derived alike. Which operations are
778// primitive is a fact about the instance author's obligations, and asking a
779// caller to know it before they can find `length` would be the wrong tax.
780//
781// `fold` (the alias of combine_all) gets no object: an alias does not need a
782// second name.
