6// The structural adapter between beman.fingertree and beman.tree_algorithms.
7//
8// Opt-in: neither library includes this header or links its target, and
9// neither depends on the other. Including it is what makes a FingerTree a
10// tree the recursive algorithms can see: fold_with, unfold_with and refold
11// in the tree's own representation, fold_fix and unfold_fix through its Fix
12// form, and both the explicit and the lookup fold_map.
13//
14// What the algorithms see is the REPRESENTATION, not the sequence. One layer
15// is one real node of the tree as fingertree stores it: Empty, Single or
16// Deep on the spine, Leaf, Node2 or Node3 below a digit, with digit order
17// and spine position preserved. A sequence operation is free to rebalance
18// that shape; a structural fold reports whatever shape is there. Folding the
19// elements left to right gives the sequence back, which is the law that ties
20// this adapter to fingertree's own Foldable.
21//
22// The representation has two recursive sorts -- the tree at a spine level,
23// and an element node -- and the verbs recurse through one position type.
24// FingerTreePart is that type: a finite tagged value holding one or the
25// other. It does not change with spine depth, so a fold over a tree of any
26// size instantiates the same handful of templates. A part is both the handle
27// a projection hands out and the piece an embedding assembles.
28//
29// Capability wrappers are reached through their backing tree, which is the
30// representation they have: fold_map<Result>(f, combine, identity,
31// index.backing_tree()). The payload a structural algorithm then sees is the
32// backing tree's, an Interval<T> for the interval index and not its T.
33

168template <typename P>
169inline constexpr bool is_payload_ref = false;
170template <typename T>
171inline constexpr bool is_payload_ref<FingerPayloadRef<T>> = true;

173// Whether a payload of type T can be made from a slot handed over as SLOT
174// (a P&& to consume, a const P& to keep). A shared slot is always copied
175// from; an owned one is moved or copied as it was handed over.
176template <typename SLOT, typename T>
177concept payload_source =
178 (is_payload_ref<std::remove_cvref_t<SLOT>> && std::same_as<typename std::remove_cvref_t<SLOT>::value_type, T> &&
179 std::copy_constructible<T>) ||
180 (!is_payload_ref<std::remove_cvref_t<SLOT>> && std::constructible_from<T, SLOT>);
181
182template <typename T, typename SLOT>
183constexpr auto make_payload(SLOT&& slot) -> T {
184 if constexpr (is_payload_ref<std::remove_cvref_t<SLOT>>) {
185 return T(slot.get());
186 } else {
187 return T(std::forward<SLOT>(slot));
188 }
189}

200template <typename P, typename A>
201struct FingerTreeFFunctorImpl {
202 template <typename Fn>
203 constexpr auto fmap(this auto&&, Fn&& fn, const FingerTreeF<P, A>& layer)
204 requires std::invocable<Fn&, const A&> &&
205 transpose::detail::holdable_value<std::invoke_result_t<Fn&, const A&>> &&
206 std::constructible_from<std::remove_cvref_t<std::invoke_result_t<Fn&, const A&>>,
207 std::invoke_result_t<Fn&, const A&>> &&
208 std::move_constructible<std::remove_cvref_t<std::invoke_result_t<Fn&, const A&>>> &&
209 std::copy_constructible<P>
210 {
211 using B = std::remove_cvref_t<std::invoke_result_t<Fn&, const A&>>;
212 using Layer = FingerTreeF<P, B>;
213 const auto slot = [&fn](const tree_algorithms::child_slot_t<A>& source) -> tree_algorithms::child_slot_t<B> {
214 if (!source)
215 return tree_algorithms::child_slot_t<B>{};
216 return tree_algorithms::make_slot<B>(std::invoke(fn, *source));
217 };
218 const auto digit = [&slot](const FingerDigit<A>& source) -> FingerDigit<B> {
219 FingerDigit<B> out;
220 for (const auto& position : source)
221 out.push_back(slot(position));
222 return out;
223 };
224 return std::visit(
225 transpose::overloaded{
226 [](const FingerEmpty&) -> Layer { return FingerEmpty{}; },
227 [&](const FingerSingle<A>& s) -> Layer { return FingerSingle<B>{slot(s.element)}; },
228 [&](const FingerDeep<A>& d) -> Layer {
229 return FingerDeep<B>{digit(d.left), slot(d.spine), digit(d.right)};
230 },
231 [](const FingerLeaf<P>& l) -> Layer { return FingerLeaf<P>{l.value}; },
232 [&](const FingerNode2<A>& n) -> Layer { return FingerNode2<B>{slot(n.a), slot(n.b)}; },
233 [&](const FingerNode3<A>& n) -> Layer { return FingerNode3<B>{slot(n.a), slot(n.b), slot(n.c)}; },
234 },
235 layer);
236 }
237};
238
241template <typename P, typename A>
242struct FingerTreeFFunctorMap : transpose::derive_functor<FingerTreeFFunctorImpl<P, A>> {
243 using FingerTreeFFunctorImpl<P, A>::fmap;
244};

278template <typename TREE>
279class FingerTreePart {
280 public:
281 using tree_type = TREE;
282 using level_type = fingertree::tree::FingerTreeLevel<TREE>;
283 using node_type = typename level_type::node_type;
284 using tag_type = typename level_type::tag_type;
285
287 explicit FingerTreePart(const TREE& sequence) : d_part(std::in_place_type<level_type>, sequence) {}
288
289 explicit FingerTreePart(level_type level) : d_part(std::move(level)) {}
290 explicit FingerTreePart(node_type node) : d_part(std::move(node)) {}
291
292 [[nodiscard]] auto is_level() const noexcept -> bool { return std::holds_alternative<level_type>(d_part); }
293 [[nodiscard]] auto is_node() const noexcept -> bool { return std::holds_alternative<node_type>(d_part); }
294
296 [[nodiscard]] auto level() const& noexcept -> const level_type& {
297 assert(is_level() && "FingerTreePart::level: the part is an element node");
298 return *std::get_if<level_type>(&d_part);
299 }
300 [[nodiscard]] auto level() && -> level_type {
301 assert(is_level() && "FingerTreePart::level: the part is an element node");
302 return std::move(*std::get_if<level_type>(&d_part));
303 }
304
306 [[nodiscard]] auto node() const& noexcept -> const node_type& {
307 assert(is_node() && "FingerTreePart::node: the part is a spine level");
308 return *std::get_if<node_type>(&d_part);
309 }
310 [[nodiscard]] auto node() && -> node_type {
311 assert(is_node() && "FingerTreePart::node: the part is a spine level");
312 return std::move(*std::get_if<node_type>(&d_part));
313 }
314
316 [[nodiscard]] auto measure() const -> tag_type { return is_level() ? level().measure() : node().measure(); }
317
320 [[nodiscard]] auto sequence() const& -> TREE { return level().sequence(); }
321 [[nodiscard]] auto sequence() && -> TREE { return std::move(*this).level().sequence(); }
322
323 private:
324 std::variant<level_type, node_type> d_part;
325};

338// One layer of @p part, with @p leaf_slot choosing how a Leaf holds its
339// payload. Both projections below are this function.
340template <typename P, typename TREE, typename LeafSlot>
341auto project_layer(const FingerTreePart<TREE>& part, const LeafSlot& leaf_slot)
342 -> FingerTreeF<P, FingerTreePart<TREE>> {
343 using Part = FingerTreePart<TREE>;
344 using fingertree::tree::FingerTreeForm;
345 using fingertree::tree::FingerTreeNodeForm;
346 const auto slot = [](auto&& piece) {
347 return tree_algorithms::make_slot<Part>(Part(std::forward<decltype(piece)>(piece)));
348 };
349
350 if (part.is_node()) {
351 const auto& node = part.node();
352 if (node.form() == FingerTreeNodeForm::LEAF)
353 return FingerLeaf<P>{leaf_slot(node)};
354 auto children = node.children();
355 if (node.form() == FingerTreeNodeForm::NODE2)
356 return FingerNode2<Part>{slot(std::move(children[0])), slot(std::move(children[1]))};
357 return FingerNode3<Part>{
358 slot(std::move(children[0])), slot(std::move(children[1])), slot(std::move(children[2]))};
359 }
360
361 const auto& level = part.level();
362 if (level.form() == FingerTreeForm::EMPTY)
363 return FingerEmpty{};
364 if (level.form() == FingerTreeForm::SINGLE)
365 return FingerSingle<Part>{slot(level.element())};
366
367 FingerDeep<Part> deep;
368 for (auto& node : level.left())
369 deep.left.push_back(slot(std::move(node)));
370 if (auto spine = level.spine(); !spine.empty())
371 deep.spine = slot(std::move(spine));
372 for (auto& node : level.right())
373 deep.right.push_back(slot(std::move(node)));
374 return deep;
375}
376
377} // namespace detail

431template <typename TREE>
432class FingerTreeEmbed {
433 public:
434 using tree_type = TREE;
435 using value_type = typename TREE::value_type;
436 using part_type = FingerTreePart<TREE>;
437 using builder_type = fingertree::tree::FingerTreeStructureBuilder<TREE>;
438
440 explicit FingerTreeEmbed(const TREE& prototype) : d_builder(prototype) {}
441
442 explicit FingerTreeEmbed(builder_type builder = builder_type{}) : d_builder(std::move(builder)) {}
443
444 [[nodiscard]] auto builder() const noexcept -> const builder_type& { return d_builder; }
445
448 template <typename P>
449 requires detail::payload_source<const P&, value_type>
450 auto operator()(const FingerTreeF<P, part_type>& layer) const -> part_type {
451 return embed<P>(layer);
452 }
453
455 template <typename P>
456 requires detail::payload_source<P&&, value_type>
457 auto operator()(FingerTreeF<P, part_type>&& layer) const -> part_type {
458 return embed<P>(std::move(layer));
459 }
460
461 private:
462 builder_type d_builder;
463
464 using slot_type = tree_algorithms::child_slot_t<part_type>;
465 using level_type = typename part_type::level_type;
466 using node_type = typename part_type::node_type;
467
468 template <typename SLOT>
469 static auto node(SLOT&& slot) -> node_type {
470 assert(slot && slot->is_node() && "FingerTreeEmbed: an element position does not hold an element");
471 return std::forward_like<SLOT>(*slot).node();
472 }
473
474 template <typename DIGIT>
475 static auto digit(DIGIT&& source) -> typename level_type::digit_type {
476 typename level_type::digit_type out;
477 for (auto& slot : source)
478 out.push_back(node(std::forward_like<DIGIT>(slot)));
479 return out;
480 }
481
482 // An if-chain and not a visit: the alternatives are forwarded as the
483 // layer was handed over, and transpose::overloaded cannot rank a
484 // forwarding lambda against its own catch-all.
485 template <typename P, typename LAYER>
486 auto embed(LAYER&& layer) const -> part_type {
487 if (auto* single = std::get_if<FingerSingle<part_type>>(&layer))
488 return part_type(d_builder.single(node(std::forward_like<LAYER>(single->element))));
489 if (auto* deep = std::get_if<FingerDeep<part_type>>(&layer)) {
490 auto left = digit(std::forward_like<LAYER>(deep->left));
491 auto next = d_builder.empty();
492 if (deep->spine) {
493 assert(deep->spine->is_level() && "FingerTreeEmbed: a spine position does not hold a tree");
494 next = std::forward_like<LAYER>(*deep->spine).level();
495 }
496 auto right = digit(std::forward_like<LAYER>(deep->right));
497 return part_type(d_builder.deep(std::move(left), std::move(next), std::move(right)));
498 }
499 if (auto* leaf = std::get_if<FingerLeaf<P>>(&layer))
500 return part_type(d_builder.leaf(detail::make_payload<value_type>(std::forward_like<LAYER>(leaf->value))));
501 if (auto* node2 = std::get_if<FingerNode2<part_type>>(&layer))
502 return part_type(
503 d_builder.node2(node(std::forward_like<LAYER>(node2->a)), node(std::forward_like<LAYER>(node2->b))));
504 if (auto* node3 = std::get_if<FingerNode3<part_type>>(&layer))
505 return part_type(d_builder.node3(node(std::forward_like<LAYER>(node3->a)),
506 node(std::forward_like<LAYER>(node3->b)),
507 node(std::forward_like<LAYER>(node3->c))));
508 return part_type(d_builder.empty());
509 }
510};

521struct FingerTreeLayerFoldMap {
522 template <typename MapFn, typename Combine, typename Result, typename P>
523 constexpr auto operator()(const MapFn& map_fn,
524 const Combine& combine,
525 const Result& identity,
526 const FingerTreeF<P, Result>& layer) const -> Result {
527 return std::visit(
528 transpose::overloaded{
529 [&](const FingerEmpty&) -> Result { return identity; },
530 [&](const FingerSingle<Result>& s) -> Result { return *s.element; },
531 [&](const FingerDeep<Result>& d) -> Result {
532 std::optional<Result> acc;
533 const auto step = [&](const Result& next) {
534 if (acc)
535 acc.emplace(combine(*acc, next));
536 else
537 acc.emplace(next);
538 };
539 for (const auto& position : d.left)
540 step(*position);
541 if (d.spine)
542 step(*d.spine);
543 for (const auto& position : d.right)
544 step(*position);
545 return acc ? std::move(*acc) : identity;
546 },
547 [&](const FingerLeaf<P>& l) -> Result { return map_fn(finger_payload(l.value)); },
548 [&](const FingerNode2<Result>& n) -> Result { return combine(*n.a, *n.b); },
549 [&](const FingerNode3<Result>& n) -> Result { return combine(combine(*n.a, *n.b), *n.c); },
550 },
551 layer);
552 }
553};
