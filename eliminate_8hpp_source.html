6// Elimination of a one-of sum with heterogeneous arms.
7//
8// `std::visit` requires every arm to return one type, so an algebra over a
9// variant-shaped layer that rebuilds the layer at a new element type spells
10// `-> Layer<B>` on every arm, and a visitor whose arms naturally produce
11// different things names their union by hand. `eliminate` removes that: its
12// result type is the JOIN OF THE ARMS -- the one type every arm returns when
13// they agree, and otherwise `std::variant` of the distinct result types, in
14// alternative order. The join is a value type and nothing more. No operation
15// in this library reads it as a grade; `fmap` over such a result receives the
16// whole variant (docs/transpose/decisions.md#value-side-grading).
17//
18// Deduplication is by identity, in alternative order, and NOT canonical
19// (sorted) order: a variant's alternatives are positional, and the join of
20// arms that rebuild a `variant<Zero, Succ<B>>` must be exactly that type,
21// which a sorted set would not be. That is the one way this differs from the
22// error_set's canonical set, and it is why the two share only the dedupe
23// (docs/transpose/decisions.md#heterogeneous-arm-elimination).
24//
25// `eliminate_indexed` passes the alternative's index tag to the arm as well,
26// for arms that must tell two alternatives of one type apart, or that want
27// the position as a template argument.
28//
29// Proposed beside `overloaded` (docs/transpose/decisions.md#overloaded-home),
30// whose consteval catch-all is the exhaustiveness diagnostic: an alternative
31// no arm accepts is a compile error naming the alternative. Wording for both
32// is pending, with `overloaded`'s.

158template <class JOINED, class RESULT>
159constexpr auto into_join(RESULT &&result) noexcept(
160 std::is_same_v<std::remove_cvref_t<RESULT>, JOINED>
161 ? std::is_nothrow_constructible_v<JOINED, RESULT>
162 : std::is_nothrow_constructible_v<
163 JOINED, std::in_place_type_t<std::remove_cvref_t<RESULT>>,
164 RESULT>) -> JOINED {
165 using Result = std::remove_cvref_t<RESULT>;
166 if constexpr (std::is_same_v<Result, JOINED>) {
167 return std::forward<RESULT>(result);
168 } else {
169 return JOINED(std::in_place_type<Result>, std::forward<RESULT>(result));
170 }
171}

243template <class VARIANT, class HANDLER>
244 requires detail::variant_specialization<VARIANT>
245[[nodiscard]] constexpr auto eliminate(
246 VARIANT &&variant,
247 HANDLER &&handler) noexcept(detail::eliminate_nothrow_v<VARIANT, HANDLER>)
248 -> detail::eliminate_result_t<VARIANT, HANDLER> {
249 using Joined = detail::eliminate_result_t<VARIANT, HANDLER>;
250 return std::visit(
251 [&handler]<class ALTERNATIVE>(ALTERNATIVE &&alternative) -> Joined {
252 return detail::into_join<Joined>(
253 std::invoke(handler, std::forward<ALTERNATIVE>(alternative)));
254 },
255 std::forward<VARIANT>(variant));
256}

268template <class VARIANT, class HANDLER>
269 requires detail::variant_specialization<VARIANT>
270[[nodiscard]] constexpr auto
271eliminate_indexed(VARIANT &&variant, HANDLER &&handler) noexcept(
272 detail::eliminate_indexed_nothrow_v<VARIANT, HANDLER>)
273 -> detail::eliminate_indexed_result_t<VARIANT, HANDLER> {
274 using Joined = detail::eliminate_indexed_result_t<VARIANT, HANDLER>;
275 using Variant = std::remove_cvref_t<VARIANT>;
276 return [&]<std::size_t... INDICES>(
277 std::index_sequence<INDICES...>) -> Joined {
278 Joined (*const arms[])(VARIANT &&, HANDLER &) = {
279 [](VARIANT &&v, HANDLER &h) -> Joined {
280 return detail::into_join<Joined>(
281 std::invoke(h, std::in_place_index<INDICES>,
282 std::get<INDICES>(std::forward<VARIANT>(v))));
283 }...};
284 return arms[variant.index()](std::forward<VARIANT>(variant), handler);
285 }(std::make_index_sequence<std::variant_size_v<Variant>>{});
286}
