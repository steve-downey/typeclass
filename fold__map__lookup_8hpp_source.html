25// Lookup-based overloads of fold_map (Decision 4, as amended 2026-07-14).
26//
27// The everyday spelling: fold_map<Result>(map_fn, combine, identity, t)
28// resolves the structural ingredients — the layer fold, the layer fmap,
29// and (for trees in their own representation) the projection — through
30// typeclass-object lookup, so call sites pass only the elementwise
31// pieces and the data. The fully-explicit forms in fold_map.hpp remain
32// the escape hatch and the surface with no mechanism dependency.
33//
34// The lookup variables here follow the pattern of beman.transpose, the
35// Paper A (P3200) bundled-customization facility, whose functor<Layer>
36// supplies the layer fmap; this header consumes the mechanism and never
37// respecifies it.

61template <class MapFn, class Combine, class Result, class Layer>
62constexpr auto layer_fold_map(const MapFn& map_fn, const Combine& combine, const Result& identity, const Layer& layer)
63 -> Result {
64 static_assert(transpose::has_instance_v<decltype(layer_fold<std::remove_cvref_t<Layer> >)>,
65 "no layer-fold instance: specialize "
66 "beman::tree_algorithms::layer_fold_typeclass for this layer type, or use "
67 "the explicit forms in fold_map.hpp");
68 return layer_fold<std::remove_cvref_t<Layer> >(map_fn, combine, identity, layer);
69}

91template <typename Result, typename MapFn, typename Combine, typename Tree>
92constexpr auto fold_map(const MapFn& map_fn, const Combine& combine, const Result& identity, const Tree& tree)
93 -> Result {
94 static_assert(transpose::has_instance_v<decltype(project<Tree>)>,
95 "no projection instance: specialize "
96 "beman::tree_algorithms::project_typeclass for this tree type, or use the "
97 "explicit forms in fold_map.hpp");
98 auto algebra = [&](const auto& layer) -> Result { return layer_fold_map(map_fn, combine, identity, layer); };
99 auto fmap_fn = [](auto&& fn, const auto& layer) { return layer_fmap(std::forward<decltype(fn)>(fn), layer); };
100 return fold_with<Result>(algebra, fmap_fn, project<Tree>, &tree);
101}
