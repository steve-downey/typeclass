22template <class STRUCTURE>
23struct FingerTreeSequenceTraversableImpl {
24 using element_type = typename STRUCTURE::value_type;
25
26 template <class APPLICATIVE, class F>
27 constexpr auto traverse(this auto&&, const APPLICATIVE& applicative, F&& function, const STRUCTURE& source)
28 requires std::invocable<F&, const element_type&> &&
29 transpose::applicative_object_for<
30 APPLICATIVE,
31 std::remove_cvref_t<std::invoke_result_t<F&, const element_type&>>> &&
32 std::move_constructible<std::remove_cvref_t<std::invoke_result_t<F&, const element_type&>>> &&
33 std::constructible_from<std::remove_cvref_t<std::invoke_result_t<F&, const element_type&>>,
34 std::invoke_result_t<F&, const element_type&>> &&
35 detail::rebindable<STRUCTURE,
36 transpose::applicative_value_t<
37 std::remove_cvref_t<std::invoke_result_t<F&, const element_type&>>>>
38 {
39 using Context = std::remove_cvref_t<std::invoke_result_t<F&, const element_type&>>;
40 using U = transpose::applicative_value_t<Context>;
41 auto builder = finger_tree_rebind<STRUCTURE, U>::make_builder(source);
42
43 // The pointers exist only while the vector primitive invokes the
44 // mapper. Deferred effects and reconstruction own their own data.
45 std::vector<const element_type*> positions;
46 source.for_each([&](const element_type& value) { positions.push_back(std::addressof(value)); });
47 auto accumulated = transpose::traversable<decltype(positions)>.traverse(
48 applicative,
49 [&](const element_type* value) -> Context { return std::invoke(function, *value); },
50 positions);
51 return applicative.invoke(
52 [builder = std::move(builder)](std::vector<U> values) mutable {
53 return std::invoke(builder, std::move(values));
54 },
55 std::move(accumulated));
56 }
57};
