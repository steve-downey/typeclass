6// The canonical set of types: a pack deduplicated by identity and sorted by
7// a strict total order on types, so that two spellings of the same set are
8// one type. This is framework-neutral machinery -- it knows nothing about
9// what the set is FOR -- and it lives here, beside grade.hpp, so that any
10// model or any other canonical sum can build on it without acquiring the
11// vocabulary of the first client. See
12// docs/transpose/decisions.md#canonical-set-home.
13//
14// DEDUPLICATION is by `std::is_same`, never by name: the name is an ordering
15// key, not an equality.
16//
17// TYPE ORDERING is P2830's job. Until `std::type_order` is available this
18// header uses an interim ordering read out of the compiler's own function
19// signature, valid only for the documented restrictions: named types,
20// external linkage, stable spelling. Two distinct types that render to the
21// same name would be silently conflated, so a client checks
22// `names_are_distinct` and makes that case ill-formed rather than a wrong
23// answer.
24//
25// THE ORDER IS PART OF THE ABI. Any type whose layout follows the canonical
26// order -- a tuple of slots in sorted position, say -- has a different layout
27// under a different order, and two translation units that disagreed about
28// the order would violate the ODR silently. The order therefore carries an
29// inline namespace keyed on which order is in use, and an order-dependent
30// type carries the same key. Today there is one key,
31// `ordered_by_type_name`; a P2830 order arrives as a second key beside it,
32// so that mixing the two fails to link instead of corrupting a value. See
33// docs/transpose/decisions.md#type-order-abi.

52template <class T>
53consteval auto type_name() -> std::string_view {
54#if defined(_MSC_VER) && !defined(__clang__)
55 // MSVC has no __PRETTY_FUNCTION__. __FUNCSIG__ spells the specialization
56 // as `... type_name<T>(void)`, so the name is what sits between the
57 // template argument list and the last `>`. Should that spelling ever
58 // change, fall back to the whole signature, which still names T and so
59 // is still an injective key -- a duller order, not a broken one.
60 const std::string_view signature{__FUNCSIG__};
61 constexpr std::string_view marker{"type_name<"};
62
63 const auto found = signature.find(marker);
64 const auto stop = signature.rfind('>');
65 if (found == std::string_view::npos || stop == std::string_view::npos ||
66 stop <= found + marker.size()) {
67 return signature;
68 }
69 const auto start = found + marker.size();
70 return signature.substr(start, stop - start);
71#else
72 const std::string_view signature{__PRETTY_FUNCTION__};
73 constexpr std::string_view marker{"T = "};
74
75 const auto start = signature.find(marker) + marker.size();
76 auto stop = signature.find(';', start);
77 if (stop == std::string_view::npos) {
78 stop = signature.rfind(']');
79 }
80 return signature.substr(start, stop - start);
81#endif
82}

85template <class LEFT, class RIGHT>
86inline constexpr bool type_precedes_v = type_name<LEFT>() < type_name<RIGHT>();
87
90template <class... TYPES>
91consteval auto names_are_distinct() -> bool {
92 constexpr std::size_t count = sizeof...(TYPES);
93 const std::array<std::string_view, count> names{type_name<TYPES>()...};
94 for (std::size_t i = 0; i < count; ++i) {
95 for (std::size_t j = i + 1; j < count; ++j) {
96 if (names[i] == names[j]) {
97 return false;
98 }
99 }
100 }
101 return true;
102}
103
107template <class... TYPES>
108inline constexpr bool names_are_distinct_v = names_are_distinct<TYPES...>();
