35// A second tree you don't own — and this time, no Fix at all.
36//
37// FringeTree<T> below is a faithful port of a pre-existing persistent
38// sequence tree (smd::tree::FringeTree): values live exclusively at
39// leaves, branches carry only structure plus a cached measure (the leaf
40// count), concatenation is O(1) by structural sharing. Where the
41// BinaryTree adapter (binary_tree.hpp) demonstrates conversion —
42// to_fix/from_fix copying between representations — this adapter is
43// direct-only: a projection and an embedding connect the tree to
44// fold_with/unfold_with in its own representation, and no Fix form is
45// provided or needed. The layer holds child handles by value (branches
46// are never null), so there is no Box here either.
47//
48// The cached measure is the bridge to measured-sequence designs: the
49// embedding below routes through branch(), which computes the measure,
50// so trees built by unfold_with maintain the invariant by construction.
51
52// ---------------------------------------------------------------------
53// The existing tree, ported as-is.
54// ---------------------------------------------------------------------
55
56// 1484acca-10e3-4500-93e1-092ccec4e3f6
64template <class T>
65class FringeTree {
66 struct Empty {};
67 struct Leaf {
68 T d_value;
69 };
70 struct Branch {
71 std::size_t d_measure;
72 std::shared_ptr<FringeTree> d_left;
73 std::shared_ptr<FringeTree> d_right;
74 };
75
76 std::variant<Empty, Leaf, Branch> d_data;
77
78 public:
79 using value_type = T;
80
82 static auto empty() -> FringeTree { return FringeTree(Empty{}); }
83
85 static auto leaf(T value) -> FringeTree { return FringeTree(Leaf{std::move(value)}); }
86
89 static auto branch(FringeTree left, FringeTree right) -> FringeTree {
90 auto left_ptr = std::make_shared<FringeTree>(std::move(left));
91 auto right_ptr = std::make_shared<FringeTree>(std::move(right));
92 auto measure = left_ptr->measure() + right_ptr->measure();
93 return FringeTree(Branch{measure, std::move(left_ptr), std::move(right_ptr)});
94 }
95
97 auto is_empty() const -> bool { return std::holds_alternative<Empty>(d_data); }
99 auto is_leaf() const -> bool { return std::holds_alternative<Leaf>(d_data); }
101 auto is_branch() const -> bool { return std::holds_alternative<Branch>(d_data); }
102
104 auto measure() const -> std::size_t {
105 if (is_empty()) {
106 return 0U;
107 }
108 if (is_leaf()) {
109 return 1U;
110 }
111 return std::get<Branch>(d_data).d_measure;
112 }
113
115 auto value() const -> const T& {
116 assert(is_leaf());
117 return std::get<Leaf>(d_data).d_value;
118 }
119
121 auto left() const -> const FringeTree& {
122 assert(is_branch());
123 return *std::get<Branch>(d_data).d_left;
124 }
125
127 auto right() const -> const FringeTree& {
128 assert(is_branch());
129 return *std::get<Branch>(d_data).d_right;
130 }
131
133 auto flatten() const -> std::vector<T> {
134 if (is_empty()) {
135 return {};
136 }
137 if (is_leaf()) {
138 return {value()};
139 }
140 auto l = left().flatten();
141 auto r = right().flatten();
142 l.insert(l.end(), r.begin(), r.end());
143 return l;
144 }
145
147 static auto concat(const FringeTree& left_tree, const FringeTree& right_tree) -> FringeTree {
148 if (left_tree.is_empty()) {
149 return right_tree;
150 }
151 if (right_tree.is_empty()) {
152 return left_tree;
153 }
154 return branch(left_tree, right_tree);
155 }
156
158 auto cons(T x) const -> FringeTree { return concat(leaf(std::move(x)), *this); }
159
161 auto snoc(T x) const -> FringeTree { return concat(*this, leaf(std::move(x))); }
162
164 static auto from_sequence(std::vector<T> values) -> FringeTree {
165 auto result = empty();
166 for (auto& v : values) {
167 result = result.snoc(std::move(v));
168 }
169 return result;
170 }
171
172 private:
173 explicit FringeTree(Empty e) : d_data(std::move(e)) {}
174 explicit FringeTree(Leaf l) : d_data(std::move(l)) {}
175 explicit FringeTree(Branch b) : d_data(std::move(b)) {}
176};

179// ---------------------------------------------------------------------
180// Views: decomposition at either end, in Paper C's vocabulary.
181// ---------------------------------------------------------------------
182
183// Evidence, not (yet) proposed surface: these mirror the FingerTree
184// interface of Paper C — view_l/view_r returning an optional View of the
185// end element and the rest — spelled as free functions so the ported
186// class above stays byte-identical to its anchored form. The contract
187// is deliberately a QUOTIENT contract: the rest's element sequence is
188// specified, its shape is not. A representation is free to rebalance
189// internally (FingerTree does, aggressively); nothing observable through
190// views, concat, or the elementwise folds can tell. Sequence-generic
191// algorithms — cons/snoc, from_range, fold_left, sequence equality —
192// derive from empty/leaf/concat plus the views alone; see
193// examples/tree_algorithms/sequence_algorithms.cpp.
194
195// f4f21f0a-6b7e-4f0d-9a3c-58f1c2d7ab26
198template <typename T>
199struct FringeView {
200 T d_value;
201 FringeTree<T> d_rest;
202};
203
207template <typename T>
208auto view_l(const FringeTree<T>& tree) -> std::optional<FringeView<T> > {
209 if (tree.is_empty()) {
210 return std::nullopt;
211 }
212 if (tree.is_leaf()) {
213 return FringeView<T>{tree.value(), FringeTree<T>::empty()};
214 }
215 auto sub = view_l(tree.left());
216 if (!sub) {
217 return view_l(tree.right());
218 }
219 sub->d_rest = FringeTree<T>::concat(sub->d_rest, tree.right());
220 return sub;
221}
222
224template <typename T>
225auto view_r(const FringeTree<T>& tree) -> std::optional<FringeView<T> > {
226 if (tree.is_empty()) {
227 return std::nullopt;
228 }
229 if (tree.is_leaf()) {
230 return FringeView<T>{tree.value(), FringeTree<T>::empty()};
231 }
232 auto sub = view_r(tree.right());
233 if (!sub) {
234 return view_r(tree.left());
235 }
236 sub->d_rest = FringeTree<T>::concat(tree.left(), sub->d_rest);
237 return sub;
238}

269// ---------------------------------------------------------------------
270// Functor instance.
271// ---------------------------------------------------------------------

273// a692305b-eb87-43fb-98a0-901a582d7ddf
277template <typename T, typename A>
278struct FringeTreeFFunctorImpl {
279 template <typename Fn>
280 constexpr auto fmap(this auto&&, Fn&& fn, const FringeTreeF<T, A>& layer) {
281 using B = std::remove_cvref_t<std::invoke_result_t<Fn, const A&> >;
282 return std::visit(transpose::overloaded{
283 [](const FringeEmpty&) -> FringeTreeF<T, B> { return FringeEmpty{}; },
284 [](const FringeLeaf<T>& l) -> FringeTreeF<T, B> { return FringeLeaf<T>{l.value}; },
285 [&fn](const FringeBranch<A>& b) -> FringeTreeF<T, B> {
286 return FringeBranch<B>{std::invoke(fn, b.left), std::invoke(fn, b.right)};
287 },
288 },
289 layer);
290 }
291};
292
295template <typename T, typename A>
296struct FringeTreeFFunctorMap : transpose::derive_functor<FringeTreeFFunctorImpl<T, A> > {
297 using FringeTreeFFunctorImpl<T, A>::fmap;
298};

322struct FringeTreeProjectFn {
323 template <typename T>
324 auto operator()(const FringeTree<T>* t) const -> FringeTreeF<T, const FringeTree<T>*> {
325 if (t->is_empty()) {
326 return FringeEmpty{};
327 }
328 if (t->is_leaf()) {
329 return FringeLeaf<T>{t->value()};
330 }
331 return FringeBranch<const FringeTree<T>*>{&t->left(), &t->right()};
332 }
333};

341struct FringeTreeEmbedFn {
342 template <typename T>
343 auto operator()(FringeTreeF<T, FringeTree<T> >&& layer) const -> FringeTree<T> {
344 return std::visit(transpose::overloaded{

348 return FringeTree<T>::branch(std::move(b.left), std::move(b.right));
