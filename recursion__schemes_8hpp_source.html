22// The three recursion verbs over Fix<F>. Each takes the base functor's fmap
23// explicitly as a callable parameter; result carriers are leading explicit
24// template parameters because C++ cannot infer fold carriers through
25// recursive calls.
26//
27// Lookup-based overloads (which resolve fmap through beman.transpose's
28// functor<Layer> lookup) live in recursion_schemes_lookup.hpp (Decision 4);
29// this header has no typeclass dependency.

41template <typename Result, template <typename> class F, typename Algebra, typename FMap>
42constexpr auto fold_fix(const Algebra& algebra, const FMap& fmap_fn, const Fix<F>& tree) -> Result {
43 const auto& layer = unwrap_fix(tree);
44 auto evaluated =
45 fmap_fn([&](const Fix<F>& child) -> Result { return fold_fix<Result>(algebra, fmap_fn, child); }, layer);
46 return algebra(evaluated);
47}

50// The build verbs' recursive lambdas take the child seed by value, not by
51// reference as fold_fix's take the child tree. A layer holds a complete
52// seed inline (child_slot_t<Seed> is a std::optional), so the reference
53// fmap_fn hands the lambda points into the parent frame's layer, and MSVC's
54// constant evaluator (through 19.51) cannot read through it while the
55// recursive frame is initializing its own layer: "read of an uninitialized
56// symbol" on the slot's storage. Seeds are cheap handles by design, so the
57// copy costs nothing; fold_with keeps its children by reference because a
58// projection may hand out the subtree itself.
68template <template <typename> class F, typename Coalgebra, typename FMap, typename Seed>
69constexpr auto unfold_fix(const Coalgebra& coalgebra, const FMap& fmap_fn, const Seed& seed) -> Fix<F> {
70 auto layer = coalgebra(seed);
71 auto expanded = fmap_fn([&](Seed child) -> Fix<F> { return unfold_fix<F>(coalgebra, fmap_fn, child); }, layer);
72 return wrap_fix<F>(std::move(expanded));
73}

97// The direct verbs: fold and build over a tree in its own representation,
98// with no Fix<F> and no conversion. The extra ingredient is supplied
99// explicitly — a projection exposing one layer of the user's tree as a
100// base-functor value, or its dual, an embedding rebuilding one layer of
101// the user's tree from a completed base-functor layer.
102//
103// fold_with(algebra, fmap_fn, project, tree) is refold with project as
104// the coalgebra; unfold_with(coalgebra, fmap_fn, embed, seed) is refold
105// with embed as the algebra. fold_fix is the degenerate case where
106// project is unwrap_fix. The child handles inside a projected layer need
107// not be the tree type itself — a projection may hand out pointers or
108// other cheap handles, and the recursion follows whatever project accepts.
109
123// 6c596985-e990-4808-a504-2b651c34cebc
124template <typename Result, typename Algebra, typename FMap, typename Project, typename Tree>
125constexpr auto fold_with(const Algebra& algebra, const FMap& fmap_fn, const Project& project, const Tree& tree)
126 -> Result {
127 auto layer = project(tree);
128 auto evaluated = fmap_fn(
129 [&](const auto& child) -> Result { return fold_with<Result>(algebra, fmap_fn, project, child); }, layer);
130 return algebra(evaluated);
131}
