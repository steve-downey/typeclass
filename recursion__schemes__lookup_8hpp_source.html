24// Lookup-based overloads of the three recursion verbs (Decision 4).
25//
26// These are the everyday spelling: fmap is resolved through
27// beman.transpose's functor<Layer> lookup, keyed on the concrete layer type, so call sites pass
28// only the algebra/coalgebra and the data. The explicit-fmap primary verbs
29// in recursion_schemes.hpp remain the per-call-site escape hatch — and the
30// surface that stays fully valid independent of the Paper A lookup
31// mechanism this header consumes.

41// f6936d93-a4aa-4e85-afd1-092d4bbb1ba1
42template <class Fn, class Layer>
43constexpr auto layer_fmap(Fn&& fn, const Layer& layer) {
44 static_assert(transpose::has_instance_v<decltype(transpose::functor<std::remove_cvref_t<Layer> >)>,
45 "no Functor instance: specialize beman::transpose::functor "
46 "for this layer type, or use the explicit-fmap overloads in "
47 "recursion_schemes.hpp");
48 return transpose::functor<std::remove_cvref_t<Layer> >.fmap(std::forward<Fn>(fn), layer);
49}
50// f6936d93-a4aa-4e85-afd1-092d4bbb1ba1 end

58template <typename Result, template <typename> class F, typename Algebra>
59constexpr auto fold_fix(const Algebra& algebra, const Fix<F>& tree) -> Result {
60 const auto& layer = unwrap_fix(tree);
61 auto evaluated =
62 layer_fmap([&](const Fix<F>& child) -> Result { return fold_fix<Result>(algebra, child); }, layer);
63 return algebra(evaluated);
64}
