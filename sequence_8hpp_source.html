6// Foldable and Traversable instances for std::vector.
7//
8// std::vector is the shared traversable *structure* for all three of Paper A's
9// motivating domains. Pairing this one Traversable instance with three
10// different applicative contexts gives the three front-door results through a
11// single API:
12//
13// traverse(f, vector<optional<T>>) -> optional<vector<T>> (fallible value)
14// traverse(f, vector<sender<T>>) -> sender<vector<T>> (deferred result)
15// traverse(f, vector<zip_list<T>>) -> zip_list<vector<T>> (lanewise / SIMD)
16//
17// The traverse primitive sequences each element's effect left-to-right, lifting
18// element results back into a vector while preserving the original shape.
19//
20// This header stands in for <vector>, which in the proposal declares
21// the instances itself (docs/transpose/decisions.md#instance-placement).
22// Include it, not only <vector>, before the first use of a typeclass on a
23// std::vector, in every translation unit that makes one: a use that precedes
24// the instance is ill-formed, no diagnostic required. transpose.hpp includes
25// it.

46template <class VALUE_TYPE>
47struct VectorFoldableImpl {
48 using element_type = VALUE_TYPE;
49
50 // The trailing return type is the Impl-keeps-its-basis-SFINAE-friendly
51 // invariant apply.hpp records, applied to the fold basis: computed only
52 // in the body, the result type would be established by instantiating
53 // that body, and an availability probe would diagnose from inside it
54 // rather than fail to match. The monoid is an instance object over that
55 // result type, passed by value; the Impl never chooses one
56 // (docs/transpose/decisions.md#monoid-selection).
58 template <class FUNCTION, class MONOID>
59 requires monoid_object<MONOID, std::remove_cvref_t<std::invoke_result_t<
60 FUNCTION &, const VALUE_TYPE &>>>
61 constexpr auto fold_map(this auto &&, FUNCTION &&function,
62 const std::vector<VALUE_TYPE> &values,
63 MONOID monoid)
64 -> std::remove_cvref_t<
65 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>> {
66 using Result = std::remove_cvref_t<
67 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>;
68 Result accumulated = monoid.identity();
69 for (const auto &value : values) {
70 accumulated = monoid.combine(std::move(accumulated),
71 std::invoke(function, value));
72 }
73 return accumulated;
74 }
75
77 constexpr auto length(this auto &&, const std::vector<VALUE_TYPE> &values)
78 -> std::size_t {
79 return values.size();
80 }
81};

226template <class VALUE_TYPE>
227template <class APPLICATIVE, class FUNCTION, class VALUES>
228constexpr auto vector_traversable_impl<VALUE_TYPE>::traverse(
229 this auto &&, const APPLICATIVE &applicative, FUNCTION &&function,
230 VALUES &&values)
231 requires std::same_as<std::remove_cvref_t<VALUES>,
232 std::vector<VALUE_TYPE>> &&
233 std::invocable<FUNCTION &, detail::traversal_argument_t<
234 vector_traversable_impl, VALUES>> &&
235 applicative_object_for<
236 APPLICATIVE,
237 std::remove_cvref_t<std::invoke_result_t<
238 FUNCTION &, detail::traversal_argument_t<
239 vector_traversable_impl, VALUES>>>>
240{
241 // const VALUE_TYPE& for an lvalue or const vector, VALUE_TYPE&& for a
242 // non-const rvalue one: the category traverse's callers inferred the
243 // context from, so the constraint above asks about the only call made.
244 using Argument =
245 detail::traversal_argument_t<vector_traversable_impl, VALUES>;
246 using Effect =
247 std::remove_cvref_t<std::invoke_result_t<FUNCTION &, Argument>>;
248
249 if constexpr (detail::collecting_applicative<APPLICATIVE, Effect>) {
250 std::vector<Effect> effects;
251 effects.reserve(values.size());
252 for (auto &value : values) {
253 effects.push_back(
254 std::invoke(function, static_cast<Argument>(value)));
255 }
256 return applicative.collect(std::move(effects));
257 } else {
258 using Element = applicative_value_t<Effect>;
259
260 std::vector<Element> collected;
261 collected.reserve(values.size());
262
263 auto accumulated = applicative.pure(std::move(collected));
264 for (auto &value : values) {
265 accumulated = applicative.invoke(
266 detail::vector_append<Element>, std::move(accumulated),
267 std::invoke(function, static_cast<Argument>(value)));
268 }
269 return accumulated;
270 }
271}
