6// Functor, Foldable, Traversable and Monoid instances for std::vector.
7//
8// std::vector is the shared traversable *structure* for all three of Paper A's
9// motivating domains. Pairing this one Traversable instance with three
10// different applicative contexts gives the three front-door results through a
11// single API:
12//
13// traverse(f, vector<optional<T>>) -> optional<vector<T>> (fallible value)
14// traverse(f, vector<sender<T>>) -> sender<vector<T>> (deferred result)
15// traverse(f, vector<zip_list<T>>) -> zip_list<vector<T>> (lanewise / SIMD)
16//
17// The traverse primitive sequences each element's effect left-to-right, lifting
18// element results back into a vector while preserving the original shape.
19//
20// This header stands in for <vector>, which in the proposal declares
21// the instances itself (docs/transpose/decisions.md#instance-placement).
22// Include it, not only <vector>, before the first use of a typeclass on a
23// std::vector, in every translation unit that makes one: a use that precedes
24// the instance is ill-formed, no diagnostic required. beman/transpose.hpp
25// includes it.

47template <class VALUE_TYPE>
48struct VectorFunctorImpl {
49 // Constrained for the reason the optional instance's is.
50 template <class F>
51 requires std::invocable<F &, const VALUE_TYPE &> &&
52 detail::holdable_value<
53 std::invoke_result_t<F &, const VALUE_TYPE &>>
54 constexpr auto fmap(this auto &&, F &&function,
55 const std::vector<VALUE_TYPE> &values) {
56 // `F &`, unlike the optional instance above: that one forwards the
57 // callable into a single std::invoke, this one invokes a captured
58 // lvalue once per element.
59 using Result = std::invoke_result_t<F &, const VALUE_TYPE &>;
60 std::vector<std::remove_cvref_t<Result>> output;
61 output.reserve(values.size());
62
63 std::ranges::transform(values, std::back_inserter(output),
64 [&function](const VALUE_TYPE &v) {
65 return std::invoke(function, v);
66 });
67
68 return output;
69 }
70};

86template <class VALUE_TYPE>
87struct Monoid<std::vector<VALUE_TYPE>> {
88 using value_type = std::vector<VALUE_TYPE>;
89
90 constexpr auto identity() const -> std::vector<VALUE_TYPE> { return {}; }
91
92 constexpr auto combine(std::vector<VALUE_TYPE> lhs,
93 const std::vector<VALUE_TYPE> &rhs) const
94 -> std::vector<VALUE_TYPE> {
95 lhs.insert(lhs.end(), rhs.begin(), rhs.end());

105template <class VALUE_TYPE>
106struct VectorFoldableImpl {
107 using element_type = VALUE_TYPE;
108
109 // The trailing return type is the Impl-keeps-its-basis-SFINAE-friendly
110 // invariant apply.hpp records, applied to the fold basis: computed only
111 // in the body, the result type would be established by instantiating
112 // that body, and an availability probe would diagnose from inside it
113 // rather than fail to match. The monoid is an instance object over that
114 // result type, passed by value; the Impl never chooses one
115 // (docs/transpose/decisions.md#monoid-selection).
117 template <class FUNCTION, class MONOID>
118 requires monoid_object<MONOID, std::remove_cvref_t<std::invoke_result_t<
119 FUNCTION &, const VALUE_TYPE &>>>
120 constexpr auto fold_map(this auto &&, FUNCTION &&function,
121 const std::vector<VALUE_TYPE> &values,
122 MONOID monoid)
123 -> std::remove_cvref_t<
124 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>> {
125 using Result = std::remove_cvref_t<
126 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>;
127 Result accumulated = monoid.identity();
128 for (const auto &value : values) {
129 accumulated = monoid.combine(std::move(accumulated),
130 std::invoke(function, value));
131 }
132 return accumulated;
133 }
134
136 constexpr auto length(this auto &&, const std::vector<VALUE_TYPE> &values)
137 -> std::size_t {
138 return values.size();
139 }
140};

285template <class VALUE_TYPE>
286template <class APPLICATIVE, class FUNCTION, class VALUES>
287constexpr auto vector_traversable_impl<VALUE_TYPE>::traverse(
288 this auto &&, const APPLICATIVE &applicative, FUNCTION &&function,
289 VALUES &&values)
290 requires std::same_as<std::remove_cvref_t<VALUES>,
291 std::vector<VALUE_TYPE>> &&
292 std::invocable<FUNCTION &, detail::traversal_argument_t<
293 vector_traversable_impl, VALUES>> &&
294 applicative_object_for<
295 APPLICATIVE,
296 std::remove_cvref_t<std::invoke_result_t<
297 FUNCTION &, detail::traversal_argument_t<
298 vector_traversable_impl, VALUES>>>>
299{
300 // const VALUE_TYPE& for an lvalue or const vector, VALUE_TYPE&& for a
301 // non-const rvalue one: the category traverse's callers inferred the
302 // context from, so the constraint above asks about the only call made.
303 using Argument =
304 detail::traversal_argument_t<vector_traversable_impl, VALUES>;
305 using Effect =
306 std::remove_cvref_t<std::invoke_result_t<FUNCTION &, Argument>>;
307
308 if constexpr (detail::collecting_applicative<APPLICATIVE, Effect>) {
309 std::vector<Effect> effects;
310 effects.reserve(values.size());
311 for (auto &value : values) {
312 effects.push_back(
313 std::invoke(function, static_cast<Argument>(value)));
314 }
315 return applicative.collect(std::move(effects));
316 } else {
317 using Element = applicative_value_t<Effect>;
318
319 std::vector<Element> collected;
320 collected.reserve(values.size());
321
322 auto accumulated = applicative.pure(std::move(collected));
323 for (auto &value : values) {
324 accumulated = applicative.invoke(
325 detail::vector_append<Element>, std::move(accumulated),
326 std::invoke(function, static_cast<Argument>(value)));
327 }
328 return accumulated;
329 }
330}
