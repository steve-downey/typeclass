6// Foldable and Traversable instances for std::vector.
7//
8// std::vector is the shared traversable *structure* for all three of Paper A's
9// motivating domains. Pairing this one Traversable instance with three
10// different applicative contexts gives the three front-door results through a
11// single API:
12//
13// traverse(f, vector<optional<T>>) -> optional<vector<T>> (fallible value)
14// traverse(f, vector<sender<T>>) -> sender<vector<T>> (deferred result)
15// traverse(f, vector<zip_list<T>>) -> zip_list<vector<T>> (lanewise / SIMD)
16//
17// The traverse primitive sequences each element's effect left-to-right, lifting
18// element results back into a vector while preserving the original shape.

38template <class VALUE_TYPE>
39struct VectorFoldableImpl {
40 using element_type = VALUE_TYPE;
41
42 // The trailing return type is the Impl-keeps-its-basis-SFINAE-friendly
43 // invariant apply.hpp records, applied to the fold basis: computed only
44 // in the body, the result type would be established by instantiating
45 // that body, and an availability probe would diagnose from inside it
46 // rather than fail to match. The monoid is an instance object over that
47 // result type, passed by value; the Impl never chooses one
48 // (docs/transpose/decisions.md#monoid-selection).
50 template <class FUNCTION, class MONOID>
51 requires monoid_object<MONOID, std::remove_cvref_t<std::invoke_result_t<
52 FUNCTION &, const VALUE_TYPE &>>>
53 constexpr auto fold_map(this auto &&, FUNCTION &&function,
54 const std::vector<VALUE_TYPE> &values,
55 MONOID monoid)
56 -> std::remove_cvref_t<
57 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>> {
58 using Result = std::remove_cvref_t<
59 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>;
60 Result accumulated = monoid.identity();
61 for (const auto &value : values) {
62 accumulated = monoid.combine(std::move(accumulated),
63 std::invoke(function, value));
64 }
65 return accumulated;
66 }
67
69 constexpr auto length(this auto &&, const std::vector<VALUE_TYPE> &values)
70 -> std::size_t {
71 return values.size();
72 }
73};

203template <class VALUE_TYPE>
204template <class APPLICATIVE, class FUNCTION>
205constexpr auto vector_traversable_impl<VALUE_TYPE>::traverse(
206 this auto &&, const APPLICATIVE &applicative, FUNCTION &&function,
207 const std::vector<VALUE_TYPE> &values) {
208 using Effect = std::remove_cvref_t<
209 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>;
210
211 if constexpr (detail::collecting_applicative<APPLICATIVE, Effect>) {
212 std::vector<Effect> effects;
213 effects.reserve(values.size());
214 for (const auto &value : values) {
215 effects.push_back(std::invoke(function, value));
216 }
217 return applicative.collect(std::move(effects));
218 } else {
219 using Element = applicative_value_t<Effect>;
220
221 std::vector<Element> collected;
222 collected.reserve(values.size());
223
224 auto accumulated = applicative.pure(std::move(collected));
225 for (const auto &value : values) {
226 accumulated = applicative.invoke(detail::vector_append<Element>,
227 std::move(accumulated),
228 std::invoke(function, value));
229 }
230 return accumulated;
231 }
232}
