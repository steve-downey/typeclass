6// Functor, Foldable, Traversable and Monoid instances for std::vector.
7//
8// std::vector is the shared traversable *structure* for all three of Paper A's
9// motivating domains. Pairing this one Traversable instance with three
10// different applicative contexts gives the three front-door results through a
11// single API:
12//
13// traverse(f, vector<optional<T>>) -> optional<vector<T>> (fallible value)
14// traverse(f, vector<sender<T>>) -> sender<vector<T>> (deferred result)
15// traverse(f, vector<zip_list<T>>) -> zip_list<vector<T>> (lanewise / SIMD)
16//
17// The traverse primitive sequences each element's effect left-to-right, lifting
18// element results back into a vector while preserving the original shape.
19//
20// This header stands in for <vector>, which in the proposal declares
21// the instances itself (docs/transpose/decisions.md#instance-placement).
22// Include it, not only <vector>, before the first use of a typeclass on a
23// std::vector, in every translation unit that makes one: a use that precedes
24// the instance is ill-formed, no diagnostic required. beman/transpose.hpp
25// includes it.

47template <class VALUE_TYPE>
48struct VectorFunctorImpl {
49 // Constrained for the reason the optional instance's is. No exception
50 // specification: the result is a vector, which allocates.
51 template <class F>
52 requires std::invocable<F &, const VALUE_TYPE &> &&
53 detail::holdable_value<
54 std::invoke_result_t<F &, const VALUE_TYPE &>>
55 [[nodiscard]] constexpr auto fmap(this auto &&, F &&function,
56 const std::vector<VALUE_TYPE> &values) {
57 // `F &`, unlike the optional instance above: that one forwards the
58 // callable into a single std::invoke, this one invokes a captured
59 // lvalue once per element.
60 using Result = std::invoke_result_t<F &, const VALUE_TYPE &>;
61 std::vector<std::remove_cvref_t<Result>> output;
62 output.reserve(values.size());
63
64 std::ranges::transform(values, std::back_inserter(output),
65 [&function](const VALUE_TYPE &v) {
66 return std::invoke(function, v);
67 });
68
69 return output;
70 }
71};

87template <class VALUE_TYPE>
88struct Monoid<std::vector<VALUE_TYPE>> {
89 using value_type = std::vector<VALUE_TYPE>;
90
91 constexpr auto identity() const noexcept -> std::vector<VALUE_TYPE> {
92 return {};
93 }
94
95 // No exception specification: concatenation allocates.
96 constexpr auto combine(std::vector<VALUE_TYPE> lhs,
97 const std::vector<VALUE_TYPE> &rhs) const
98 -> std::vector<VALUE_TYPE> {
99 lhs.insert(lhs.end(), rhs.begin(), rhs.end());
100 return lhs;
101 }
102};

109template <class VALUE_TYPE>
110struct VectorFoldableImpl {
111 using element_type = VALUE_TYPE;
112
113 // The fold's exception specification: the identity, each application of
114 // the function and its combination into the accumulator, the assignment
115 // of the combination back, and the move of the result out.
117 template <class FUNCTION, class MONOID>
118 static constexpr bool fold_map_nothrow =
119 noexcept(std::declval<const MONOID &>().identity()) &&
120 noexcept(std::declval<const MONOID &>().combine(
121 std::declval<std::remove_cvref_t<
122 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>>(),
123 std::invoke(std::declval<FUNCTION &>(),
124 std::declval<const VALUE_TYPE &>()))) &&
125 std::is_nothrow_assignable_v<
126 std::remove_cvref_t<
127 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>> &,
128 decltype(std::declval<const MONOID &>().combine(
129 std::declval<std::remove_cvref_t<
130 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>>(),
131 std::invoke(std::declval<FUNCTION &>(),
132 std::declval<const VALUE_TYPE &>())))> &&
133 std::is_nothrow_move_constructible_v<std::remove_cvref_t<
134 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>>;
135
136 // The trailing return type is the Impl-keeps-its-basis-SFINAE-friendly
137 // invariant apply.hpp records, applied to the fold basis: computed only
138 // in the body, the result type would be established by instantiating
139 // that body, and an availability probe would diagnose from inside it
140 // rather than fail to match. The monoid is an instance object over that
141 // result type, passed by value; the Impl never chooses one
142 // (docs/transpose/decisions.md#monoid-selection).
144 template <class FUNCTION, class MONOID>
145 requires monoid_object<MONOID, std::remove_cvref_t<std::invoke_result_t<
146 FUNCTION &, const VALUE_TYPE &>>>
147 [[nodiscard]] constexpr auto
148 fold_map(this auto &&, FUNCTION &&function,
149 const std::vector<VALUE_TYPE> &values,
150 MONOID monoid) noexcept(fold_map_nothrow<FUNCTION, MONOID>)
151 -> std::remove_cvref_t<
152 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>> {
153 using Result = std::remove_cvref_t<
154 std::invoke_result_t<FUNCTION &, const VALUE_TYPE &>>;
155 Result accumulated = monoid.identity();
156 for (const auto &value : values) {
157 accumulated = monoid.combine(std::move(accumulated),
158 std::invoke(function, value));
159 }
160 return accumulated;
161 }
162
164 [[nodiscard]] constexpr auto
165 length(this auto &&, const std::vector<VALUE_TYPE> &values) noexcept
166 -> std::size_t {
167 return values.size();
168 }
169};

320template <class VALUE_TYPE>
321template <class APPLICATIVE, class FUNCTION, class VALUES>
322[[nodiscard]] constexpr auto vector_traversable_impl<VALUE_TYPE>::traverse(
323 this auto &&, const APPLICATIVE &applicative, FUNCTION &&function,
324 VALUES &&values)
325 requires std::same_as<std::remove_cvref_t<VALUES>,
326 std::vector<VALUE_TYPE>> &&
327 std::invocable<FUNCTION &, detail::traversal_argument_t<
328 vector_traversable_impl, VALUES>> &&
329 applicative_object_for<
330 APPLICATIVE,
331 std::remove_cvref_t<std::invoke_result_t<
332 FUNCTION &, detail::traversal_argument_t<
333 vector_traversable_impl, VALUES>>>>
334{
335 // const VALUE_TYPE& for an lvalue or const vector, VALUE_TYPE&& for a
336 // non-const rvalue one: the category traverse's callers inferred the
337 // context from, so the constraint above asks about the only call made.
338 using Argument =
339 detail::traversal_argument_t<vector_traversable_impl, VALUES>;
340 using Effect =
341 std::remove_cvref_t<std::invoke_result_t<FUNCTION &, Argument>>;
342
343 if constexpr (detail::collecting_applicative<APPLICATIVE, Effect>) {
344 std::vector<Effect> effects;
345 effects.reserve(values.size());
346 for (auto &value : values) {
347 effects.push_back(
348 std::invoke(function, static_cast<Argument>(value)));
349 }
350 return applicative.collect(std::move(effects));
351 } else {
352 using Element = applicative_value_t<Effect>;
353
354 std::vector<Element> collected;
355 collected.reserve(values.size());
356
357 auto accumulated = applicative.pure(std::move(collected));
358 for (auto &value : values) {
359 accumulated = applicative.invoke(
360 detail::vector_append<Element>, std::move(accumulated),
361 std::invoke(function, static_cast<Argument>(value)));
362 }
363 return accumulated;
364 }
365}
