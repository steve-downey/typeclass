30// Stand-ins for the continuations the bind-basis derivations hand to bind.
31// A derivation's continuation is a lambda over an element whose type only
32// bind knows, so its validity cannot be spelled at the derived member's
33// declaration directly. These name it instead: each call operator is
34// constrained on what the matching continuation does with the element, and
35// spells the continuation's result type, so a bind that checks its
36// continuation -- the shipped instances spell their result through
37// invoke_result_t on it -- fails the probe softly for a function the
38// derivation could not have applied. Like probe_witness, each body
39// establishes only the result type.
40//
41// Each call operator's exception specification is that of the continuation
42// it stands in for, so a derived operation's exception specification --
43// computed by asking whether bind with the probe is noexcept -- is the one
44// the body's own continuation would give (see
45// docs/transpose/decisions.md#operation-noexcept).
46
50template <class IMPL, class FUNCTION>
51struct fmap_continuation_probe {
52 template <class ARGUMENT>
53 requires std::invocable<FUNCTION &, ARGUMENT> &&
54 requires(const IMPL &impl) {
55 impl.pure(std::invoke(std::declval<FUNCTION &>(),
56 std::declval<ARGUMENT>()));
57 }
58 constexpr auto operator()(ARGUMENT &&) const
59 noexcept(noexcept(std::declval<const IMPL &>().pure(
60 std::invoke(std::declval<FUNCTION &>(), std::declval<ARGUMENT>()))))
61 -> decltype(std::declval<const IMPL &>().pure(std::invoke(
62 std::declval<FUNCTION &>(), std::declval<ARGUMENT>()))) {
63 std::unreachable();
64 }
65};

229 template <class FUNCTION, class MA>
230 [[nodiscard]] constexpr auto
231 fmap(this auto &&self, FUNCTION &&function,
232 MA &&ma) noexcept(fmap_nothrow<decltype(self), FUNCTION, MA>())
233 requires requires(const Impl &impl) {
234 impl.fmap(std::forward<FUNCTION>(function), std::forward<MA>(ma));
235 } || requires(const Impl &impl) {
236 // The continuation the body binds with, pure . function -- not
237 // function itself, which a bind would accept only if it already
238 // returned a monadic value.
239 impl.bind(std::forward<MA>(ma),
240 detail::fmap_continuation_probe<Impl, FUNCTION>{});
241 }
242 {
243 if constexpr (requires {
244 impl_of(self).fmap(std::forward<FUNCTION>(function),
245 std::forward<MA>(ma));
246 }) {
247 return impl_of(self).fmap(std::forward<FUNCTION>(function),
248 std::forward<MA>(ma));
249 } else {
250 return impl_of(self).bind(std::forward<MA>(ma), [&](auto &&a) {
251 return impl_of(self).pure(
252 std::invoke(function, std::forward<decltype(a)>(a)));
253 });
254 }

257 // invoke: n-ary lift synthesized from bind + pure (left-nested binds):
258 // invoke(f, m1, ..., mn) = m1 >>= \a1 -> ... mn >>= \an ->
259 // pure(f(a1...an))
260 // Prefers a native Impl::invoke when present, like Applicative. This
261 // derivation assumes a synchronous bind (the continuation is invoked
262 // before bind returns), which holds for every Monad instance in this
263 // repository.
264 template <class FUNCTION, class FIRST, class... REST>
265 [[nodiscard]] constexpr auto
266 invoke(this auto &&self, FUNCTION &&function, FIRST &&first,
267 REST &&...rest) noexcept(invoke_nothrow<decltype(self), FUNCTION,
268 FIRST, REST...>())
269 requires requires(const Impl &impl) {
270 impl.invoke(std::forward<FUNCTION>(function),
271 std::forward<FIRST>(first),
272 std::forward<REST>(rest)...);
273 } || requires {
274 // The derivation's own requirement: binding FIRST, then each of
275 // REST in turn, reaches a continuation that can apply FUNCTION
276 // to every element and embed the result with pure. The
277 // continuation probe spells each step, so a function that
278 // cannot take the elements fails here rather than in the body.
279 // The probe follows the derived path all the way down, as the
280 // body does once it has not found a native Impl::invoke.
281 self.bind(
282 std::forward<FIRST>(first),
283 detail::invoke_continuation_probe<decltype(self), FUNCTION,
284 std::tuple<>, REST...>{});
285 }
286 {
287 if constexpr (requires {
288 impl_of(self).invoke(std::forward<FUNCTION>(function),
289 std::forward<FIRST>(first),
290 std::forward<REST>(rest)...);
291 }) {
292 return impl_of(self).invoke(std::forward<FUNCTION>(function),
293 std::forward<FIRST>(first),
294 std::forward<REST>(rest)...);
295 } else {
296 return self.bind(std::forward<FIRST>(first), [&](auto &&head) {
297 if constexpr (sizeof...(REST) == 0) {
298 return self.pure(std::invoke(
299 function, std::forward<decltype(head)>(head)));
300 } else {
301 return self.invoke(
302 [&function, &head](auto &&...tail) {
303 return std::invoke(
304 function, head,
305 std::forward<decltype(tail)>(tail)...);
306 },
307 std::forward<REST>(rest)...);
308 }
309 });
310 }
311 }

313 // join: flatten nested monad.
314 // join mma = mma >>= id
315 // Prefers a native Impl::join. join/bind is a mutually-derivable pair
316 // (join = bind(., id); bind is recoverable from join + fmap), so the
317 // fallback addresses Impl directly, matching fmap's bind-basis branch --
318 // the one member in this step where the derivation, not just the probe,
319 // addresses Impl rather than self.
320 template <class MMA>
321 [[nodiscard]] constexpr auto
322 join(this auto &&self,
323 MMA &&mma) noexcept(join_nothrow<decltype(self), MMA>())
324 requires requires(const Impl &impl) {
325 impl.join(std::forward<MMA>(mma));
326 } || requires(const Impl &impl) {
327 impl.bind(std::forward<MMA>(mma), detail::identity_continuation{});
328 }
329 {
330 if constexpr (requires {
331 impl_of(self).join(std::forward<MMA>(mma));
332 }) {
333 return impl_of(self).join(std::forward<MMA>(mma));
334 } else {
335 return impl_of(self).bind(std::forward<MMA>(mma),
336 detail::identity_continuation{});
337 }
338 }

340 // kleisli: forward Kleisli composition (>=>).
341 // (f >=> g) a = f a >>= g
342 // Prefers a native Impl::kleisli. Whether the composition is valid
343 // depends on the type of "a", which is not known until the composed
344 // function is called, so nothing about f and g can be checked here. The
345 // derived kleisli therefore requires only what building the composition
346 // needs, and the composition's call operator carries the real
347 // constraint: calling it with an argument f cannot take, or whose
348 // result g cannot continue, fails softly at the call.
349 template <class F, class G>
350 [[nodiscard]] constexpr auto
351 kleisli(this auto &&self, F f,
352 G g) noexcept(kleisli_nothrow<decltype(self), F, G>())
353 requires requires(const Impl &impl) { impl.kleisli(f, g); } ||
354 (std::move_constructible<F> && std::move_constructible<G>)
355 {
356 if constexpr (requires { impl_of(self).kleisli(f, g); }) {
357 return impl_of(self).kleisli(f, g);
358 } else {
359 return detail::kleisli_composition<
360 std::remove_reference_t<decltype(self)>, F, G>{
361 self, std::move(f), std::move(g)};
362 }
363 }

594 requires(const OBJ &obj, const CONTEXT &context,

596 obj.pure(element);

615// -- Operation objects --
616//
617// See functor.hpp for the shape, the motivation, and why the namespace is
618// nested. Still evidence, not proposed wording -- but evidence of one more
619// thing now: that the operation-object layer is uniform across the whole
620// typeclass family, including the member nobody is proposing. If the spelling
621// only worked for the operations D3200R0 happens to propose, that would say
622// it had been fitted to those operations instead of to the mechanism.
623//
624// It also settles a naming question the free-function API had ducked. `bind`
625// could not be a function template here without reading as a competitor to
626// std::bind; the old spelling was `mbind`, a wart adopted to dodge exactly
627// that. An object needs no such dodge: `bind` is a variable, it is
628// never found by ADL, and it never enters overload resolution with std::bind.
629// The typeclass operation gets its real name.
