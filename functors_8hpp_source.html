30// Reusable demo base functors beyond the expression tree. Currently the
31// NatF family: unary (Peano) naturals, the smallest recursive datatype
32// with a genuinely recursive position, used by the paper and tests to
33// demonstrate unfold_fix and refold. Layout is types -> functor_typeclass
34// instance -> smart constructors/converters, so the partial specialization
35// is visible before any function here that dispatches through the lookup
36// verbs for NatF layers.
37
38// ---------------------------------------------------------------------
39// NatF — unary naturals.
40// ---------------------------------------------------------------------
41
42// 7970dd71-b5dd-4fa0-8e80-735a26833b65
44struct Zero {};
45
48template <typename A>
49struct Succ {
50 child_slot_t<A> pred;
51};
52
54template <typename A>
55using NatF = std::variant<Zero, Succ<A> >;
56// 7970dd71-b5dd-4fa0-8e80-735a26833b65 end
57
59using Nat = Fix<NatF>;

61// ---------------------------------------------------------------------
62// Functor instance.
63// ---------------------------------------------------------------------
64
65// edbdeea5-9b3e-4d9a-88d5-fb2610c9174d
68template <typename A>
69struct NatFFunctorImpl {
70 template <typename Fn>
71 constexpr auto fmap(this auto&&, Fn&& fn, const NatF<A>& layer) {
72 using B = std::remove_cvref_t<std::invoke_result_t<Fn, const A&> >;
73 return std::visit(
74 transpose::overloaded{
75 [](const Zero&) -> NatF<B> { return Zero{}; },
76 [&fn](const Succ<A>& s) -> NatF<B> { return Succ<B>{make_slot<B>(std::invoke(fn, *s.pred))}; },
77 },
78 layer);
79 }
80};
81
84template <typename A>
85struct NatFFunctorMap : transpose::derive_functor<NatFFunctorImpl<A> > {
86 using NatFFunctorImpl<A>::fmap;
87};

116constexpr auto nat_from_int(int n) -> Nat {
117 return unfold_fix<NatF>(
118 [](int m) -> NatF<int> {
119 if (m <= 0) {
120 return Zero{};
121 }
122 return Succ<int>{make_slot<int>(m - 1)};
123 },
124 n);
125}

128constexpr auto nat_to_int(const Nat& nat) -> int {

131 return std::visit(transpose::overloaded{
132 [](const Zero&) { return 0; },
133 [](const Succ<int>& s) { return *s.pred + 1; },
134 },
135 layer);
136 },
