21// Monoid pattern invariants:
22// - A monoid is an *instance object*: anything with `identity()` and
23// `combine(lhs, rhs)` over one value type. `monoid_object<M, T>` is the
24// concept.
25// - Monoid<T> is the registry: the one canonical instance for a type, where
26// one exists (string and vector concatenation). monoid<T> is the lookup
27// object generic algorithms default to when the caller passes nothing.
28// - Where a type carries more than one monoid -- numbers, booleans -- no
29// instance is registered; the caller passes one (`sum_monoid<int>{}`,
30// `max_monoid<int>{}`, `any_monoid{}`, ...). The value type is never
31// wrapped to select a monoid. See docs/decisions.md#monoid-selection.
32// - Instances may carry state. The library may copy them and pass them by
33// value; the expectation (not a requirement) is that they are shaped like
34// stateless objects whose calls resolve statically.
35// - identity and combine must stay coherent for a single associative law
36// domain.
37

44struct Monoid;

48template <class VALUE_TYPE>
49inline constexpr Monoid<VALUE_TYPE> monoid = Monoid<VALUE_TYPE>{};

61template <class M, class T>
62concept monoid_object = requires(const M &m, const T &a, const T &b) {
63 { m.identity() } -> std::convertible_to<T>;
64 { m.combine(a, b) } -> std::convertible_to<T>;
65};

117template <class T>
118struct max_monoid {
119 using value_type = T;
120
121 constexpr auto identity() const -> T {
122 if constexpr (std::numeric_limits<T>::has_infinity) {
123 return -std::numeric_limits<T>::infinity();
124 } else {
125 return std::numeric_limits<T>::lowest();
126 }

129 constexpr auto combine(const T &lhs, const T &rhs) const -> T {

198// Combinators: an instance built from instances. Operands are held by value
199// as their concrete types, so composing instances erases nothing the

207template <class M>
208struct dual_monoid {
209 using value_type = monoid_value_t<M>;
210
211 M d_inner{};
212
213 constexpr auto identity() const -> value_type { return d_inner.identity(); }
214
215 constexpr auto combine(const value_type &lhs, const value_type &rhs) const
216 -> value_type {
217 return d_inner.combine(rhs, lhs);
218 }
219};

224template <class... M>
225struct tuple_monoid {
226 using value_type = std::tuple<monoid_value_t<M>...>;
227
228 std::tuple<M...> d_inners{};
229
230 constexpr auto identity() const -> value_type {
231 return std::apply(
232 [](const M &...inner) { return value_type{inner.identity()...}; },
233 d_inners);
234 }
235
236 constexpr auto combine(const value_type &lhs, const value_type &rhs) const
237 -> value_type {
238 return [&]<std::size_t... I>(std::index_sequence<I...>) {
239 return value_type{std::get<I>(d_inners).combine(
240 std::get<I>(lhs), std::get<I>(rhs))...};
241 }(std::index_sequence_for<M...>{});
242 }
243};
