20// Monoid pattern invariants:
21// - A monoid is an *instance object*: anything with `identity()` and
22// `combine(lhs, rhs)` over one value type. `monoid_object<M, T>` is the
23// concept.
24// - Monoid<T> is the registry: the one canonical instance for a type, where
25// one exists (string and vector concatenation). monoid<T> is the lookup
26// object generic algorithms default to when the caller passes nothing.
27// - Where a type carries more than one monoid -- numbers, booleans -- no
28// instance is registered; the caller passes one (`sum_monoid<int>{}`,
29// `max_monoid<int>{}`, `any_monoid{}`, ...). The value type is never
30// wrapped to select a monoid. See
31// docs/transpose/decisions.md#monoid-selection.
32// - Instances may carry state. The library may copy them and pass them by
33// value; the expectation (not a requirement) is that they are shaped like
34// stateless objects whose calls resolve statically.
35// - identity and combine must stay coherent for a single associative law
36// domain.
37// - identity and combine carry computed exception specifications, so that a
38// fold's own specification, which is computed through them, can be true
39// (docs/transpose/decisions.md#operation-noexcept).

64template <class M, class T>
65concept monoid_object = requires(const M &m, const T &a, const T &b) {
66 { m.identity() } -> std::convertible_to<T>;
67 { m.combine(a, b) } -> std::convertible_to<T>;

128template <class T>
129struct max_monoid {
130 using value_type = T;
131
132 constexpr auto identity() const
133 noexcept(std::is_nothrow_copy_constructible_v<T>) -> T {
134 if constexpr (std::numeric_limits<T>::has_infinity) {
135 return -std::numeric_limits<T>::infinity();
136 } else {
137 return std::numeric_limits<T>::lowest();

141 constexpr auto combine(const T &lhs, const T &rhs) const
142 noexcept(noexcept(lhs < rhs) && std::is_nothrow_copy_constructible_v<T>)
143 -> T {
144 return lhs < rhs ? rhs : lhs;

224template <class M>
225struct dual_monoid {

228 M d_inner{};

230 constexpr auto identity() const noexcept(noexcept(d_inner.identity()))

244template <class... M>
245struct tuple_monoid {
246 using value_type = std::tuple<monoid_value_t<M>...>;
247
248 std::tuple<M...> d_inners{};
249
250 constexpr auto identity() const noexcept(
251 (noexcept(std::declval<const M &>().identity()) && ...) &&
252 std::is_nothrow_constructible_v<
253 value_type, decltype(std::declval<const M &>().identity())...>)
254 -> value_type {
255 return std::apply(
256 [](const M &...inner) { return value_type{inner.identity()...}; },
257 d_inners);
258 }
259
260 constexpr auto combine(const value_type &lhs, const value_type &rhs) const
261 noexcept(
262 (noexcept(std::declval<const M &>().combine(
263 std::declval<const monoid_value_t<M> &>(),
264 std::declval<const monoid_value_t<M> &>())) &&
265 ...) &&
266 std::is_nothrow_constructible_v<
267 value_type, decltype(std::declval<const M &>().combine(
268 std::declval<const monoid_value_t<M> &>(),
269 std::declval<const monoid_value_t<M> &>()))...>)
270 -> value_type {
271 return [&]<std::size_t... I>(std::index_sequence<I...>) {
272 return value_type{std::get<I>(d_inners).combine(
273 std::get<I>(lhs), std::get<I>(rhs))...};
274 }(std::index_sequence_for<M...>{});
275 }
276};
