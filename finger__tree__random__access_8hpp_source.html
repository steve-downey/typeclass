5// Persistent random-access sequence backed by a finger tree.
6//
7// The Tree template parameter accepts any fully-specialised finger tree type
8// whose measure is an element count (sum_monoid<std::size_t> over
9// UnitMeasure):
10//
11// FingerTreeRandomAccess<int> // FingerTree<int>-backed (default)
12// FingerTreeRandomAccess<int, SomeOtherTree<int>> // any tree with the same measure
13//
14// This mirrors std::stack<T, Container>: the element type T plus an optional
15// fully-specialised backing tree. Reason for the full-specialisation approach
16// (not template-template): different finger tree variants have different
17// template signatures; taking the complete instantiation is robust to drift.

49template <typename T,
50 typename Tree = FingerTree<T, beman::transpose::sum_monoid<std::size_t>, UnitMeasure<T, std::size_t>>>
51class FingerTreeRandomAccess {
52 Tree d_tree;
53
54 public:
55 using value_type = T;
56
58 auto backing_tree() const noexcept -> const Tree& { return d_tree; }
59
60 template <class F>
61 void for_each(F&& function) const
62 requires std::invocable<F&, const T&>
63 {
64 d_tree.for_each([&](const T& value) { std::invoke(function, value); });
65 }
66
68 FingerTreeRandomAccess() : d_tree(Tree{}) {}
69
71 explicit FingerTreeRandomAccess(Tree tree) : d_tree(std::move(tree)) {}
72
74 static auto from_sequence(std::vector<T> values) -> FingerTreeRandomAccess {
75 return FingerTreeRandomAccess(Tree::from_sequence(std::move(values)));
76 }
77
79 auto size() const -> std::size_t { return d_tree.measure(); }
80
82 auto empty() const -> bool { return d_tree.is_empty(); }
83
86 auto at(std::size_t index) const -> std::optional<T> {
87 if (index >= size())
88 return std::nullopt;
89 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
90 if (!sp.has_value())
91 return std::nullopt;
92 return sp->d_pivot;
93 }
94
97 auto push_back(T value) const -> FingerTreeRandomAccess {
98 return FingerTreeRandomAccess(d_tree.snoc(std::move(value)));
99 }
100
103 auto push_front(T value) const -> FingerTreeRandomAccess {
104 return FingerTreeRandomAccess(d_tree.cons(std::move(value)));
105 }
106
109 auto insert(std::size_t index, T value) const -> FingerTreeRandomAccess {
110 auto parts = d_tree.split_at([index](std::size_t prefix) { return prefix > index; });
111 return FingerTreeRandomAccess(
112 Tree::concat(Tree::concat(parts.d_left, Tree::leaf(std::move(value))), parts.d_right));
113 }
114
118 auto erase(std::size_t index) const -> FingerTreeRandomAccess {
119 if (index >= size())
120 return *this;
121 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
122 if (!sp.has_value())
123 return *this;
124 return FingerTreeRandomAccess(Tree::concat(sp->d_left, sp->d_right));
125 }
126
130 auto update(std::size_t index, T value) const -> FingerTreeRandomAccess {
131 if (index >= size())
132 return *this;
133 auto sp = d_tree.split([index](std::size_t prefix) { return prefix > index; });
134 if (!sp.has_value())
135 return *this;
136 return FingerTreeRandomAccess(Tree::concat(sp->d_left.snoc(std::move(value)), sp->d_right));
137 }
138
140 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
141};

160template <class T, class Tree>
161struct FingerTreeRandomAccessFoldableImpl {
162 using element_type = T;
163
164 // The monoid is an instance object over the result type, passed by
165 // value; the Impl never chooses one (beman.transpose
166 // docs/transpose/decisions.md#monoid-selection).
167 template <class F, class RESULT_MONOID>
168 auto fold_map(this auto&&,
169 F&& function,
170 const beman::fingertree::tree::FingerTreeRandomAccess<T, Tree>& sequence,
171 RESULT_MONOID monoid) -> std::remove_cvref_t<std::invoke_result_t<F&, const T&>>
172 requires std::invocable<F&, const T&> &&
173 beman::transpose::monoid_object<RESULT_MONOID,
174 std::remove_cvref_t<std::invoke_result_t<F&, const T&>>>
175 {
176 using Result = std::remove_cvref_t<std::invoke_result_t<F&, const T&>>;
177 Result accumulated = monoid.identity();
178 sequence.for_each([&](const T& value) {
179 accumulated = monoid.combine(std::move(accumulated), std::invoke(function, value));
180 });
181 return accumulated;
182 }
183};
