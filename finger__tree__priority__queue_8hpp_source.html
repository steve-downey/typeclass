30template <typename T>
31struct optional_min_monoid {
32 using value_type = std::optional<T>;
33
34 auto identity() const -> std::optional<T> { return std::nullopt; }
35
36 auto combine(const std::optional<T>& lhs, const std::optional<T>& rhs) const -> std::optional<T> {
37 if (!lhs.has_value())
38 return rhs;
39 if (!rhs.has_value())
40 return lhs;
41 return *lhs <= *rhs ? lhs : rhs;
42 }
43};
44
46template <typename T>
47struct optional_max_monoid {
48 using value_type = std::optional<T>;
49
50 auto identity() const -> std::optional<T> { return std::nullopt; }
51
52 auto combine(const std::optional<T>& lhs, const std::optional<T>& rhs) const -> std::optional<T> {
53 if (!lhs.has_value())
54 return rhs;
55 if (!rhs.has_value())
56 return lhs;
57 return *lhs >= *rhs ? lhs : rhs;
58 }
59};

92template <typename T, typename Tree = FingerTree<T, PriorityMonoid<T>, PriorityMeasure<T>>>
93class FingerTreePriorityQueue {
94 Tree d_tree;
95
96 public:
97 using value_type = T;
98
99 explicit FingerTreePriorityQueue(Tree tree) : d_tree(std::move(tree)) {}
100
102 auto backing_tree() const noexcept -> const Tree& { return d_tree; }
103
104 template <class F>
105 void for_each(F&& function) const
106 requires std::invocable<F&, const T&>
107 {
108 d_tree.for_each([&](const T& value) { std::invoke(function, value); });
109 }
110
111 FingerTreePriorityQueue() : d_tree(Tree{}) {}
112
113 static auto from_values(std::vector<T> values) -> FingerTreePriorityQueue {
114 return FingerTreePriorityQueue{Tree::from_sequence(std::move(values))};
115 }
116
117 auto empty() const -> bool { return d_tree.is_empty(); }
118
120 auto size() const -> std::size_t { return std::get<0>(d_tree.measure()); }
121
122 auto min() const -> std::optional<T> { return std::get<1>(d_tree.measure()); }
123
124 auto max() const -> std::optional<T> { return std::get<2>(d_tree.measure()); }
125
126 auto push(T value) const -> FingerTreePriorityQueue {
127 return FingerTreePriorityQueue{d_tree.snoc(std::move(value))};
128 }
129
130 auto pop_min() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
131 auto current = min();
132 if (!current.has_value())
133 return std::nullopt;
134 T global_min = *current;
135 auto sp = d_tree.split([global_min](const PriorityTag<T>& p) {
136 const auto& m = std::get<1>(p);
137 return m.has_value() && *m <= global_min;
138 });
139 if (!sp.has_value())
140 return std::nullopt;
141 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
142 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
143 }
144
145 auto pop_max() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
146 auto current = max();
147 if (!current.has_value())
148 return std::nullopt;
149 T global_max = *current;
150 auto sp = d_tree.split([global_max](const PriorityTag<T>& p) {
151 const auto& m = std::get<2>(p);
152 return m.has_value() && *m >= global_max;
153 });
154 if (!sp.has_value())
155 return std::nullopt;
156 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
157 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
158 }
159
160 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
161};

167template <class T, class A, class U>
168 requires(!std::same_as<T, U>) && std::totally_ordered<U> && std::copy_constructible<U> &&
169 std::copy_constructible<A>
170struct finger_tree_rebind<tree::FingerTree<T, tree::PriorityMonoid<T>, tree::PriorityMeasure<T>, A>, U> {
171 using source_type = tree::FingerTree<T, tree::PriorityMonoid<T>, tree::PriorityMeasure<T>, A>;
172 using type = tree::FingerTree<U, tree::PriorityMonoid<U>, tree::PriorityMeasure<U>, A>;
173
174 static auto make_builder(const source_type& source) {
175 // The extrema monoids are stateless; retain the count component and
176 // allocator while rebinding the two payload-dependent components.
177 auto monoid = source.get_monoid();
178 tree::PriorityMonoid<U> rebound{
179 {std::get<0>(monoid.d_inners), tree::optional_min_monoid<U>{}, tree::optional_max_monoid<U>{}}};
180 return detail::configured_tree_builder<type>{source.get_allocator(), std::move(rebound)};
181 }
182};

196template <class T, class Tree>
197struct FingerTreePriorityQueueFoldableImpl {
198 using element_type = T;
199
200 // The monoid is an instance object over the result type, passed by
201 // value; the Impl never chooses one (beman.transpose
202 // docs/transpose/decisions.md#monoid-selection).
203 template <class F, class RESULT_MONOID>
204 auto fold_map(this auto&&,
205 F&& function,
206 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue,
207 RESULT_MONOID monoid) -> std::remove_cvref_t<std::invoke_result_t<F&, const T&>>
208 requires std::invocable<F&, const T&> &&
209 beman::transpose::monoid_object<RESULT_MONOID,
210 std::remove_cvref_t<std::invoke_result_t<F&, const T&>>>
211 {
212 using Result = std::remove_cvref_t<std::invoke_result_t<F&, const T&>>;
213 Result accumulated = monoid.identity();
214 queue.for_each([&](const T& value) {
215 accumulated = monoid.combine(std::move(accumulated), std::invoke(function, value));
216 });
217 return accumulated;
218 }
219};
