28template <typename T>
29struct optional_min_monoid {
30 using value_type = std::optional<T>;
31
32 auto identity() const -> std::optional<T> { return std::nullopt; }
33
34 auto combine(const std::optional<T>& lhs, const std::optional<T>& rhs) const -> std::optional<T> {
35 if (!lhs.has_value())
36 return rhs;
37 if (!rhs.has_value())
38 return lhs;
39 return *lhs <= *rhs ? lhs : rhs;
40 }
41};

44template <typename T>
45struct optional_max_monoid {
46 using value_type = std::optional<T>;
47
48 auto identity() const -> std::optional<T> { return std::nullopt; }

53 if (!rhs.has_value())

90template <typename T, typename Tree = FingerTree<T, PriorityMonoid<T>, PriorityMeasure<T>>>
91class FingerTreePriorityQueue {
92 Tree d_tree;
93
94 explicit FingerTreePriorityQueue(Tree tree) : d_tree(std::move(tree)) {}
95
96 public:
97 FingerTreePriorityQueue() : d_tree(Tree{}) {}
98
99 static auto from_values(std::vector<T> values) -> FingerTreePriorityQueue {
100 return FingerTreePriorityQueue{Tree::from_sequence(std::move(values))};
101 }
102
103 auto empty() const -> bool { return d_tree.is_empty(); }
104
106 auto size() const -> std::size_t { return std::get<0>(d_tree.measure()); }
107
108 auto min() const -> std::optional<T> { return std::get<1>(d_tree.measure()); }
109
110 auto max() const -> std::optional<T> { return std::get<2>(d_tree.measure()); }
111
112 auto push(T value) const -> FingerTreePriorityQueue {
113 return FingerTreePriorityQueue{d_tree.snoc(std::move(value))};
114 }
115
116 auto pop_min() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
117 auto current = min();
118 if (!current.has_value())
119 return std::nullopt;
120 T global_min = *current;
121 auto sp = d_tree.split([global_min](const PriorityTag<T>& p) {
122 const auto& m = std::get<1>(p);
123 return m.has_value() && *m <= global_min;
124 });
125 if (!sp.has_value())
126 return std::nullopt;
127 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
128 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
129 }
130
131 auto pop_max() const -> std::optional<std::pair<T, FingerTreePriorityQueue>> {
132 auto current = max();
133 if (!current.has_value())
134 return std::nullopt;
135 T global_max = *current;
136 auto sp = d_tree.split([global_max](const PriorityTag<T>& p) {
137 const auto& m = std::get<2>(p);
138 return m.has_value() && *m >= global_max;
139 });
140 if (!sp.has_value())
141 return std::nullopt;
142 return std::pair<T, FingerTreePriorityQueue>{sp->d_pivot,
143 FingerTreePriorityQueue{Tree::concat(sp->d_left, sp->d_right)}};
144 }
145
146 auto to_vector() const -> std::vector<T> { return d_tree.flatten(); }
147};

153template <class T, class Tree>
154struct FingerTreePriorityQueueFoldableImpl {
155 using element_type = T;
156
157 // The monoid is an instance object over the result type, passed by
158 // value; the Impl never chooses one (beman.transpose
159 // docs/transpose/decisions.md#monoid-selection).
160 template <class F, class RESULT_MONOID>
161 auto fold_map(this auto&&,
162 F&& function,
163 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue,
164 RESULT_MONOID monoid) -> std::remove_cvref_t<std::invoke_result_t<F&, const T&>> {
165 using Result = std::remove_cvref_t<std::invoke_result_t<F&, const T&>>;
166 return std::ranges::fold_left(
167 queue.to_vector(), Result(monoid.identity()), [&](Result acc, const auto& value) {
168 return monoid.combine(std::move(acc), std::invoke(function, value));
169 });
170 }
171};

193 template <class APPLICATIVE, class F>
194 auto traverse(this auto&&,
195 const APPLICATIVE& applicative,
196 F&& function,
197 const beman::fingertree::tree::FingerTreePriorityQueue<T, Tree>& queue)
198 requires std::invocable<F&, const T&> &&
199 beman::transpose::applicative_object_for<APPLICATIVE,
200 std::remove_cvref_t<std::invoke_result_t<F&, const T&>>>
201 {
202 using Context = std::remove_cvref_t<std::invoke_result_t<F&, const T&>>;
203 using U = beman::transpose::applicative_value_t<Context>;
204
205 auto accumulated = applicative.pure(std::vector<U>{});
206
207 for (const auto& value : queue.to_vector()) {
208 auto lifted = std::invoke(function, value);
209 accumulated = applicative.invoke(
210 [](std::vector<U> values, U element) {
211 values.push_back(std::move(element));
212 return values;
213 },
214 std::move(accumulated),
215 std::move(lifted));
216 }
217
218 return applicative.invoke(
219 [](std::vector<U> values) {
220 return beman::fingertree::tree::FingerTreePriorityQueue<U>::from_values(std::move(values));
221 },
222 std::move(accumulated));
223 }
224};
