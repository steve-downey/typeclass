583// -- Operation objects --
584//
585// See functor.hpp for the shape and the motivation. `traverse` was a free
586// function template doing this lookup-and-call; as an object it is not found
587// by argument-dependent lookup and cannot be overloaded, which is the
588// property that makes a lookup-based verb predictable. Its constraints are
589// the ones the function template carried, so a structure or context with no
590// instance is still a constraint that does not match rather than a
591// diagnostic from inside the body.
592
595struct traverse_fn {
596 template <class F, class T,
597 class POLICY = std::remove_cvref_t<
598 decltype(applicative<traverse_context_t<F, T>>)>>
599 requires applicative_object_for<POLICY, traverse_context_t<F, T>>
600 constexpr auto operator()(F &&function, T &&value,
601 POLICY policy = POLICY{}) const {
602 const auto &map = traversable<std::remove_cvref_t<T>>;
603 return map.traverse(policy, std::forward<F>(function),
604 std::forward<T>(value));
605 }
606};
607
608// `for_each` gets no operation object. The derived operation is `traverse`
609// with the arguments the other way round, so an object would be a second name
610// for the same call. (Sitting beside std::for_each is not the reason: `invoke`
611// sits beside std::invoke and keeps its object; see
612// docs/transpose/decisions.md#operation-object-wording.)
613
614// \rSec3[transpose.alg.traverse]{traverse}
615
646inline constexpr traverse_fn traverse{};
