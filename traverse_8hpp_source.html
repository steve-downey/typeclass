520// -- Operation objects --
521//
522// See functor.hpp for the shape and the motivation. `traverse` was a free
523// function template doing this lookup-and-call; as an object it is not found
524// by argument-dependent lookup and cannot be overloaded, which is the
525// property that makes a lookup-based verb predictable. Its constraints are
526// the ones the function template carried, so a structure or context with no
527// instance is still a constraint that does not match rather than a
528// diagnostic from inside the body.
529
532struct traverse_fn {
533 template <class F, class T,
534 class POLICY = std::remove_cvref_t<
535 decltype(applicative<traverse_context_t<F, T>>)>>
536 requires applicative_object_for<POLICY, traverse_context_t<F, T>>
537 constexpr auto operator()(F &&function, T &&value,
538 POLICY policy = POLICY{}) const {
539 const auto &map = traversable<std::remove_cvref_t<T>>;
540 return map.traverse(policy, std::forward<F>(function),
541 std::forward<T>(value));
542 }
543};
544
545// `for_each` gets no operation object. The derived operation is `traverse`
546// with the arguments the other way round, so an object would be a second name
547// for the same call. (Sitting beside std::for_each is not the reason: `invoke`
548// sits beside std::invoke and keeps its object; see
549// docs/transpose/decisions.md#operation-object-wording.)
550
551// \rSec3[transpose.alg.traverse]{traverse}
552
583inline constexpr traverse_fn traverse{};
