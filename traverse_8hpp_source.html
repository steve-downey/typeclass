628// -- Operation objects --
629//
630// See functor.hpp for the shape and the motivation. `traverse` was a free
631// function template doing this lookup-and-call; as an object it is not found
632// by argument-dependent lookup and cannot be overloaded, which is the
633// property that makes a lookup-based verb predictable. Its constraints are
634// the ones the function template carried, so a structure or context with no
635// instance is still a constraint that does not match rather than a
636// diagnostic from inside the body.
637
640struct traverse_fn {
641 template <class F, class T,
642 class POLICY = std::remove_cvref_t<
643 decltype(applicative<traverse_context_t<F, T>>)>>
644 requires applicative_object_for<POLICY, traverse_context_t<F, T>> &&
645 requires(POLICY &policy) {
646 traversable<std::remove_cvref_t<T>>.traverse(
647 policy, std::declval<F>(), std::declval<T>());
648 }
649 [[nodiscard]] constexpr auto operator()(F &&function, T &&value,
650 POLICY policy = POLICY{}) const
651 noexcept(noexcept(traversable<std::remove_cvref_t<T>>.traverse(
652 policy, std::forward<F>(function), std::forward<T>(value)))) {
653 const auto &map = traversable<std::remove_cvref_t<T>>;
654 return map.traverse(policy, std::forward<F>(function),
655 std::forward<T>(value));
656 }
657};

659// `for_each` gets no operation object. The derived operation is `traverse`
660// with the arguments the other way round, so an object would be a second name
661// for the same call. (Sitting beside std::for_each is not the reason: `invoke`
662// sits beside std::invoke and keeps its object; see
663// docs/transpose/decisions.md#operation-object-wording.)

704// `transpose` flips a structure of contextual values into a single contextual
705// value of the structure, preserving shape:
706//
707// structure<context<T>> -> context<structure<T>>
708//
709// It is `traverse` with the identity function; the context's applicative is
710// inferred from the structure's element type.
711
714struct transpose_fn {
715 template <class T>
716 requires transposable_structure<T> && requires {
717 traversable<std::remove_cvref_t<T>>.transpose(std::declval<T>());
718 }
719 [[nodiscard]] constexpr auto operator()(T &&value) const
720 noexcept(noexcept(traversable<std::remove_cvref_t<T>>.transpose(
721 std::forward<T>(value)))) {
722 const auto &map = traversable<std::remove_cvref_t<T>>;
723 return map.transpose(std::forward<T>(value));
724 }
725};
