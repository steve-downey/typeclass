5// Structural access to a FingerTree: the representation as it is.
6//
7// The sequence interface of FingerTree deliberately hides the tree's shape;
8// cons, snoc, split and append are free to rebalance it. A structural
9// algorithm needs the opposite: the real Empty/Single/Deep forms of the
10// spine and the real Leaf/Node2/Node3 forms of its elements, with digit
11// order and spine position preserved. This header is that access and nothing
12// else. It names no algorithm library, so any structural consumer can be
13// written against it; beman.tree_algorithms is reached through the optional
14// beman/typeclass_tree_adapters/finger_tree.hpp.
15//
16// The representation has two recursive sorts, and both are the same C++
17// type at every depth:
18//
19// FingerTreeLevel<TREE> the tree at one level of the spine. Level 0 is
20// the sequence; level d holds nodes of height d.
21// FingerTreeNode<T, MONOID> one element: a Leaf (height 0) or a Node2 or
22// Node3 of nodes one lower.
23//
24// Both are owning values. They share the immutable structure they describe,
25// so copying one is O(1), no payload is copied, and neither can outlive what
26// it refers to. Reading a level or a node hands out more of the same.
27//
28// FingerTreeStructureBuilder<TREE> constructs the same forms under one
29// allocator and one monoid instance, and computes every cached measure. A
30// level that is not a sequence never escapes as a TREE: sequence() is the
31// only way out, and its precondition is that the level is one.

56enum class FingerTreeForm : std::uint8_t { EMPTY, SINGLE, DEEP };

59enum class FingerTreeNodeForm : std::uint8_t { LEAF, NODE2, NODE3 };

68template <typename T, typename MONOID>
69class FingerTreeNode {
70 using E = ft::Elem<T, MONOID>;
71 using EP = ft::ElemPtr<T, MONOID>;
72
73 EP d_elem;
74 std::size_t d_height;
75
76 FingerTreeNode(EP elem, std::size_t height) noexcept : d_elem(std::move(elem)), d_height(height) {}
77
78 template <typename>
79 friend class FingerTreeLevel;
80 template <typename>
81 friend class FingerTreeStructureBuilder;
82
83 public:
84 using value_type = T;
85 using tag_type = ft::tag_t<MONOID>;
86 using monoid_type = MONOID;
87 using children_type = std::inplace_vector<FingerTreeNode, 3>;
88
89 [[nodiscard]] auto form() const noexcept -> FingerTreeNodeForm {
90 return static_cast<FingerTreeNodeForm>(d_elem->d_data.index());
91 }
92
94 [[nodiscard]] auto height() const noexcept -> std::size_t { return d_height; }
95
97 [[nodiscard]] auto measure() const noexcept -> const tag_type& { return d_elem->d_measure; }
98
100 [[nodiscard]] auto value() const noexcept -> const T& {
101 const auto* leaf = std::get_if<typename E::Leaf>(&d_elem->d_data);
102 assert(leaf != nullptr && "FingerTreeNode::value: not a leaf");
103 return leaf->d_value;
104 }
105
109 [[nodiscard]] auto share_value() const noexcept -> std::shared_ptr<const T> {
110 return std::shared_ptr<const T>(d_elem, std::addressof(value()));
111 }
112
115 [[nodiscard]] auto children() const -> children_type {
116 children_type out;
117 if (const auto* n2 = std::get_if<typename E::Node2>(&d_elem->d_data)) {
118 out.push_back(FingerTreeNode(n2->a, d_height - 1));
119 out.push_back(FingerTreeNode(n2->b, d_height - 1));
120 } else if (const auto* n3 = std::get_if<typename E::Node3>(&d_elem->d_data)) {
121 out.push_back(FingerTreeNode(n3->a, d_height - 1));
122 out.push_back(FingerTreeNode(n3->b, d_height - 1));
123 out.push_back(FingerTreeNode(n3->c, d_height - 1));
124 }
125 return out;
126 }
127};

135template <typename TREE>
136class FingerTreeLevel {
137 static_assert(ft::is_finger_tree<TREE>, "FingerTreeLevel requires a FingerTree specialization");
138
139 TREE d_tree;
140 std::size_t d_level = 0;
141
142 FingerTreeLevel(TREE tree, std::size_t level) : d_tree(std::move(tree)), d_level(level) {}
143
144 template <typename>
145 friend class FingerTreeStructureBuilder;
146
147 public:
148 using tree_type = TREE;
149 using value_type = typename TREE::value_type;
150 using tag_type = typename TREE::tag_type;
151 using monoid_type = typename TREE::monoid_type;
152 using allocator_type = typename TREE::allocator_type;
153 using node_type = FingerTreeNode<value_type, monoid_type>;
154 using digit_type = std::inplace_vector<node_type, 4>;
155
157 explicit FingerTreeLevel(const TREE& sequence) : d_tree(sequence) {}
158
159 [[nodiscard]] auto form() const noexcept -> FingerTreeForm {
160 return static_cast<FingerTreeForm>(d_tree.d_repr.index());
161 }
162
163 [[nodiscard]] auto empty() const noexcept -> bool { return d_tree.empty(); }
164
167 [[nodiscard]] auto level() const noexcept -> std::size_t { return d_level; }
168
170 [[nodiscard]] auto measure() const -> tag_type { return d_tree.measure(); }
171
173 [[nodiscard]] auto element() const -> node_type {
174 const auto* single = std::get_if<typename TREE::Single>(&d_tree.d_repr);
175 assert(single != nullptr && "FingerTreeLevel::element: not a Single");
176 return node_type(single->d_elem, d_level);
177 }
178
181 [[nodiscard]] auto left() const -> digit_type { return digit(deep().d_left); }
182
185 [[nodiscard]] auto right() const -> digit_type { return digit(deep().d_right); }
186
189 [[nodiscard]] auto spine() const -> FingerTreeLevel {
190 const auto& d = deep();
191 if (d.d_spine)
192 return FingerTreeLevel(*d.d_spine, d_level + 1);
193 return FingerTreeLevel(TREE(d_tree.get_allocator(), d_tree.get_monoid()), d_level + 1);
194 }
195
198 [[nodiscard]] auto is_sequence() const noexcept -> bool { return empty() || d_level == 0; }
199
201 [[nodiscard]] auto sequence() const& -> TREE {
202 assert(is_sequence() && "FingerTreeLevel::sequence: an inner spine level is not a sequence");
203 return d_tree;
204 }
205
206 [[nodiscard]] auto sequence() && -> TREE {
207 assert(is_sequence() && "FingerTreeLevel::sequence: an inner spine level is not a sequence");
208 return std::move(d_tree);
209 }
210
211 private:
212 auto deep() const -> const typename TREE::Deep& {
213 const auto* d = std::get_if<typename TREE::DeepPtr>(&d_tree.d_repr);
214 assert(d != nullptr && "FingerTreeLevel: not a Deep");
215 return **d;
216 }
217
218 auto digit(const typename TREE::Digit& source) const -> digit_type {
219 digit_type out;
220 for (const auto& elem : source)
221 out.push_back(node_type(elem, d_level));
222 return out;
223 }
224};

235template <typename TREE>
236class FingerTreeStructureBuilder {
237 static_assert(ft::is_finger_tree<TREE>, "FingerTreeStructureBuilder requires a FingerTree specialization");
238
239 public:
240 using tree_type = TREE;
241 using value_type = typename TREE::value_type;
242 using monoid_type = typename TREE::monoid_type;
243 using allocator_type = typename TREE::allocator_type;
244 using level_type = FingerTreeLevel<TREE>;
245 using node_type = typename level_type::node_type;
246 using digit_type = typename level_type::digit_type;
247
249 explicit FingerTreeStructureBuilder(const TREE& prototype)
250 : d_alloc(prototype.get_allocator()), d_monoid(prototype.get_monoid()) {}
251
252 explicit FingerTreeStructureBuilder(const allocator_type& alloc = allocator_type{},
253 const monoid_type& monoid = monoid_type{})
254 : d_alloc(alloc), d_monoid(monoid) {}
255
256 [[nodiscard]] auto get_allocator() const noexcept -> allocator_type { return d_alloc; }
257 [[nodiscard]] auto get_monoid() const noexcept -> monoid_type { return d_monoid; }
258
260 [[nodiscard]] auto leaf(value_type value) const -> node_type {
261 return node_type(ft::make_leaf<value_type, monoid_type>(d_alloc, TREE::meas_fn(), std::move(value)), 0);
262 }
263
265 [[nodiscard]] auto node2(node_type a, node_type b) const -> node_type {
266 assert(a.d_height == b.d_height && "FingerTreeStructureBuilder::node2: children differ in height");
267 const auto height = a.d_height + 1;
268 return node_type(
269 ft::make_node2<value_type, monoid_type>(d_alloc, d_monoid, std::move(a.d_elem), std::move(b.d_elem)),
270 height);
271 }
272
274 [[nodiscard]] auto node3(node_type a, node_type b, node_type c) const -> node_type {
275 assert(a.d_height == b.d_height && b.d_height == c.d_height &&
276 "FingerTreeStructureBuilder::node3: children differ in height");
277 const auto height = a.d_height + 1;
278 return node_type(ft::make_node3<value_type, monoid_type>(
279 d_alloc, d_monoid, std::move(a.d_elem), std::move(b.d_elem), std::move(c.d_elem)),
280 height);
281 }
282
284 [[nodiscard]] auto empty() const -> level_type { return level_type(TREE(d_alloc, d_monoid), 0); }
285
287 [[nodiscard]] auto single(node_type element) const -> level_type {
288 const auto level = element.d_height;
289 return level_type(TREE::make_single(d_alloc, d_monoid, std::move(element.d_elem)), level);
290 }
291
299 [[nodiscard]] auto deep(digit_type left, level_type spine, digit_type right) const -> level_type {
300 assert(!left.empty() && !right.empty() && "FingerTreeStructureBuilder::deep: empty digit");
301 const auto level = left.front().d_height;
302 assert(uniform(left, level) && uniform(right, level) &&
303 "FingerTreeStructureBuilder::deep: digit nodes differ in height");
304 assert((spine.empty() || spine.d_level == level + 1) &&
305 "FingerTreeStructureBuilder::deep: spine is not one level below the digits");
306
307 typename TREE::SpinePtr next;
308 if (!spine.empty()) {
309 assert(TREE::alloc_equal(d_alloc, spine.d_tree.get_allocator()) &&
310 "FingerTreeStructureBuilder::deep: spine was allocated by another resource");
311 next = TREE::allocate_spine(d_alloc, std::move(spine.d_tree));
312 }
313 return level_type(
314 TREE::make_deep(d_alloc, d_monoid, elems(std::move(left)), std::move(next), elems(std::move(right))),
315 level);
316 }
317
318 private:
319 [[no_unique_address]] allocator_type d_alloc;
320 [[no_unique_address]] monoid_type d_monoid;
321
322 static auto uniform(const digit_type& digit, std::size_t height) noexcept -> bool {
323 for (const auto& node : digit) {
324 if (node.d_height != height)
325 return false;
326 }
327 return true;
328 }
329
330 static auto elems(digit_type&& digit) -> typename TREE::Digit {
331 typename TREE::Digit out;
332 for (auto& node : digit)
333 out.push_back(std::move(node.d_elem));
334 return out;
335 }
336};
