6// Non-normative demonstration type. sender models a deferred/asynchronous
7// computational context for Paper A's second motivating domain. It is a
8// deliberately minimal stand-in for a std::execution sender (P2300): a lazy
9// value whose computation runs only when get() is called. Traversing a
10// structure of senders transposes into a single sender of the structure, e.g.
11// vector<sender<T>> becomes sender<vector<T>>, with no work performed until the
12// resulting sender is run. It is illustrative, not a proposed standard type.
13//
14// WHAT IT SHOWS, PRECISELY. That the front door preserves laziness for a
15// copyable, single-value deferred computation, and that it does so through
16// the same traverse and transpose a caller uses for optional. That is the
17// whole of it. This type has no completion signatures, no error or stopped
18// channel, no environment or scheduler, no one-shot operation state -- none
19// of what makes a P2300 sender a sender. The Applicative surface is checked
20// against genuine senders separately, at examples/transpose/p2300_adapter.hpp.
21//
22// AND WHAT IT HIDES. Being a std::function<T()> wrapper makes this type
23// invariant under composition: sender<vector<T>> composed with sender<T> is
24// again sender<vector<T>>, so the vector traversal's accumulator loop
25// type-checks. A real sender is not invariant -- every combinator names a new
26// type -- so transposing a runtime-sized structure of real senders is not
27// merely slower here, it cannot be written as a loop at all. The convenience
28// that makes this a good demonstration is the same property that conceals
29// that limit. See docs/transpose/decisions.md#p2300-front-door-shape.

50template <class T>
51struct sender {
52 using value_type = T;
53
54 std::function<T()> run;
55
56 sender() = default;
57
59 template <class F>
60 requires(!std::same_as<std::remove_cvref_t<F>, sender>) &&
61 std::invocable<const std::remove_cvref_t<F> &>
62 explicit sender(F &&work) : run(box(std::forward<F>(work))) {}
63
65 auto get() const -> T { return run(); }
66
67 private:
68 template <class F>
69 static auto box(F &&work) -> std::function<T()> {
70 using WORK = std::remove_cvref_t<F>;
71 if constexpr (std::copy_constructible<WORK>) {
72 return std::function<T()>(std::forward<F>(work));
73 } else {
74 return [shared = std::make_shared<const WORK>(
75 std::forward<F>(work))]() -> T { return (*shared)(); };
76 }
77 }
78
79 public:
81 static auto ready(T value) -> sender {
82 return sender{[value = std::move(value)]() -> T { return value; }};
83 }
84
86 friend auto operator==(const sender &left, const sender &right) -> bool
87 requires requires(const T &a, const T &b) {
88 { a == b } -> std::convertible_to<bool>;
89 }
90 {
91 return left.get() == right.get();
92 }
93};

103template <class T>
104struct SenderApplicativeImpl {
105 template <class VALUE>
106 auto pure(this auto &&, VALUE &&value) {
107 using U = std::remove_cvref_t<VALUE>;
108 return sender<U>::ready(U(std::forward<VALUE>(value)));
109 }

133 template <class FUNCTION, class FIRST, class... REST>
134 auto invoke(this auto &&, FUNCTION &&function, sender<FIRST> first,
135 sender<REST>... rest)
136 -> sender<std::remove_cvref_t<std::invoke_result_t<
137 const std::remove_cvref_t<FUNCTION> &, FIRST, REST...>>>
138 requires std::constructible_from<std::remove_cvref_t<FUNCTION>,
139 FUNCTION> &&
140 detail::holdable_value<std::invoke_result_t<
141 const std::remove_cvref_t<FUNCTION> &, FIRST, REST...>>
142 {
143 using U = std::remove_cvref_t<std::invoke_result_t<
144 const std::remove_cvref_t<FUNCTION> &, FIRST, REST...>>;
145 return sender<U>{[function = std::remove_cvref_t<FUNCTION>(
146 std::forward<FUNCTION>(function)),
147 first = std::move(first),
148 ... rest = std::move(rest)]() -> U {
149 // Running the operands as arguments to std::invoke would leave
150 // their relative order unspecified -- the calls are
151 // indeterminately sequenced, and GCC 16 runs them in reverse.
152 // That cannot implement the composition order the traversal
153 // promises. The initializer-clauses of a braced-init-list are
154 // sequenced left to right ([dcl.init.list]/4), so running the
155 // operands into a tuple first pins the order, and std::apply
156 // then applies the plain function to the results.
157 std::tuple<FIRST, REST...> values{first.get(), rest.get()...};
158 return std::apply(
159 [&function](auto &&...value) -> U {
160 return std::invoke(function,
161 std::forward<decltype(value)>(value)...);
162 },
163 std::move(values));
164 }};
165 }
166};
