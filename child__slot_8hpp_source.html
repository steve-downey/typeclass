25// Child storage for base-functor layers: indirection exactly at the knot.
26//
27// A base functor F needs indirection at its recursive positions only when
28// they hold the fixed point itself: F<Fix<F>> contains children of an
29// incomplete type, so those must sit behind a pointer (Box). Every other
30// instantiation of F — the F<Result> a fold hands to its algebra, the
31// F<Seed> a coalgebra produces — is at a complete type, and its children
32// can live inline on the stack. A layer type that hardwires Box pays one
33// heap allocation per child per node in every materialized layer, for
34// nothing the algorithm needs; measured on a 2^20-node fold, that
35// allocation is the entire difference between fold_fix and hand-written
36// recursion.
37//
38// child_slot chooses per position: Box at Fix<F> (matched by partial
39// specialization, which needs only Fix's declaration, not completeness),
40// std::optional everywhere else. Both are nullable, contextually
41// convertible to bool, and dereference with *, so layer code reads the
42// same either way; make_slot constructs whichever the position requires.
43
44// 9dd7168f-4960-40f1-9366-db7d88caec2c
53template <typename A>
54struct child_slot {
55 using type = std::optional<A>;

60template <template <typename> class F>
61struct child_slot<Fix<F> > {
62 using type = Box<Fix<F> >;
63};

65template <typename A>
66using child_slot_t = typename child_slot<A>::type;

72template <typename A, typename... Args>
73constexpr auto make_slot(Args&&... args) -> child_slot_t<A> {
74 if constexpr (std::is_same_v<child_slot_t<A>, Box<A> >) {
75 return make_box<A>(std::forward<Args>(args)...);
76 } else {
77 return std::optional<A>(std::in_place, std::forward<Args>(args)...);
78 }
79}
