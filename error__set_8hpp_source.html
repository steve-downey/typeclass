6// error_set: the set of error types a computation may raise.
7//
8// NOMINAL, not a structural sum. `error_set_of<Es...>` is its own type. The
9// public surface is deliberately impoverished -- injection, subsumption,
10// membership, visitation -- because "the set of errors this computation may
11// raise" is an interpretation, not a representation, and interpretations
12// need names. See docs/transpose/decisions.md#error-set-identity.
13//
14// VALUE LEVEL: a WITNESSED SUBSET, not a one-of union. The TYPE says "may
15// raise {A,B}"; a VALUE says "did raise", and did-raise is a non-empty subset
16// of may-raise: one witness per raised type, at most. Storage is per-type
17// slots (a tuple of `std::optional<Es>...`) bounded by |grade|, never
18// allocation, never a discriminated one-of. This is the amendment recorded
19// at docs/transpose/decisions.md#error-set-identity, made by
20// docs/transpose/decisions.md#accumulation-evidence: the short-circuit
21// applicative object only ever produces a singleton witness (exactly the old
22// "one alternative" behavior, preserved as a special case); the accumulating
23// object (docs/transpose/transpose-grading-plan.md#accumulating-object)
24// combines witnesses from multiple failures, left-biased per type via
25// `error_set_combine` -- two witnesses of the SAME type keep the leftmost,
26// which is deterministic only because left-to-right traversal is already
27// normative (docs/transpose/decisions.md#applicative-objects).
28//
29// CANONICAL BY CONSTRUCTION. `error_set<Es...>` is the spelling users and the
30// framework write; it is an alias that sorts and deduplicates its pack, so
31// `error_set<A,B>` and `error_set<B,A>` are THE SAME TYPE. That is what makes
32// grade arithmetic well-defined and inclusions unique -- sorted normalization
33// is names-not-positions. `error_set_of` accepts only an already-canonical
34// pack and says so with a static_assert; forming a non-canonical one is the
35// invariant violation the class exists to prevent.
36//
37// The conversions are exactly the ⊆ widenings and nothing else. There is no
38// narrowing conversion, no conversion between incomparable sets, and no
39// accessor that hands out the underlying variant. Having only the subset
40// conversions is the coherence argument compiled into the overload set.
41//
42// TYPE ORDERING is P2830's job. Until it is available the canonical set uses
43// an interim ordering read out of the compiler's own function signature,
44// valid only for the documented restrictions: named types, external linkage,
45// stable spelling. Two distinct types that render to the same name would be
46// silently conflated, so that case is a hard error rather than a wrong
47// answer -- see the static_assert in error_set_of. The set machinery and the
48// order live in detail/canonical_set.hpp, which has no error vocabulary
49// (docs/transpose/decisions.md#canonical-set-home); this header is its
50// first client.
51//
52// THE ORDER IS PART OF THE ABI. `error_set_of` stores its witnesses as a
53// tuple in canonical order, so its layout follows the order in use. It is
54// therefore declared in the inline namespace `ordered_by_type_name`, the
55// same key the order itself carries, so that the day a P2830 order exists a
56// translation unit built against it cannot be linked with one built against
57// the name order -- the mismatch fails at link time rather than violating
58// the ODR silently. User code and the wording spell `error_set<...>` and
59// `error_set_of<...>` as before; the key is part of the type's identity, not
60// of its spelling. See docs/transpose/decisions.md#type-order-abi.

118// -- Per-type witness storage ----------------------------------------------
119// One std::optional slot per alternative, never a discriminated one-of. See
120// docs/transpose/decisions.md#accumulation-evidence for why: the grade join
121// needs no monoid, but accumulating VALUE-level evidence does, and the
122// semigroup it needs is "per type, keep one witness" -- which a tuple of
123// optionals gives for free, with no allocation and no extra discriminant to
124// keep in sync.
125
131template <class ERROR, class INJECTED>
132constexpr auto witness_slot(INJECTED &&error) noexcept(
133 std::is_nothrow_constructible_v<ERROR, INJECTED>) -> std::optional<ERROR> {
134 if constexpr (std::is_same_v<ERROR, std::remove_cvref_t<INJECTED>>) {
135 return std::optional<ERROR>(std::forward<INJECTED>(error));
136 } else {
137 return std::optional<ERROR>{};
138 }
139}

145template <class ERROR, class... NARROWER>
146constexpr auto widen_slot(const error_set_of<NARROWER...> &narrower) noexcept(
147 std::is_nothrow_copy_constructible_v<ERROR>) -> std::optional<ERROR> {
148 if constexpr (error_set_of<NARROWER...>::template contains<ERROR>()) {
149 return narrower.template witness<ERROR>();

151 return std::optional<ERROR>{};

220template <class... ERRORS>
221class error_set_of {
222 static_assert(
223 detail::error_set_is_canonical_v<ERRORS...>,
224 "error_set_of requires a canonical (sorted, deduplicated) pack. "
225 "Spell error_set<...> instead, which canonicalizes for you.");
226 static_assert(
227 detail::error_set_names_distinct_v<ERRORS...>,
228 "Two distinct error types render to the same name, so the interim "
229 "type ordering cannot separate them. The P2830 fallback is valid "
230 "only for named types with external linkage and a stable spelling; "
231 "this is outside those restrictions.");

290
292 template <std::size_t... IDX>
293 constexpr auto combine_witnesses(const error_set_of &other,
294 std::index_sequence<IDX...>) const
295 noexcept((std::is_nothrow_copy_constructible_v<ERRORS> && ...))
296 -> std::tuple<std::optional<ERRORS>...> {
297 return std::tuple<std::optional<ERRORS>...>{
298 (std::get<IDX>(d_witnesses).has_value()
299 ? std::get<IDX>(d_witnesses)
300 : std::get<IDX>(other.d_witnesses))...};
301 }
302
304 constexpr auto to_variant() const -> std::variant<ERRORS...> {
305 if (witness_count() != 1) {
306 detail::error_set_precondition_violated(
307 "beman::transpose::error_set::visit requires exactly one "
308 "witness, but this value witnesses a different number. Use "
309 "witness<E>() per alternative for a multi-witness subset; "
310 "see docs/transpose/decisions.md#accumulation-evidence.");
311 }
312
313 std::optional<std::variant<ERRORS...>> found;
314 auto try_slot = [&found](const auto &slot) {
315 if (!found.has_value() && slot.has_value()) {
316 found = std::variant<ERRORS...>{*slot};
317 }
318 };
319 std::apply(
320 [&try_slot](const auto &...slots) { (try_slot(slots), ...); },
321 d_witnesses);
322 return *found;
323 }
324
326 std::tuple<std::optional<ERRORS>...> d_witnesses;
327};

567// -- recover: narrowing the grade by set difference ------------------------
568// Stage recover-narrowing
569// (docs/transpose/transpose-grading-plan.md#recover-narrowing). Error
570// vocabulary, per grade.hpp's own comment that `recover` lives in the model's
571// header, not the framework's.
572//
573// docs/transpose/decisions.md#recover-grade-inference: the handled set {H} is
574// ANNOTATED -- explicit template arguments, never inferred from the
575// handler's shape -- and CHECKED: one static_assert enforces H is a subset
576// of the computation's grade (reusing error_set_of::contains, the same
577// membership primitive docs/transpose/decisions.md#multi-witness-elimination
578// names), another enforces the handler's result shape. The resulting grade
579// `(e \ H) ∪ raised` is then a direct, non-recursive computation -- no
580// lattice fixpoint, because one recover call is not a recursive fold
581// ("inference only where no fixpoint is required").
582//
583// docs/transpose/decisions.md#multi-witness-elimination: recover needs no new
584// public verb. Elimination is a static fold over H with runtime presence
585// filtering, built entirely from the existing public API --
586// error_set_of's injecting constructor and `combined_with` -- never from
587// private access to the representation.

659// -- Value level: carrying witnesses forward through the PUBLIC API only ---
660// No private access, no friendship: exactly the injecting constructor and
661// `combined_with` that any user of error_set already has. This is what makes
662// the multi-witness-elimination fence a fence rather than an oversight --
663// recover's mechanism is not privileged.
664
670template <class E, class HANDLED_LIST, class TARGET, class SOURCE>
671constexpr void carry_witness(std::optional<TARGET> &accumulated,
672 const SOURCE &source) {
673 if constexpr (!list_contains_v<E, HANDLED_LIST> &&
674 TARGET::template contains<E>()) {
675 if (source.template holds<E>()) {
676 TARGET injected(*source.template witness<E>());
677 accumulated = accumulated.has_value()
678 ? accumulated->combined_with(injected)
679 : injected;
680 }
681 }
682}

687template <class TARGET, class VALUE, class... RS>
688constexpr void
689carry_raised(std::optional<TARGET> &accumulated,
690 const std::expected<VALUE, error_set_of<RS...>> &result) {
691 (carry_witness<RS, type_list<>>(accumulated, result.error()), ...);

698template <class H, class HANDLER, class VALUE, class TARGET, class SOURCE>
699constexpr void apply_handler(HANDLER &handler, std::optional<VALUE> &recovered,
700 std::optional<TARGET> &accumulated,
701 const SOURCE &error) {
702 if (error.template holds<H>()) {
703 auto result = std::invoke(handler, *error.template witness<H>());
704 if constexpr (std::is_same_v<std::remove_cvref_t<decltype(result)>,
705 VALUE>) {
706 if (!recovered.has_value()) {
707 recovered = std::move(result);
708 }
709 } else {
710 if (result.has_value()) {
711 if (!recovered.has_value()) {
712 recovered = *result;
713 }
714 } else {
715 carry_raised(accumulated, result);
716 }
717 }
718 }
719}

807template <class... HANDLED, class VALUE, class... ERRORS, class HANDLER>
808 requires(sizeof...(ERRORS) > 0)
809[[nodiscard]] constexpr auto
810recover(const std::expected<VALUE, error_set_of<ERRORS...>> &computation,
811 HANDLER &&
812 handler) noexcept(detail::recover_nothrow_v<HANDLER, VALUE,
813 error_set_of<ERRORS...>,
814 HANDLED...>) ->
815 typename std::enable_if_t<
816 (sizeof...(ERRORS) > 0),
817 detail::recover_return<HANDLER, VALUE, error_set_of<ERRORS...>,
818 HANDLED...>>::type {
819 static_assert(sizeof...(HANDLED) > 0,
820 "recover needs at least one HANDLED type.");
821 static_assert(
822 (error_set_of<ERRORS...>::template contains<HANDLED>() && ...),
823 "recover's HANDLED types must be members of the computation's grade "
824 "-- CHECKED, per docs/transpose/decisions.md#recover-grade-inference. "
825 "The "
826 "handled set is ANNOTATED (explicit template arguments on recover), "
827 "never inferred.");
828 static_assert(
829 (detail::is_valid_recover_result_v<
830 typename detail::handler_result_of<HANDLER, VALUE, HANDLED>::type,
831 VALUE> &&
832 ...),
833 "recover's handler must return either the recovered VALUE directly, "
834 "or std::expected<VALUE, error_set<...>> if the recovery attempt can "
835 "itself raise. See "
836 "docs/transpose/decisions.md#recover-grade-inference.");
837
838 using Grade = error_set_of<ERRORS...>;
839 using FinalGrade =
840 detail::recover_final_grade_t<HANDLER, VALUE, Grade, HANDLED...>;
841 using Return = detail::recover_return_t<HANDLER, VALUE, Grade, HANDLED...>;
842
843 if (computation.has_value()) {
844 return Return(*computation);
845 }
846
847 const Grade &error = computation.error();
848 std::optional<VALUE> recovered;
849 std::optional<FinalGrade> accumulated;
850
851 (detail::carry_witness<ERRORS, detail::type_list<HANDLED...>>(accumulated,
852 error),
853 ...);
854 (detail::apply_handler<HANDLED>(handler, recovered, accumulated, error),
855 ...);
856
857 if constexpr (std::is_same_v<Return, VALUE>) {
858 // FinalGrade is ∅: every carry_witness/carry_raised call above was a
859 // no-op by construction (TARGET::contains<E>() is unconditionally
860 // false for error_set_of<>), so `accumulated` can never actually hold
861 // a value here -- every present witness was necessarily HANDLED and
862 // recovered.
863 return *recovered;
864 } else {
865 if (accumulated.has_value()) {
866 return Return(std::unexpect, *accumulated);
867 }
868 return Return(*recovered);
869 }
870}

872// -- recover at the empty grade: vacuous ----------------------------------
873// docs/transpose/decisions.md#generic-closure-at-empty-grade. Generic code
874// that recovers from E must stay well-formed where an upstream stage has
875// proved E impossible. At the empty grade there is no witness to hand a
876// handler, so the operation is the identity on its operand, and asks nothing
877// of the handler -- not membership of HANDLED in the grade, which is empty,
878// and not that the handler be callable. The handled set is still annotated;
879// what is waived is the check, and only here, where the grade itself says
880// there is nothing to check
881// (docs/transpose/decisions.md#recover-grade-inference records the exception).
882// The uniform spelling is the user's to write, by `fallible` below or by hand;
883// this overload never produces it.

898 is_nothrow_copy_constructible_v<
899 std::expected<VALUE, error_set_of<>>>)
900 -> std::expected<VALUE, error_set_of<>> {
901 static_assert(sizeof...(HANDLED) > 0,
902 "recover needs at least one HANDLED type.");
903 return computation;

1095 Result r = std::invoke(handler, *errors.template witness<E>());

1099 }

1101 (step(std::type_identity<ERRORS>{}), ...);
1102 return std::move(*acc);
