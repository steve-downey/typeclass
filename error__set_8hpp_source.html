6// error_set: the set of error types a computation may raise.
7//
8// NOMINAL, not a structural sum. `error_set_of<Es...>` is its own type. The
9// public surface is deliberately impoverished -- injection, subsumption,
10// membership, visitation -- because "the set of errors this computation may
11// raise" is an interpretation, not a representation, and interpretations
12// need names. See docs/transpose/decisions.md#error-set-identity.
13//
14// VALUE LEVEL: a WITNESSED SUBSET, not a one-of union. The TYPE says "may
15// raise {A,B}"; a VALUE says "did raise", and did-raise is a non-empty subset
16// of may-raise: one witness per raised type, at most. Storage is per-type
17// slots (a tuple of `std::optional<Es>...`) bounded by |grade|, never
18// allocation, never a discriminated one-of. This is the amendment recorded
19// at docs/transpose/decisions.md#error-set-identity, made by
20// docs/transpose/decisions.md#accumulation-evidence: the short-circuit
21// applicative object only ever produces a singleton witness (exactly the old
22// "one alternative" behavior, preserved as a special case); the accumulating
23// object (docs/transpose/transpose-grading-plan.md#accumulating-object)
24// combines witnesses from multiple failures, left-biased per type via
25// `error_set_combine` -- two witnesses of the SAME type keep the leftmost,
26// which is deterministic only because left-to-right traversal is already
27// normative (docs/transpose/decisions.md#applicative-objects).
28//
29// CANONICAL BY CONSTRUCTION. `error_set<Es...>` is the spelling users and the
30// framework write; it is an alias that sorts and deduplicates its pack, so
31// `error_set<A,B>` and `error_set<B,A>` are THE SAME TYPE. That is what makes
32// grade arithmetic well-defined and inclusions unique -- sorted normalization
33// is names-not-positions. `error_set_of` accepts only an already-canonical
34// pack and says so with a static_assert; forming a non-canonical one is the
35// invariant violation the class exists to prevent.
36//
37// The conversions are exactly the ⊆ widenings and nothing else. There is no
38// narrowing conversion, no conversion between incomparable sets, and no
39// accessor that hands out the underlying variant. Having only the subset
40// conversions is the coherence argument compiled into the overload set.
41//
42// TYPE ORDERING is P2830's job. Until it is available this header uses an
43// interim ordering read out of the compiler's own function signature, valid
44// only for the documented restrictions: named types, external linkage,
45// stable spelling. Two distinct types that render to the same name would be
46// silently conflated, so that case is a hard error rather than a wrong
47// answer -- see the static_assert in error_set_of.

78template <class T>
79consteval auto type_name() -> std::string_view {
80#if defined(_MSC_VER) && !defined(__clang__)
81 // MSVC has no __PRETTY_FUNCTION__. __FUNCSIG__ spells the specialization
82 // as `... type_name<T>(void)`, so the name is what sits between the
83 // template argument list and the last `>`. Should that spelling ever
84 // change, fall back to the whole signature, which still names T and so
85 // is still an injective key -- a duller order, not a broken one.
86 const std::string_view signature{__FUNCSIG__};
87 constexpr std::string_view marker{"type_name<"};
88
89 const auto found = signature.find(marker);
90 const auto stop = signature.rfind('>');
91 if (found == std::string_view::npos || stop == std::string_view::npos ||
92 stop <= found + marker.size()) {
93 return signature;
94 }
95 const auto start = found + marker.size();
96 return signature.substr(start, stop - start);
97#else
98 const std::string_view signature{__PRETTY_FUNCTION__};
99 constexpr std::string_view marker{"T = "};
100
101 const auto start = signature.find(marker) + marker.size();
102 auto stop = signature.find(';', start);
103 if (stop == std::string_view::npos) {
104 stop = signature.rfind(']');
105 }
106 return signature.substr(start, stop - start);
107#endif
108}

219template <class... ERRORS>
220consteval auto names_are_distinct() -> bool {
221 constexpr std::size_t count = sizeof...(ERRORS);
222 const std::array<std::string_view, count> names{type_name<ERRORS>()...};
223 for (std::size_t i = 0; i < count; ++i) {
224 for (std::size_t j = i + 1; j < count; ++j) {
225 if (names[i] == names[j]) {
226 return false;
227 }
228 }
229 }
230 return true;
231}

262// -- Per-type witness storage ----------------------------------------------
263// One std::optional slot per alternative, never a discriminated one-of. See
264// docs/transpose/decisions.md#accumulation-evidence for why: the grade join
265// needs no monoid, but accumulating VALUE-level evidence does, and the
266// semigroup it needs is "per type, keep one witness" -- which a tuple of
267// optionals gives for free, with no allocation and no extra discriminant to
268// keep in sync.
269
275template <class ERROR, class INJECTED>
276constexpr auto witness_slot(INJECTED &&error) -> std::optional<ERROR> {
277 if constexpr (std::is_same_v<ERROR, std::remove_cvref_t<INJECTED>>) {
278 return std::optional<ERROR>(std::forward<INJECTED>(error));
279 } else {
280 return std::optional<ERROR>{};
281 }
282}
283
288template <class ERROR, class... NARROWER>
289constexpr auto widen_slot(const error_set_of<NARROWER...> &narrower)
290 -> std::optional<ERROR> {
291 if constexpr (error_set_of<NARROWER...>::template contains<ERROR>()) {
292 return narrower.template witness<ERROR>();
293 } else {
294 return std::optional<ERROR>{};
295 }
296}

334template <class... ERRORS>
335class error_set_of {
336 static_assert(
337 detail::error_set_is_canonical_v<ERRORS...>,
338 "error_set_of requires a canonical (sorted, deduplicated) pack. "
339 "Spell error_set<...> instead, which canonicalizes for you.");
340 static_assert(
341 detail::error_set_names_distinct_v<ERRORS...>,
342 "Two distinct error types render to the same name, so the interim "
343 "type ordering cannot separate them. The P2830 fallback is valid "
344 "only for named types with external linkage and a stable spelling; "
345 "this is outside those restrictions.");

347 public:

349 template <class ERROR>
350 static constexpr auto contains() noexcept -> bool;

355 constexpr error_set_of(ERROR &&error);
356
357 template <class... NARROWER>
358 requires(sizeof...(NARROWER) > 0) &&
359 (!std::is_same_v<error_set_of<NARROWER...>,
360 error_set_of<ERRORS...>>) &&
361 (error_set_has_v<NARROWER, ERRORS...> && ...)
362 constexpr error_set_of(
363 const error_set_of<NARROWER...> &narrower);
364
365 // \ref{transpose.errset.obs}, observers
366 template <class ERROR>
367 requires(error_set_has_v<ERROR, ERRORS...>)
368 constexpr auto holds() const noexcept -> bool;

372 constexpr auto witness() const -> const std::optional<ERROR> &;

376 template <class HANDLER>

379 // \ref{transpose.errset.ops}, operations
387 friend auto operator==(const error_set_of &, const error_set_of &)
388 -> bool = default;
389
390 constexpr auto combined_with(const error_set_of &other) const
391 -> error_set_of;
392
393 private:
395 explicit constexpr error_set_of(
396 std::tuple<std::optional<ERRORS>...> witnesses)
397 : d_witnesses(std::move(witnesses)) {}
398
400 template <std::size_t... IDX>
401 constexpr auto combine_witnesses(const error_set_of &other,
402 std::index_sequence<IDX...>) const
403 -> std::tuple<std::optional<ERRORS>...> {
404 return std::tuple<std::optional<ERRORS>...>{
405 (std::get<IDX>(d_witnesses).has_value()
406 ? std::get<IDX>(d_witnesses)
407 : std::get<IDX>(other.d_witnesses))...};
408 }
409
411 constexpr auto to_variant() const -> std::variant<ERRORS...> {
412 if (witness_count() != 1) {
413 detail::error_set_precondition_violated(
414 "beman::transpose::error_set::visit requires exactly one "
415 "witness, but this value witnesses a different number. Use "
416 "witness<E>() per alternative for a multi-witness subset; "
417 "see docs/transpose/decisions.md#accumulation-evidence.");
418 }
419
420 std::optional<std::variant<ERRORS...>> found;
421 auto try_slot = [&found](const auto &slot) {
422 if (!found.has_value() && slot.has_value()) {
423 found = std::variant<ERRORS...>{*slot};
424 }
425 };
426 std::apply(
427 [&try_slot](const auto &...slots) { (try_slot(slots), ...); },
428 d_witnesses);
429 return *found;
430 }
431
433 std::tuple<std::optional<ERRORS>...> d_witnesses;
434};

645// -- recover: narrowing the grade by set difference ------------------------
646// Stage recover-narrowing
647// (docs/transpose/transpose-grading-plan.md#recover-narrowing). Error
648// vocabulary, per grade.hpp's own comment that `recover` lives in the model's
649// header, not the framework's.
650//
651// docs/transpose/decisions.md#recover-grade-inference: the handled set {H} is
652// ANNOTATED -- explicit template arguments, never inferred from the
653// handler's shape -- and CHECKED: one static_assert enforces H is a subset
654// of the computation's grade (reusing error_set_of::contains, the same
655// membership primitive docs/transpose/decisions.md#multi-witness-elimination
656// names), another enforces the handler's result shape. The resulting grade
657// `(e \ H) ∪ raised` is then a direct, non-recursive computation -- no
658// lattice fixpoint, because one recover call is not a recursive fold
659// ("inference only where no fixpoint is required").
660//
661// docs/transpose/decisions.md#multi-witness-elimination: recover needs no new
662// public verb. Elimination is a static fold over H with runtime presence
663// filtering, built entirely from the existing public API --
664// error_set_of's injecting constructor and `combined_with` -- never from
665// private access to the representation.

774// -- Value level: carrying witnesses forward through the PUBLIC API only ---
775// No private access, no friendship: exactly the injecting constructor and
776// `combined_with` that any user of error_set already has. This is what makes
777// the multi-witness-elimination fence a fence rather than an oversight --
778// recover's mechanism is not privileged.
779
785template <class E, class HANDLED_LIST, class TARGET, class SOURCE>
786constexpr void carry_witness(std::optional<TARGET> &accumulated,
787 const SOURCE &source) {
788 if constexpr (!list_contains_v<E, HANDLED_LIST> &&
789 TARGET::template contains<E>()) {
790 if (source.template holds<E>()) {
791 TARGET injected(*source.template witness<E>());
792 accumulated = accumulated.has_value()
793 ? accumulated->combined_with(injected)
794 : injected;
795 }
796 }

813template <class H, class HANDLER, class VALUE, class TARGET, class SOURCE>
814constexpr void apply_handler(HANDLER &handler, std::optional<VALUE> &recovered,
815 std::optional<TARGET> &accumulated,
816 const SOURCE &error) {
817 if (error.template holds<H>()) {
818 auto result = std::invoke(handler, *error.template witness<H>());
819 if constexpr (std::is_same_v<std::remove_cvref_t<decltype(result)>,
820 VALUE>) {
821 if (!recovered.has_value()) {
822 recovered = std::move(result);
823 }
824 } else {
825 if (result.has_value()) {
826 if (!recovered.has_value()) {
827 recovered = *result;
828 }
829 } else {
830 carry_raised(accumulated, result);
831 }
832 }
833 }
834}

848
867template <class... HANDLED, class VALUE, class... ERRORS, class HANDLER>
868constexpr auto
869recover(const std::expected<VALUE, error_set_of<ERRORS...>> &computation,
870 HANDLER &&handler)
871 -> detail::recover_return_t<HANDLER, VALUE, error_set_of<ERRORS...>,
872 HANDLED...> {
873 static_assert(sizeof...(HANDLED) > 0,
874 "recover needs at least one HANDLED type.");
875 static_assert(
876 (error_set_of<ERRORS...>::template contains<HANDLED>() && ...),
877 "recover's HANDLED types must be members of the computation's grade "
878 "-- CHECKED, per docs/transpose/decisions.md#recover-grade-inference. "
879 "The "
880 "handled set is ANNOTATED (explicit template arguments on recover), "
881 "never inferred.");
882 static_assert(
883 (detail::is_valid_recover_result_v<
884 typename detail::handler_result_of<HANDLER, VALUE, HANDLED>::type,
885 VALUE> &&
886 ...),
887 "recover's handler must return either the recovered VALUE directly, "
888 "or std::expected<VALUE, error_set<...>> if the recovery attempt can "
889 "itself raise. See "
890 "docs/transpose/decisions.md#recover-grade-inference.");
891
892 using Grade = error_set_of<ERRORS...>;
893 using FinalGrade =
894 detail::recover_final_grade_t<HANDLER, VALUE, Grade, HANDLED...>;
895 using Return = detail::recover_return_t<HANDLER, VALUE, Grade, HANDLED...>;
896
897 if (computation.has_value()) {
898 return Return(*computation);
899 }
900
901 const Grade &error = computation.error();
902 std::optional<VALUE> recovered;
903 std::optional<FinalGrade> accumulated;
904
905 (detail::carry_witness<ERRORS, detail::type_list<HANDLED...>>(accumulated,
906 error),
907 ...);
908 (detail::apply_handler<HANDLED>(handler, recovered, accumulated, error),
909 ...);
910
911 if constexpr (std::is_same_v<Return, VALUE>) {
912 // FinalGrade is ∅: every carry_witness/carry_raised call above was a
913 // no-op by construction (TARGET::contains<E>() is unconditionally
914 // false for error_set_of<>), so `accumulated` can never actually hold
915 // a value here -- every present witness was necessarily HANDLED and
916 // recovered.
917 return *recovered;
918 } else {
919 if (accumulated.has_value()) {
920 return Return(std::unexpect, *accumulated);
921 }
922 return Return(*recovered);
923 }
924}
