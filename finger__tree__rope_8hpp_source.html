43template <typename Tree = FingerTree<std::string, beman::transpose::sum_monoid<std::size_t>, RopeChunkMeasure>>
44class FingerTreeRope {
45 Tree d_tree;
46
48 auto split_chars(std::size_t pos) const -> std::pair<FingerTreeRope, FingerTreeRope> {
49 if (pos == 0)
50 return {FingerTreeRope{}, *this};
51 if (pos >= size_bytes())
52 return {*this, FingerTreeRope{}};
53
54 auto split = d_tree.split([pos](std::size_t prefix) { return prefix > pos; });
55 if (!split.has_value())
56 return {*this, FingerTreeRope{}};
57
58 auto left_prefix_bytes = split->d_left.measure();
59 auto local = pos - left_prefix_bytes;
60
61 const auto& pivot = split->d_pivot;
62 assert(local < pivot.size());
63
64 auto left = split->d_left;
65 if (local > 0)
66 left = left.snoc(pivot.substr(0, local));
67
68 auto right = split->d_right;
69 if (local < pivot.size())
70 right = right.cons(pivot.substr(local));
71
72 return {FingerTreeRope{std::move(left)}, FingerTreeRope{std::move(right)}};
73 }
74
75 explicit FingerTreeRope(Tree tree) : d_tree(std::move(tree)) {}
76
77 public:
78 using value_type = std::string;
79
81 FingerTreeRope() : d_tree(Tree{}) {}

84 static auto from_chunks(std::vector<std::string> chunks) -> FingerTreeRope {
85 return FingerTreeRope{Tree::from_sequence(std::move(chunks))};

91 static auto from_text(std::string_view text, std::size_t chunk_size = 16) -> FingerTreeRope {
92 std::vector<std::string> chunks;
93 chunks.reserve((text.size() / chunk_size) + 1);
94 for (std::size_t i = 0; i < text.size(); i += chunk_size) {
95 const auto n = std::min(chunk_size, text.size() - i);
96 chunks.emplace_back(text.substr(i, n));
97 }
98 return from_chunks(std::move(chunks));
99 }
100
102 auto size_bytes() const -> std::size_t { return d_tree.measure(); }
103
105 auto to_string() const -> std::string {
106 std::string out;
107 out.reserve(size_bytes());
108 std::ranges::for_each(d_tree.flatten(), [&out](const std::string& c) { out += c; });
109 return out;
110 }
111
114 auto insert(std::size_t pos, std::string_view text) const -> FingerTreeRope {
115 auto [left, right] = split_chars(pos);
116 auto middle = from_text(text);
117 return FingerTreeRope{Tree::concat(Tree::concat(left.d_tree, middle.d_tree), right.d_tree)};
118 }
119
122 auto erase(std::size_t pos, std::size_t count) const -> FingerTreeRope {
123 auto [left, rest] = split_chars(pos);
124 auto [drop, right] = rest.split_chars(count);
125 static_cast<void>(drop);
126 return FingerTreeRope{Tree::concat(left.d_tree, right.d_tree)};
127 }
128
131 auto replace(std::size_t pos, std::size_t count, std::string_view text) const -> FingerTreeRope {
132 return erase(pos, count).insert(pos, text);
133 }
134
136 auto chunks() const -> std::vector<std::string> { return d_tree.flatten(); }
137};

143template <class Tree>
144struct FingerTreeRopeFoldableImpl {
145 using element_type = std::string;
146
147 // The monoid is an instance object over the result type, passed by
148 // value; the Impl never chooses one (beman.transpose
149 // docs/transpose/decisions.md#monoid-selection).
150 template <class F, class RESULT_MONOID>
151 auto fold_map(this auto&&,
152 F&& function,
153 const beman::fingertree::tree::FingerTreeRope<Tree>& rope,
154 RESULT_MONOID monoid) -> std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>
155 requires std::invocable<F&, const std::string&> &&
156 beman::transpose::monoid_object<RESULT_MONOID,
157 std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>>
158 {
159 using Result = std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>;
160 return std::ranges::fold_left(rope.chunks(), Result(monoid.identity()), [&](Result acc, const auto& value) {
161 return monoid.combine(std::move(acc), std::invoke(function, value));
162 });
163 }
164};
