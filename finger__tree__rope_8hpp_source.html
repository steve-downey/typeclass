40template <typename Tree = FingerTree<std::string, beman::transpose::sum_monoid<std::size_t>, RopeChunkMeasure>>
41class FingerTreeRope {
42 Tree d_tree;
43
45 auto split_chars(std::size_t pos) const -> std::pair<FingerTreeRope, FingerTreeRope> {
46 if (pos == 0)
47 return {FingerTreeRope{}, *this};
48 if (pos >= size_bytes())
49 return {*this, FingerTreeRope{}};
50
51 auto split = d_tree.split([pos](std::size_t prefix) { return prefix > pos; });
52 if (!split.has_value())
53 return {*this, FingerTreeRope{}};
54
55 auto left_prefix_bytes = split->d_left.measure();
56 auto local = pos - left_prefix_bytes;
57
58 const auto& pivot = split->d_pivot;
59 assert(local < pivot.size());
60
61 auto left = split->d_left;
62 if (local > 0)
63 left = left.snoc(pivot.substr(0, local));
64
65 auto right = split->d_right;
66 if (local < pivot.size())
67 right = right.cons(pivot.substr(local));
68
69 return {FingerTreeRope{std::move(left)}, FingerTreeRope{std::move(right)}};
70 }
71
72 explicit FingerTreeRope(Tree tree) : d_tree(std::move(tree)) {}
73
74 public:
76 FingerTreeRope() : d_tree(Tree{}) {}
77
79 static auto from_chunks(std::vector<std::string> chunks) -> FingerTreeRope {
80 return FingerTreeRope{Tree::from_sequence(std::move(chunks))};
81 }
82
86 static auto from_text(std::string_view text, std::size_t chunk_size = 16) -> FingerTreeRope {
87 std::vector<std::string> chunks;
88 chunks.reserve((text.size() / chunk_size) + 1);
89 for (std::size_t i = 0; i < text.size(); i += chunk_size) {
90 const auto n = std::min(chunk_size, text.size() - i);
91 chunks.emplace_back(text.substr(i, n));
92 }
93 return from_chunks(std::move(chunks));
94 }
95
97 auto size_bytes() const -> std::size_t { return d_tree.measure(); }
98
100 auto to_string() const -> std::string {
101 std::string out;
102 out.reserve(size_bytes());
103 std::ranges::for_each(d_tree.flatten(), [&out](const std::string& c) { out += c; });
104 return out;
105 }
106
109 auto insert(std::size_t pos, std::string_view text) const -> FingerTreeRope {
110 auto [left, right] = split_chars(pos);
111 auto middle = from_text(text);
112 return FingerTreeRope{Tree::concat(Tree::concat(left.d_tree, middle.d_tree), right.d_tree)};
113 }
114
117 auto erase(std::size_t pos, std::size_t count) const -> FingerTreeRope {
118 auto [left, rest] = split_chars(pos);
119 auto [drop, right] = rest.split_chars(count);
120 static_cast<void>(drop);
121 return FingerTreeRope{Tree::concat(left.d_tree, right.d_tree)};
122 }
123
126 auto replace(std::size_t pos, std::size_t count, std::string_view text) const -> FingerTreeRope {
127 return erase(pos, count).insert(pos, text);
128 }
129
131 auto chunks() const -> std::vector<std::string> { return d_tree.flatten(); }
132};

138template <class Tree>
139struct FingerTreeRopeFoldableImpl {
140 using element_type = std::string;
141
142 // The monoid is an instance object over the result type, passed by
143 // value; the Impl never chooses one (beman.transpose
144 // docs/transpose/decisions.md#monoid-selection).
145 template <class F, class RESULT_MONOID>
146 auto fold_map(this auto&&,
147 F&& function,
148 const beman::fingertree::tree::FingerTreeRope<Tree>& rope,
149 RESULT_MONOID monoid) -> std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>> {
150 using Result = std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>;
151 return std::ranges::fold_left(rope.chunks(), Result(monoid.identity()), [&](Result acc, const auto& value) {
152 return monoid.combine(std::move(acc), std::invoke(function, value));
153 });
154 }
155};

172template <class Tree>
173struct FingerTreeRopeTraversableImpl {
174 using element_type = std::string;
175
176 template <class APPLICATIVE, class F>
177 auto traverse(this auto&&,
178 const APPLICATIVE& applicative,
179 F&& function,
180 const beman::fingertree::tree::FingerTreeRope<Tree>& rope)
181 requires std::invocable<F&, const std::string&> &&
182 beman::transpose::applicative_object_for<
183 APPLICATIVE,
184 std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>>
185 {
186 using Context = std::remove_cvref_t<std::invoke_result_t<F&, const std::string&>>;
187 using U = beman::transpose::applicative_value_t<Context>;
188
189 auto accumulated = applicative.pure(std::vector<U>{});
190
191 for (const auto& chunk : rope.chunks()) {
192 auto lifted = std::invoke(function, chunk);
193 accumulated = applicative.invoke(
194 [](std::vector<U> values, U element) {
195 values.push_back(std::move(element));
196 return values;
197 },
198 std::move(accumulated),
199 std::move(lifted));
200 }
201
202 // Result always uses the default (-backed) FingerTreeRope.
203 return applicative.invoke(
204 [](std::vector<U> values) {
205 return beman::fingertree::tree::FingerTreeRope<>::from_chunks(std::move(values));
206 },
207 std::move(accumulated));
208 }
209};

211template <class Tree>
212struct FingerTreeRopeTraversableMap : beman::transpose::derive_traversable<FingerTreeRopeTraversableImpl<Tree>> {
213 using FingerTreeRopeTraversableImpl<Tree>::traverse;
214};
