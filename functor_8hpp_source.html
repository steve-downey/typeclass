36// Functor pattern invariants:
37// - Instances are single lookup objects that provide fmap(F, T).
38// - replace is a derived object operation implemented from fmap.
39// - Dispatch happens through a provided object or functor<Concrete>.
40// - Keep lookup explicit through typeclass objects, not ADL overloads.

45template <class Impl>
46struct derive_functor : protected Impl {
48 template <class FUNCTION, class T>
49 constexpr auto fmap(this auto &&self, FUNCTION &&function, T &&value)
50 requires requires(const Impl &impl) {
51 impl.fmap(std::forward<FUNCTION>(function), std::forward<T>(value));
52 }
53 {
54 return impl_of(self).fmap(std::forward<FUNCTION>(function),
55 std::forward<T>(value));
56 }

62 template <class T, class U>
63 constexpr auto replace(this auto &&self, T &&value, U &&replacement)
64 requires requires(const Impl &impl) {
65 impl.replace(std::forward<T>(value), std::forward<U>(replacement));
66 } || requires(const Impl &impl) {
67 impl.fmap(
68 detail::constant_function<std::remove_cvref_t<U>>{
69 std::forward<U>(replacement)},
70 std::forward<T>(value));
71 }
72 {
73 if constexpr (requires {
74 impl_of(self).replace(std::forward<T>(value),
75 std::forward<U>(replacement));
76 }) {
77 return impl_of(self).replace(std::forward<T>(value),
78 std::forward<U>(replacement));
79 } else {
80 // One-way derivation (not a mutually-derivable pair): route
81 // through self, not Impl, so a shadow on a wrapping Map is
82 // still reached.
83 return self.fmap(
84 detail::constant_function<std::remove_cvref_t<U>>{
85 std::forward<U>(replacement)},
86 std::forward<T>(value));
87 }
88 }
89
90 private:
92 template <class SELF>
93 static constexpr decltype(auto) impl_of(SELF &&self) {
94 return static_cast<impl_ref_t<Impl, SELF>>(self);
95 }
96};

105// -- Operation objects --
106//
107// This comment is the reference for the operation objects; the other
108// typeclass headers point back to it. The rulings are
109// docs/decisions.md#operation-objects, #lookup-variable-naming and
110// #empty-predicate-name.
111//
112// A fourth lookup mode, on top of implicit lookup, explicit object argument,
113// and NTTP pinning (examples/lookup_modes_example.cpp): the operation as an
114// object that does the lookup itself. `fmap(f, v)` names the operation the
115// caller means, rather than the object the operation is reached through.
116//
117// Two cases motivate them. First, everything-is-local code -- a container on
118// the stack that one algorithm maps or traverses -- where naming a lookup
119// object costs a line to save nothing. Second, and the reason the pattern is
120// not merely cosmetic: a *non-template* caller can not carry the typeclass as
121// an NTTP, because there is no template parameter list to hang it on. The
122// operation object resolves the instance at the call site instead.
123//
124// They are NOT customization points, and D3200R0's case for a bundled
125// mechanism depends on the difference. Customization happens once, at
126// functor<T> and its siblings. There is no per-operation hook, no
127// ADL path in, and nothing specializable that would make `fmap` mean
128// something other than what the instance's own fmap means. The familiar name
129// for this shape is "customization point object"; it is the wrong name here,
130// and these are operation objects over a customizable bundle.
131//
132// Shape, uniform across every typeclass: deduce the type that keys the lookup
133// from one designated argument, default a trailing `const auto &TC` NTTP to
134// the lookup for that type, and call through it.
135//
136// ONE NAMESPACE FOR THE WHOLE FAMILY. Every typeclass operation lives in one
137// namespace, not one namespace per typeclass. Here that namespace is
138// beman::transpose, which is already nested and already scoped to exactly
139// this facility; the standard spelling this maps to is `std::typeclass`.
140//
141// The reason is coherence, and it is part of the design rather than a
142// consequence of it. A name in here means one thing across the entire family:
143// `fmap` is Functor's map whichever instance supplies it, and Applicative is
144// free to provide the Functor names because they can not mean anything else.
145// That commitment is already load-bearing elsewhere -- it is what lets an
146// algorithm inherit from two typeclass instances at once and call both
147// unqualified (see examples/algorithm_object_example.cpp). Collecting the
148// operations in one namespace makes the commitment checkable in one place. A
149// namespace per typeclass would let `fmap` mean two things and hide it.
150//
151// It also keeps the operations away from std, where `empty`, `length`,
152// `invoke`, `to_vector`, `any_of` and `all_of` all exist with related
153// meanings. `beman::transpose::empty` is not something a caller drags into
154// scope by accident, and `std::typeclass::empty` would not be either.
155//
156// NAMING. The lookup variables get the plain names -- `functor<T>`,
157// `applicative<T>`, `foldable<T>`, `traversable<T>`, `monad<T>` -- because
158// they are the user-facing surface. What that collides with gets the longer,
159// more descriptive name: the CRTP bases became `derive_functor<Impl>` and
160// friends, which says what they do (derive the rest of a typeclass from an
161// instance's primitives) instead of relying on a capital letter to tell
162// `Functor` from `functor`.
163//
164// One consequence worth knowing: a local named `applicative` now shadows
165// `applicative<T>`. Locals holding a looked-up instance are spelled
166// `applicative_map` here, matching the existing `traversable_map` and
167// `monad_map` parameter names.
168//
169// The one name where coherence is under real strain is `empty`, which means
170// Foldable's "holds nothing" here and Alternative's "identity of alt" in the
171// sibling compile-time-scheme tree. Adding Alternative would claim it twice,
172// in the one namespace whose job is to make that impossible. That is the
173// ordinary C++ trouble with a short name in a widely-used namespace, and the
174// name is provisional like every other name here, more so than most. The three
175// languages with both typeclasses all give `empty` to Alternative and name
176// the Foldable predicate something else -- `null` in Haskell (Data.Foldable)
177// and PureScript (which re-exports Control.Plus's `empty` from the same
178// module, so the split is load-bearing there), `isEmpty` in Cats, whose
179// Alternative gets `empty` from MonoidK.
180//
181// C++ points the other way: std::empty, std::ranges::empty and every
182// container's .empty() are the *predicate*, and the language has no
183// nullary-constructor-named-empty convention to protect. So the plan here is
184// the opposite of the FP convention and consistent with the host language:
185// Foldable keeps `empty`, and Alternative's identity takes another name when
186// it arrives. `zero<C>()` sits naturally beside `pure<C>(v)` -- both build a
187// context and both must name it, for the same reason.
188//
189// PINNING is mode 2, unchanged: `tc.fmap(f, v)` on an instance object, which
190// is shorter than any spelling routed through these objects. An operation
191// object takes no explicit template arguments, so mode 3 is not available on
192// it -- the trailing TC above is reachable only as
193// `fmap.operator()<F, T, TC>(...)`, which is a curiosity, not an
194// API. The earlier lookup modes are all untouched; `typeclass` is an addition.
195

266template <class VALUE_TYPE>
267struct OptionalFunctorImpl {
268 template <class F>
269 constexpr auto fmap(this auto &&, F &&function,
270 const std::optional<VALUE_TYPE> &value) {
271 using Result = std::invoke_result_t<F, const VALUE_TYPE &>;
272 if (!value) {
273 return std::optional<std::remove_cvref_t<Result>>{};
274 }
275 return std::optional<std::remove_cvref_t<Result>>{
276 std::invoke(std::forward<F>(function), *value)};
277 }
278};

280template <class VALUE_TYPE>
281struct VectorFunctorImpl {
282 template <class F>
283 constexpr auto fmap(this auto &&, F &&function,
284 const std::vector<VALUE_TYPE> &values) {
285 // `F &`, unlike the optional instance above: that one forwards the
286 // callable into a single std::invoke, this one invokes a captured
287 // lvalue once per element.
288 using Result = std::invoke_result_t<F &, const VALUE_TYPE &>;
289 std::vector<std::remove_cvref_t<Result>> output;
290 output.reserve(values.size());
291
292 std::ranges::transform(values, std::back_inserter(output),
293 [&function](const VALUE_TYPE &v) {
294 return std::invoke(function, v);
295 });
296
297 return output;
298 }
299};
