39// Concrete reconstruction operations keep storage choices in their own
40// representation: shared_ptr children for BinaryTree, Box at the Fix knot.
41template <class T>
42struct binary_payload_model {
43 using element_type = T;
44 template <class U>
45 using rebind = BinaryTree<U>;
46 static constexpr auto value(const BinaryTree<T>& tree) -> const T& { return tree.value(); }
47 static constexpr auto left(const BinaryTree<T>& tree) -> const BinaryTree<T>* {
48 return tree.has_left() ? &tree.left() : nullptr;
49 }
50 static constexpr auto right(const BinaryTree<T>& tree) -> const BinaryTree<T>* {
51 return tree.has_right() ? &tree.right() : nullptr;
52 }
53 template <class U>
54 static constexpr auto node(U value, std::optional<rebind<U> > left, std::optional<rebind<U> > right) -> rebind<U> {
55 return rebind<U>::from_children_ptrs(std::move(value),
56 left ? rebind<U>::make_ptr(std::move(*left)) : nullptr,
57 right ? rebind<U>::make_ptr(std::move(*right)) : nullptr);
58 }
59};

66 static constexpr auto value(const Tree& tree) -> const T& { return tree.inner.value; }

82template <class Model>
83struct binary_payload_builder {
84 std::vector<binary_payload_shape> shape;
85 std::size_t root;
86 template <class U>
87 constexpr auto operator()(std::vector<U> values) const -> typename Model::template rebind<U> {
88 assert(values.size() == shape.size());
89 using Result = typename Model::template rebind<U>;
90 auto rebuild = [&](auto&& self, std::size_t position) -> Result {
91 std::optional<Result> left, right;
92 if (shape[position].left)
93 left.emplace(self(self, *shape[position].left));
94 if (shape[position].right)
95 right.emplace(self(self, *shape[position].right));
96 return Model::template node<U>(std::move(values[position]), std::move(left), std::move(right));
97 };
98 return rebuild(rebuild, root);
99 }
100};

102template <class Model, class Tree>
103constexpr auto record_binary(const Tree& source) {
104 std::vector<const typename Model::element_type*> positions;
105 std::vector<binary_payload_shape> shape;
106 auto record = [&](auto&& self, const Tree& tree) -> std::size_t {
107 std::optional<std::size_t> left;
108 if (auto child = Model::left(tree))
109 left = self(self, *child);
110 auto index = positions.size();
111 positions.push_back(std::addressof(Model::value(tree)));
112 shape.push_back({left, {}});
113 if (auto child = Model::right(tree)) {
114 // Recursion may reallocate shape; finish it before indexing.
115 const auto right = self(self, *child);
116 shape[index].right = right;
117 }
118 return index;
119 };
120 const auto root = record(record, source);
121 return std::pair{std::move(positions), binary_payload_builder<Model>{std::move(shape), root}};
122}

124template <class T>
125struct fringe_payload_model {
126 using element_type = T;
127 // Pre-order node kinds encode the complete topology, including empties.
128 enum class kind { empty, leaf, branch };
129 struct builder {
130 std::vector<kind> shape;
131 template <class U>
132 constexpr auto operator()(std::vector<U> values) const -> FringeTree<U> {
133 std::size_t node = 0, payload = 0;
134 auto rebuild = [&](auto&& self) -> FringeTree<U> {
135 switch (shape[node++]) {
136 case kind::empty:
137 return FringeTree<U>::empty();
138 case kind::leaf:
139 return FringeTree<U>::leaf(std::move(values[payload++]));
140 case kind::branch: {
141 auto left = self(self);
142 auto right = self(self);
143 return FringeTree<U>::branch(std::move(left), std::move(right));
144 }
145 }
146 std::terminate();
147 };
148 auto result = rebuild(rebuild);
149 assert(node == shape.size() && payload == values.size());
150 return result;
151 }
152 };
153 static constexpr auto record(const FringeTree<T>& source) {
154 std::vector<const T*> positions;
155 std::vector<kind> shape;
156 auto visit = [&](auto&& self, const FringeTree<T>& tree) -> void {
157 if (tree.is_empty())
158 shape.push_back(kind::empty);
159 else if (tree.is_leaf()) {
160 shape.push_back(kind::leaf);
161 positions.push_back(std::addressof(tree.value()));
162 } else {
163 shape.push_back(kind::branch);
164 self(self, tree.left());
165 self(self, tree.right());
166 }
167 };
168 visit(visit, source);
169 return std::pair{std::move(positions), builder{std::move(shape)}};
170 }
171};

173template <class T, class Tree>
174struct rose_payload_model {
175 using element_type = T;
176 struct builder {
177 std::vector<std::size_t> arities;
178 template <class U>
179 constexpr auto operator()(std::vector<U> values) const -> RoseTreeFix<U> {
180 assert(values.size() == arities.size());
181 std::size_t position = 0;
182 auto rebuild = [&](auto&& self) -> RoseTreeFix<U> {
183 const auto index = position++;
184 std::vector<RoseTreeFix<U> > children;
185 children.reserve(arities[index]);
186 for (std::size_t i = 0; i < arities[index]; ++i)
187 children.push_back(self(self));
188 return rose<U>(std::move(values[index]), std::move(children));
189 };
190 return rebuild(rebuild);
191 }
192 };
193 static constexpr auto record(const Tree& source) {
194 std::vector<const T*> positions;
195 std::vector<std::size_t> arities;
196 auto visit = [&](auto&& self, const Tree& tree) -> void {
197 positions.push_back(std::addressof(tree.inner.value));
198 arities.push_back(tree.inner.children.size());
199 for (const auto& child : tree.inner.children)
200 self(self, child);
201 };
202 visit(visit, source);
203 return std::pair{std::move(positions), builder{std::move(arities)}};
204 }
205};

244template <class Tree>
245struct PayloadFunctorImpl {
246 using element_type = typename payload_model<Tree>::element_type;
247 template <class F>
248 requires payload_mapper<F, element_type>
249 constexpr auto fmap(this auto&&, F&& function, const Tree& tree) {
250 auto [positions, builder] = payload_model<Tree>::record(tree);
251 using U = std::remove_cvref_t<std::invoke_result_t<F&, const element_type&> >;
252 std::vector<U> values;
253 values.reserve(positions.size());
254 for (const auto* value : positions)
255 values.emplace_back(std::invoke(function, *value));
256 return builder(std::move(values));
257 }
258};

276template <class Tree>
277struct PayloadTraversableImpl {
278 using element_type = typename payload_model<Tree>::element_type;
279 template <class Applicative, class F>
280 requires payload_mapper<F, element_type> &&
281 transpose::applicative_object_for<
282 Applicative,
283 std::remove_cvref_t<std::invoke_result_t<F&, const element_type&> > > &&
284 transpose::detail::holdable_value<transpose::applicative_value_t<
285 std::remove_cvref_t<std::invoke_result_t<F&, const element_type&> > > > &&
286 std::move_constructible<transpose::applicative_value_t<
287 std::remove_cvref_t<std::invoke_result_t<F&, const element_type&> > > >
288 constexpr auto traverse(this auto&&, const Applicative& applicative, F&& function, const Tree& tree) {
289 using Context = std::remove_cvref_t<std::invoke_result_t<F&, const element_type&> >;
290 using U = transpose::applicative_value_t<Context>;
291 auto [positions, builder] = payload_model<Tree>::record(tree);
292 // These pointers are consumed synchronously by the mapper; deferred
293 // effects own their captures, and the final closure owns only shape.
294 auto accumulated = transpose::traversable<decltype(positions)>.traverse(
295 applicative,
296 [&](const element_type* value) -> Context { return std::invoke(function, *value); },
297 positions);
298 return applicative.invoke(
299 [builder = std::move(builder)](std::vector<U> values) { return builder(std::move(values)); },
300 std::move(accumulated));
301 }
302};
303template <class Tree>
304struct PayloadTraversableMap : transpose::derive_traversable<PayloadTraversableImpl<Tree> > {};
