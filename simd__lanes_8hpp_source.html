6// Non-normative demonstration type. simd_lanes models a fixed-width lanewise
7// applicative context for Paper A's SIMD domain, backed by std::simd::vec for
8// arithmetic element types. Unlike zip_list (which has dynamic/infinite width),
9// simd_lanes has a compile-time width N matching a hardware SIMD register.
10//
11// Transposing a structure of simd_lanes produces a simd_lanes of structures:
12// vector<simd_lanes<T, N>> -> simd_lanes<vector<T>, N>
13//
14// The general storage is std::array<T, N> because the transposed result may
15// hold non-arithmetic T (e.g., vector<int>). Examples show construction from
16// std::simd::vec for the arithmetic case.

29template <class T, int N>
30struct simd_lanes {
31 using value_type = T;
32 static constexpr int width = N;
33
34 std::array<T, N> data;
35
36 static auto repeat(T value) -> simd_lanes {
37 // Built element by element rather than default-constructed and
38 // filled, so T need only be copy-constructible.
39 return simd_lanes{detail::make_array<T, static_cast<std::size_t>(N)>(
40 [&value](std::size_t) -> T { return T(value); },
41 std::make_index_sequence<static_cast<std::size_t>(N)>{})};
42 }
43
44 friend auto operator==(const simd_lanes &, const simd_lanes &)
45 -> bool = default;
46};

69template <class T, int N>
70struct SimdLanesApplicativeImpl {
71 template <class VALUE>
72 auto pure(this auto &&, VALUE &&value) {
73 using U = std::remove_cvref_t<VALUE>;
74 return simd_lanes<U, N>::repeat(U(std::forward<VALUE>(value)));
75 }
76
77 // Operands are deduced through forwarding references so that an rvalue
78 // operand's lanes are handed to `function` rather than copied. Each lane
79 // is read exactly once, so that is safe. It matters because the
80 // accumulator of a traversal into this context is an rvalue at every
81 // step: copying it would rebuild the whole accumulated result per lane.
82 // The operands are constrained on their shape for the same reason the
83 // other registered objects constrain theirs: the return type is deduced,
84 // so an operand of some other shape would reach the body and diagnose
85 // there rather than leave this overload unmatched.
86 template <class FUNCTION, class FIRST, class... REST>
87 requires detail::is_simd_lanes_of_width_v<FIRST, N> &&
88 (detail::is_simd_lanes_of_width_v<REST, N> && ...)
89 auto invoke(this auto &&, FUNCTION &&function, FIRST &&first,
90 REST &&...rest) {
91 using Result = std::invoke_result_t<
92 FUNCTION &,
93 detail::element_ref_t<decltype(std::forward<FIRST>(first).data)>,
94 detail::element_ref_t<decltype(std::forward<REST>(rest).data)>...>;
95 using U = std::remove_cvref_t<Result>;
96
97 // Lanes are constructed, not default-constructed and assigned, so U
98 // need not be default-constructible or assignable.
99 return simd_lanes<U, N>{
100 detail::make_array<U, static_cast<std::size_t>(N)>(
101 [&](std::size_t index) -> U {
102 return std::invoke(
103 function,
104 detail::forward_at(std::forward<FIRST>(first).data,
105 index),
106 detail::forward_at(std::forward<REST>(rest).data,
107 index)...);
108 },
109 std::make_index_sequence<static_cast<std::size_t>(N)>{})};
110 }
111};

114template <class T, int N>
115struct SimdLanesApplicativeMap
116 : derive_applicative<SimdLanesApplicativeImpl<T, N>> {
117 using SimdLanesApplicativeImpl<T, N>::invoke;
118 using SimdLanesApplicativeImpl<T, N>::pure;
119};

121template <class T, int N>
