37template <class T>
38struct zip_list {
39 using value_type = T;
40
41 // Invariant: when repeated has a value, this zip_list models an infinite
42 // repetition of that value and data is ignored.
43 std::vector<T> data;
44 std::optional<T> repeated{};
45
47 static auto repeat(T value) -> zip_list {
48 return zip_list{{}, std::move(value)};
49 }
50
52 auto is_repeating() const -> bool { return repeated.has_value(); }
53
56 auto finite_size() const -> std::size_t { return data.size(); }
57
61 friend auto operator==(const zip_list &left, const zip_list &right)
62 -> bool {
63 if (left.is_repeating() || right.is_repeating()) {
64 return left.repeated == right.repeated;
65 }
66 return left.data == right.data;
67 }
68};

105template <class LIST>
106constexpr auto forward_zip_list_value_at(LIST &&list, std::size_t index)
107 -> decltype(auto) {
108 if constexpr (std::is_lvalue_reference_v<LIST>) {
109 return zip_list_value_at(list, index);
110 } else {
111 using Value = typename std::remove_cvref_t<LIST>::value_type;
112 return list.is_repeating() ? Value(*list.repeated)
113 : Value(std::move(list.data[index]));
114 }
115}
116
119template <class LIST>
120using zip_list_value_ref_t =
121 decltype(forward_zip_list_value_at(std::declval<LIST>(), std::size_t{}));

135template <class FIRST, class... REST>
136auto zip_list_result_size(const FIRST &first, const REST &...rest)
137 -> std::optional<std::size_t> {
138 auto count = zip_list_finite_length(first);
139 ((count = count
140 ? std::optional<std::size_t>{std::min(
141 *count, zip_list_finite_length(rest).value_or(*count))}
142 : zip_list_finite_length(rest)),
143 ...);
144 return count;
145}

158template <class T>
159struct ZipListApplicativeImpl {
161 template <class VALUE>
162 auto pure(this auto &&, VALUE &&value) {
163 using U = std::remove_cvref_t<VALUE>;
164 return zip_list<U>::repeat(U(std::forward<VALUE>(value)));
165 }
166
175 template <class FUNCTION, class FIRST, class... REST>
176 requires detail::is_zip_list_v<FIRST> &&
177 (detail::is_zip_list_v<REST> && ...)
178 auto invoke(this auto &&, FUNCTION &&function, FIRST &&first,
179 REST &&...rest) {
180 // The callable is moved into a local `callable` and then invoked as
181 // an lvalue once per lane, so that -- not `FUNCTION` -- is the
182 // category to detect in. Operands are deduced through forwarding
183 // references so that a caller offering an rvalue has each lane handed
184 // over instead of copied; see forward_zip_list_value_at for why a
185 // repeating operand is exempt.
186 using Result =
187 std::invoke_result_t<std::remove_cvref_t<FUNCTION> &,
188 detail::zip_list_value_ref_t<FIRST>,
189 detail::zip_list_value_ref_t<REST>...>;
190
191 using U = std::remove_cvref_t<Result>;
192 auto callable = std::forward<FUNCTION>(function);
193 const auto count = detail::zip_list_result_size(first, rest...);
194
195 if (!count.has_value()) {
196 // Every operand is infinite, so every operand is repeating, and
197 // forward_zip_list_value_at copies rather than moves for exactly
198 // that case. Spelling it the same way in both branches is what
199 // keeps this result the same type as the loop's.
200 return zip_list<U>::repeat(std::invoke(
201 callable,
202 detail::forward_zip_list_value_at(std::forward<FIRST>(first),
203 0),
204 detail::forward_zip_list_value_at(std::forward<REST>(rest),
205 0)...));
206 }
207
208 zip_list<U> result;
209 result.data.reserve(*count);
210
211 for (std::size_t index = 0; index < *count; ++index) {
212 result.data.push_back(std::invoke(
213 callable,
214 detail::forward_zip_list_value_at(std::forward<FIRST>(first),
215 index),
216 detail::forward_zip_list_value_at(std::forward<REST>(rest),
217 index)...));
218 }
219
220 return result;
221 }
222};
