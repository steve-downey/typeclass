31// The fixpoint expression tree: the simplest recursive datatype in the
32// library, used throughout the paper and tests to demonstrate the pattern.
33// Layout is types -> functor_typeclass instance -> smart constructors ->
34// algebras: the functor_typeclass partial specialization must be visible
35// before any function in this header that dispatches through the lookup
36// verbs for ExprF layers.
37
38// ---------------------------------------------------------------------
39// The non-recursive base functor.
40// ---------------------------------------------------------------------
41
42// 085bb189-a48e-4262-aefd-b64f8755e959
47template <typename A>
48struct Const {
49 int val;
50};
51
54template <typename A>
55struct Add {
56 child_slot_t<A> left, right;
57};
58
61template <typename A>
62struct Mul {
63 child_slot_t<A> left, right;
64};

70template <typename A>
71using ExprF = std::variant<Const<A>, Add<A>, Mul<A> >;
72// 085bb189-a48e-4262-aefd-b64f8755e959 end

74// ---------------------------------------------------------------------
75// Functor instance.
76// ---------------------------------------------------------------------
77
78// a052ddcb-f05b-41c9-85bf-4c443ab438b7
83template <typename A>
84struct ExprFFunctorImpl {
85 template <typename Fn>
86 constexpr auto fmap(this auto&&, Fn&& fn, const ExprF<A>& layer) {
87 using B = std::remove_cvref_t<std::invoke_result_t<Fn, const A&> >;
88 return std::visit(
89 transpose::overloaded{
90 [](const Const<A>& c) -> ExprF<B> { return Const<B>{c.val}; },
91 [&fn](const Add<A>& a) -> ExprF<B> {
92 return Add<B>{make_slot<B>(std::invoke(fn, *a.left)), make_slot<B>(std::invoke(fn, *a.right))};
93 },
94 [&fn](const Mul<A>& m) -> ExprF<B> {
95 return Mul<B>{make_slot<B>(std::invoke(fn, *m.left)), make_slot<B>(std::invoke(fn, *m.right))};
96 },
97 },
98 layer);
99 }
100};
101
104template <typename A>
105struct ExprFFunctorMap : transpose::derive_functor<ExprFFunctorImpl<A> > {
106 using ExprFFunctorImpl<A>::fmap;
107};
