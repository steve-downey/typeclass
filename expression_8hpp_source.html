32// The fixpoint expression tree: the simplest recursive datatype in the
33// library, used throughout the paper and tests to demonstrate the pattern.
34// Layout is types -> functor_typeclass instance -> smart constructors ->
35// algebras: the functor_typeclass partial specialization must be visible
36// before any function in this header that dispatches through the lookup
37// verbs for ExprF layers.
38
39// ---------------------------------------------------------------------
40// The non-recursive base functor.
41// ---------------------------------------------------------------------
42
43// 085bb189-a48e-4262-aefd-b64f8755e959
48template <typename A>
49struct Const {
50 int val;
51};
52
55template <typename A>
56struct Add {
57 child_slot_t<A> left, right;
58};
59
62template <typename A>
63struct Mul {
64 child_slot_t<A> left, right;
65};

71template <typename A>
72using ExprF = std::variant<Const<A>, Add<A>, Mul<A> >;

76// Functor instance.
77// ---------------------------------------------------------------------

79// a052ddcb-f05b-41c9-85bf-4c443ab438b7
84template <typename A>
85struct ExprFFunctorImpl {
86 template <typename Fn>
87 constexpr auto fmap(this auto&&, Fn&& fn, const ExprF<A>& layer)
88 requires std::invocable<Fn&, const A&> &&
89 transpose::detail::holdable_value<std::invoke_result_t<Fn&, const A&> > &&
90 std::constructible_from<std::remove_cvref_t<std::invoke_result_t<Fn&, const A&> >,
91 std::invoke_result_t<Fn&, const A&> > &&
92 std::move_constructible<std::remove_cvref_t<std::invoke_result_t<Fn&, const A&> > >
93 {
94 using B = std::remove_cvref_t<std::invoke_result_t<Fn&, const A&> >;
95 return std::visit(
96 transpose::overloaded{
97 [](const Const<A>& c) -> ExprF<B> { return Const<B>{c.val}; },
98 [&fn](const Add<A>& a) -> ExprF<B> {
99 return Add<B>{make_slot<B>(std::invoke(fn, *a.left)), make_slot<B>(std::invoke(fn, *a.right))};
100 },
101 [&fn](const Mul<A>& m) -> ExprF<B> {
102 return Mul<B>{make_slot<B>(std::invoke(fn, *m.left)), make_slot<B>(std::invoke(fn, *m.right))};
103 },
104 },
105 layer);
106 }
107};
108
111template <typename A>
112struct ExprFFunctorMap : transpose::derive_functor<ExprFFunctorImpl<A> > {
113 using ExprFFunctorImpl<A>::fmap;
114};
