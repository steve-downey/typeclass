63template <class VALUE_TYPE>
64struct optional_applicative_impl {

101template <class VALUE_TYPE>
102struct OptionalFunctorImpl {
103 // Constrained, not left to the body: the return type is deduced, so a
104 // probe would instantiate the body and fail hard on a function that
105 // cannot take the element, or one whose result an optional cannot hold
106 // (a function returning void, say). The Map re-exposes this member, which
107 // hides derive_functor::fmap, so its constraint is the one callers see.
108 template <class F>
109 requires std::invocable<F, const VALUE_TYPE &> &&
110 detail::optional_holdable_value<
111 std::invoke_result_t<F, const VALUE_TYPE &>>
112 constexpr auto fmap(this auto &&, F &&function,
113 const std::optional<VALUE_TYPE> &value) {
114 using Result = std::invoke_result_t<F, const VALUE_TYPE &>;
115 if (!value) {
116 return std::optional<std::remove_cvref_t<Result>>{};
117 }
118 return std::optional<std::remove_cvref_t<Result>>{
119 std::invoke(std::forward<F>(function), *value)};
120 }
121};
122
124template <class VALUE_TYPE>
125struct OptionalFunctorMap : derive_functor<OptionalFunctorImpl<VALUE_TYPE>> {
126 using OptionalFunctorImpl<VALUE_TYPE>::fmap;
127};

130template <class VALUE_TYPE>

138template <class VALUE_TYPE>
139struct OptionalMonadImpl {
140 using element_type = VALUE_TYPE;
141
142 template <class VALUE>
143 constexpr auto pure(this auto &&, VALUE &&value)
144 -> std::optional<std::remove_cvref_t<VALUE>> {
145 return applicative<std::optional<VALUE_TYPE>>.pure(
146 std::forward<VALUE>(value));
147 }
148
149 // The continuation must return an optional: a -> m b, not a -> b. A
150 // plain result would be returned as-is, with a value-initialized one
151 // standing in for the empty case, which breaks the monad laws.
152 template <class A, class F>
153 constexpr auto bind(this auto &&, const std::optional<A> &ma, F &&f)
154 -> std::remove_cvref_t<std::invoke_result_t<F, const A &>>
155 requires detail::is_optional_v<std::invoke_result_t<F, const A &>>
156 {
157 using Result = std::remove_cvref_t<std::invoke_result_t<F, const A &>>;
158 if (!ma)
159 return Result{};
160 return Result{std::invoke(std::forward<F>(f), *ma)};
161 }
162};
163
165template <class VALUE_TYPE>
166struct OptionalMonadMap : derive_monad<OptionalMonadImpl<VALUE_TYPE>> {
167 using OptionalMonadImpl<VALUE_TYPE>::bind;
168 using OptionalMonadImpl<VALUE_TYPE>::pure;
169};

172template <class VALUE_TYPE>
173inline constexpr auto monad<std::optional<VALUE_TYPE>> =
174 OptionalMonadMap<VALUE_TYPE>{};

196template <class VALUE_TYPE>
197template <class FUNCTION, class FIRST, class... REST>
198 requires detail::is_optional_v<FIRST> &&
199 (detail::is_optional_v<REST> && ...) &&
200 std::invocable<FUNCTION &, detail::contained_ref_t<FIRST>,
201 detail::contained_ref_t<REST>...> &&
202 detail::optional_holdable_value<std::invoke_result_t<
203 FUNCTION &, detail::contained_ref_t<FIRST>,
204 detail::contained_ref_t<REST>...>>
205constexpr auto
206optional_applicative_impl<VALUE_TYPE>::invoke(this auto &&, FUNCTION &&function,
207 FIRST &&first, REST &&...rest)
208 -> std::optional<std::remove_cvref_t<
209 std::invoke_result_t<FUNCTION &, detail::contained_ref_t<FIRST>,
210 detail::contained_ref_t<REST>...>>> {
211 using Result = std::remove_cvref_t<
212 std::invoke_result_t<FUNCTION &, detail::contained_ref_t<FIRST>,
213 detail::contained_ref_t<REST>...>>;
214 if (first.has_value() && (... && rest.has_value())) {
215 return std::optional<Result>{std::invoke(
216 function, detail::forward_contained(std::forward<FIRST>(first)),
217 detail::forward_contained(std::forward<REST>(rest))...)};
218 }
219 return std::optional<Result>{};
220}
