34// Adapting an existing tree you don't own.
35//
36// BinaryTree<T> below is a faithful port of a pre-existing persistent
37// binary tree (smd::tree::BinaryTree): value at every node, children held
38// by shared_ptr so copies are cheap and subtrees are structurally shared.
39// That representation is part of its identity — and it is not Fix-based,
40// so the recursion verbs cannot run over it directly.
41//
42// Design choice: rather than rewrite the tree, we adapt it. BinaryTreeF
43// is the base functor describing one layer of a value-at-every-node
44// binary tree (nullable child slots mirror the tree's nullable
45// shared_ptr children; see child_slot.hpp — Box at the knot, inline
46// storage at complete types), with a Functor instance registered for it, and
47// to_fix/from_fix convert between the shared_ptr representation and
48// Fix form. The same fold_fix/unfold_fix/refold that run over ExprF run
49// unchanged over the converted tree; only the adapter knows how the tree
50// is really stored. The conversions copy — an honest cost, consistent
51// with the value-semantics stance of Decision 6.
52//
53// BinaryTree itself is not constexpr-capable (shared_ptr allocation is a
54// runtime affair in C++23); the BinaryTreeF/Fix side is fully constexpr.
55
56// ---------------------------------------------------------------------
57// The existing tree, ported as-is.
58// ---------------------------------------------------------------------

173// 91fef612-39b5-4424-8ab5-7d8c80997e2c
179template <typename T, typename A>
180struct BinaryTreeFFunctorImpl {
181 template <typename Fn>
182 constexpr auto fmap(this auto&&, Fn&& fn, const BinaryTreeF<T, A>& layer) {
183 using B = std::remove_cvref_t<std::invoke_result_t<Fn, const A&> >;
184 return BinaryTreeF<T, B>{
185 layer.value,
186 layer.left ? make_slot<B>(std::invoke(fn, *layer.left)) : child_slot_t<B>{},
187 layer.right ? make_slot<B>(std::invoke(fn, *layer.right)) : child_slot_t<B>{},
188 };
189 }
190};

194template <typename T, typename A>
195struct BinaryTreeFFunctorMap : transpose::derive_functor<BinaryTreeFFunctorImpl<T, A> > {
196 using BinaryTreeFFunctorImpl<T, A>::fmap;
197};

306 layer.left ? BinaryTree<T>::make_ptr(*layer.left) : nullptr,

308 };
