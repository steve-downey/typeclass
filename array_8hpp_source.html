6// Applicative instance for std::array<T, N> with positional (zip) semantics,
7// plus a tuple-specific transpose for the SoA <-> AoS transformation:
8//
9// tuple<array<T1, N>, array<T2, N>, ...> -> array<tuple<T1, T2, ...>, N>
10//
11// The array applicative applies functions lane-by-lane (like simd_lanes and
12// zip_list) but over a standard fixed-size container. pure(x) broadcasts x
13// to all N positions.
14//
15// The tuple transpose is a heterogeneous traversal: std::tuple cannot use the
16// homogeneous Traversable loop, so transpose_tuple uses a fold over indices
17// to combine the per-field arrays through the array applicative's invoke.
18//
19// This header stands in for <array>, which in the proposal declares
20// the instance itself (docs/transpose/decisions.md#instance-placement). Include
21// it, not only <array>, before the first use of the applicative on a
22// std::array, in every translation unit that makes one: a use that precedes the
23// instance is ill-formed, no diagnostic required. beman/transpose.hpp includes
24// it.

66template <class T, std::size_t N>
67struct array_applicative_impl {

69 template <class VALUE>

99template <std::size_t N, class... Ts, std::size_t... Is>
100auto transpose_tuple_impl(const std::tuple<std::array<Ts, N>...> &soa,
101 std::index_sequence<Is...>)
102 -> std::array<std::tuple<Ts...>, N> {
103 const auto &app = applicative<std::array<std::tuple<Ts...>, N>>;
104 return app.invoke([](const Ts &...elems) { return std::tuple{elems...}; },
105 std::get<Is>(soa)...);

114template <class T, std::size_t N>
115template <class VALUE>
116auto array_applicative_impl<T, N>::pure(this auto &&, VALUE &&value) {
117 using U = std::remove_cvref_t<VALUE>;
118 // Constructed directly rather than default-constructed and filled: every
119 // element is a copy of `value`, which is all this operation documents,
120 // and filling would additionally require U to be default-constructible
121 // and assignable.
122 return detail::make_array<U, N>(
123 [&value](std::size_t) -> U { return U(value); },
124 std::make_index_sequence<N>{});
125}

137template <class T, std::size_t N>
138template <class FUNCTION, class FIRST, class... REST>
139 requires detail::is_std_array_of_extent_v<FIRST, N> &&
140 (detail::is_std_array_of_extent_v<REST, N> && ...) &&
141 std::invocable<FUNCTION &, detail::element_ref_t<FIRST>,
142 detail::element_ref_t<REST>...> &&
143 detail::holdable_value<
144 std::invoke_result_t<FUNCTION &, detail::element_ref_t<FIRST>,
145 detail::element_ref_t<REST>...>>
146auto array_applicative_impl<T, N>::invoke(this auto &&, FUNCTION &&function,
147 FIRST &&first, REST &&...rest) {
148 using Result =
149 std::invoke_result_t<FUNCTION &, detail::element_ref_t<FIRST>,
150 detail::element_ref_t<REST>...>;
151 using U = std::remove_cvref_t<Result>;
152
153 // Constructed directly rather than default-constructed and assigned into:
154 // the operation documents only that each result element is the result of
155 // one invocation, so U need not be default-constructible or assignable.
156 return detail::make_array<U, N>(
157 [&](std::size_t index) -> U {
158 return std::invoke(
159 function, detail::forward_at(std::forward<FIRST>(first), index),
160 detail::forward_at(std::forward<REST>(rest), index)...);
161 },
162 std::make_index_sequence<N>{});
163}
