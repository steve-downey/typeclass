23// Currying support for the invoke-from-ap derivation: wraps a plain n-ary
24// function so it can be fed through one-step contextual application,
25// consuming one argument at a time and invoking once all ARITY arguments
26// are bound. This cost exists only in the derivation direction -- C++ is
27// not natively curried -- and never at invoke-native call sites.
28//
29// The arity is fixed when the partial is made, from the number of
30// arguments the caller of invoke passed, and never inferred from what the
31// function will accept. Inferring it -- invoke as soon as the function
32// accepts the bound arguments, otherwise keep binding -- turns every
33// failure into another bound argument: a function that cannot take the
34// arguments, or is given too few, yields a half-applied partial in the
35// context instead of failing. The saturating call is constrained and
36// spells its result type, so a function that cannot take the arguments
37// makes the context's ap fail substitution, and with it the derived
38// invoke's constraint (ap_chain_valid below).
39//
40// Every call copies what it binds rather than moving it: the context's ap
41// may call one held partial many times (once per element of a list), and
42// each call must see the same bound arguments.
43template <std::size_t ARITY, class FUNCTION, class... BOUND_ARGS>
44struct terminating_partial {
45 static_assert(sizeof...(BOUND_ARGS) < ARITY);
46
47 FUNCTION function;
48 std::tuple<BOUND_ARGS...> bound_args;
49
50 template <class NEXT_ARG>
51 requires(sizeof...(BOUND_ARGS) + 1 == ARITY) &&
52 std::invocable<FUNCTION &, BOUND_ARGS &..., NEXT_ARG>
53 constexpr auto operator()(NEXT_ARG &&next_arg)
54 -> std::invoke_result_t<FUNCTION &, BOUND_ARGS &..., NEXT_ARG> {
55 return std::apply(
56 [&](BOUND_ARGS &...bound) -> decltype(auto) {
57 return std::invoke(function, bound...,
58 std::forward<NEXT_ARG>(next_arg));
59 },
60 bound_args);
61 }
62
63 template <class NEXT_ARG>
64 requires(sizeof...(BOUND_ARGS) + 1 == ARITY) &&
65 std::invocable<const FUNCTION &, const BOUND_ARGS &...,
66 NEXT_ARG>
67 constexpr auto operator()(NEXT_ARG &&next_arg) const
68 -> std::invoke_result_t<const FUNCTION &, const BOUND_ARGS &...,
69 NEXT_ARG> {
70 return std::apply(
71 [&](const BOUND_ARGS &...bound) -> decltype(auto) {
72 return std::invoke(function, bound...,
73 std::forward<NEXT_ARG>(next_arg));
74 },
75 bound_args);
76 }
77
78 template <class NEXT_ARG>
79 requires(sizeof...(BOUND_ARGS) + 1 < ARITY)
80 constexpr auto operator()(NEXT_ARG &&next_arg) const
81 -> terminating_partial<ARITY, FUNCTION, BOUND_ARGS...,
82 std::remove_cvref_t<NEXT_ARG>> {
83 return {function,
84 std::tuple_cat(bound_args,
85 std::tuple<std::remove_cvref_t<NEXT_ARG>>{
86 std::forward<NEXT_ARG>(next_arg)})};
87 }
88};
89
90template <std::size_t ARITY, class FUNCTION>
91constexpr auto make_terminating_partial(FUNCTION &&function)
92 -> terminating_partial<ARITY, std::remove_cvref_t<FUNCTION>> {
93 return {std::forward<FUNCTION>(function), std::tuple<>{}};
94}

96// Whether ap_chain can fold ACCUMULATED through every argument: each step
97// self.ap(accumulated, argument) is well-formed, and its result is the
98// next step's accumulated value. The trait only names each step's type, so
99// it relies on the bases' ap being SFINAE-friendly -- the same rule the
100// class invariant already imposes on every Impl.
101template <class SELF, class ACCUMULATED, class... ARGUMENTS>
102struct ap_chain_valid : std::false_type {};
103
104template <class SELF, class ACCUMULATED>
105struct ap_chain_valid<SELF, ACCUMULATED> : std::true_type {};
106
107template <class SELF, class ACCUMULATED, class FIRST, class... REST>
108 requires requires {
109 std::declval<SELF>().ap(std::declval<ACCUMULATED>(),
110 std::declval<FIRST>());
111 }
112struct ap_chain_valid<SELF, ACCUMULATED, FIRST, REST...>
113 : ap_chain_valid<SELF,
114 decltype(std::declval<SELF>().ap(
115 std::declval<ACCUMULATED>(), std::declval<FIRST>())),
116 REST...> {};

118// Whether the invoke-from-ap derivation of SELF.invoke(function,
119// arguments...) is well-formed: pure can embed the curried function, and
120// ap can then apply it to every argument in turn, the last application
121// invoking it.
123template <class SELF, class FUNCTION, class... ARGUMENTS>
124concept invocable_through_ap =
125 ap_chain_valid<SELF,
126 decltype(std::declval<SELF>().pure(
127 make_terminating_partial<sizeof...(ARGUMENTS)>(
128 std::declval<FUNCTION>()))),
129 ARGUMENTS...>::value;

131// The evaluator that recovers one-step application (ap) from the n-ary
132// core: ap(cf, cx) = invoke(applicative_eval, cf, cx) -- GHC's
133// (<*>) = liftA2 id. The trailing return type keeps it SFINAE-visible so
134// the derived ap's constraint fails cleanly -- never hard-errors -- for
135// contexts that cannot hold callables.
136struct applicative_eval_t {
137 template <class FUNCTION, class ARGUMENT>
138 constexpr auto operator()(FUNCTION &&function, ARGUMENT &&argument) const
139 -> std::invoke_result_t<FUNCTION, ARGUMENT> {
140 return std::invoke(std::forward<FUNCTION>(function),
141 std::forward<ARGUMENT>(argument));
142 }
143};
144inline constexpr applicative_eval_t applicative_eval{};
145
146// Free-function form for generic internal use.
147template <class APPLICATIVE_MAP, class FUNCTIONS_IN_CONTEXT,
148 class ARGUMENTS_IN_CONTEXT>
149constexpr auto ap(const APPLICATIVE_MAP &map, FUNCTIONS_IN_CONTEXT &&functions,
150 ARGUMENTS_IN_CONTEXT &&arguments) {
151 return map.invoke(applicative_eval,
152 std::forward<FUNCTIONS_IN_CONTEXT>(functions),
153 std::forward<ARGUMENTS_IN_CONTEXT>(arguments));
154}
